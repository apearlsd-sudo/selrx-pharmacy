   148→   148→    nearExpiry: true,
   149→   149→    variance: true,
   150→   150→    reorder: true,

   152→   152→  })

   155→   155→  const addToast = useAppStore((s) => s.addToast)
   156→   156→
   157→   157→  const fetchReport = useCallback(async () => {
   158→   158→    if (!stockTakeId) return

   160→   160→    setError(null)
   161→   161→    try {
   162→   162→      const res = await fetch(`/api/stock-take?action=report&id=${stockTakeId}`, { headers: authHeaders() })
   163→   163→      if (res.ok) {
   164→   164→        setReport(await res.json())
   165→   165→      } else {
   166→   166→        const err = await res.json().catch(() => ({ error: 'Failed to load report' }))
   167→   167→        setError(err.error || 'Failed to load report')
   168→   168→      }
   169→   169→    } catch {
   170→   170→      setError('Network error while loading report')
   171→   171→    } finally {
   172→   172→      setLoading(false)
   173→   173→    }
   174→   174→  }, [stockTakeId])
   175→   175→
   176→   176→  useEffect(() => { fetchReport() }, [fetchReport])

   179→   179→    setExpandedSections((prev) => ({ ...prev, [section]: !prev[section] }))
   180→   180→  }

   182→   182→  const handlePrint = () => window.print()

   188→   188→      lines.push('=== EXPIRED GOODS ===')

   191→   191→        lines.push(`"${item.productName}","${item.ndc || ''}","${item.category}","${item.strength || ''}","${item.dosageForm || ''}",${item.countedQty},${item.costPrice},${item.sellingPrice},${item.totalCost},${item.potentialRevenue},"${item.expiryDate || ''}",${item.daysSinceExpiry},"${item.manufacturer || ''}","${item.vendor || ''}"`)
   192→   192→      }
   193→   193→      lines.push('')
   194→   194→      // Near expiry
   195→   195→      lines.push('=== NEAR EXPIRY (90 DAYS) ===')
   196→   196→      lines.push('Product,NDC,Category,Strength,Form,Qty,Cost Price,Selling Price,Total Cost,Potential Revenue,Expiry Date,Days To Expiry,Manufacturer,Vendor')

   198→   198→        lines.push(`"${item.productName}","${item.ndc || ''}","${item.category}","${item.strength || ''}","${item.dosageForm || ''}",${item.countedQty},${item.costPrice},${item.sellingPrice},${item.totalCost},${item.potentialRevenue},"${item.expiryDate || ''}",${item.daysToExpiry},"${item.manufacturer || ''}","${item.vendor || ''}"`)
   199→   199→      }
   200→   200→      lines.push('')
   201→   201→      // Variance
   202→   202→      lines.push('=== STOCK VARIANCE ===')

   206→   206→      }

   232→   232→        <Skeleton className="h-8 w-64" />

   236→   236→      </div>

   238→   238→  }

   252→   252→

   288→   288→  return (

   294→   294→            <ArrowLeft className="h-4 w-4 mr-1" /> Back
   295→   295→          </Button>
   296→   296→          <div className="h-10 w-10 rounded-lg bg-emerald-100 flex items-center justify-center">
