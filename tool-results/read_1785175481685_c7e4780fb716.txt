     1→'use client'
     2→
     3→import { useState, useEffect, useCallback } from 'react'
     4→import {
     5→  Package, Search, Plus, AlertTriangle, Edit, ArrowUpDown,
     6→  Download, Filter, TrendingUp, X, PackagePlus, Tags, Trash2
     7→} from 'lucide-react'
     8→import { Button } from '@/components/ui/button'
     9→import { Input } from '@/components/ui/input'
    10→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
    11→import { Badge } from '@/components/ui/badge'
    12→import { Skeleton } from '@/components/ui/skeleton'
    13→import {
    14→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
    15→} from '@/components/ui/table'
    16→import {
    17→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter,
    18→} from '@/components/ui/dialog'
    19→import { Label } from '@/components/ui/label'
    20→import { Textarea } from '@/components/ui/textarea'
    21→import {
    22→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    23→} from '@/components/ui/select'
    24→import { useAppStore } from '@/store/app-store'
    25→
    26→interface InventoryItem {
    27→  id: string
    28→  productId: string
    29→  quantity: number
    30→  lastCounted: string | null
    31→  product: {
    32→    id: string
    33→    name: string
    34→    ndc: string | null
    35→    category: string
    36→    sellingPrice: number
    37→    costPrice: number | null
    38→    status: string
    39→    reorderPoint: number
    40→    maxStock: number | null
    41→    dosageForm: string | null
    42→    strength: string | null
    43→    unitOfMeasure: string
    44→    expiryDate: string | null
    45→    batchNumber: string | null
    46→  }
    47→}
    48→
    49→export function InventoryView() {
    50→  const [items, setItems] = useState<InventoryItem[]>([])
    51→  const [loading, setLoading] = useState(true)
    52→  const [searchQuery, setSearchQuery] = useState('')
    53→  const [categoryFilter, setCategoryFilter] = useState('ALL')
    54→  const [stockFilter, setStockFilter] = useState('ALL')
    55→  const [adjustDialog, setAdjustDialog] = useState(false)
    56→  const [receiveDialog, setReceiveDialog] = useState(false)
    57→  const [addProductDialog, setAddProductDialog] = useState(false)
    58→  const [categoryDialog, setCategoryDialog] = useState(false)
    59→  const [categories, setCategories] = useState<{ id: string; name: string; description: string | null; _count?: { products: number } }[]>([])
    60→  const [newCategoryName, setNewCategoryName] = useState('')
    61→  const [newCategoryDesc, setNewCategoryDesc] = useState('')
    62→  const [savingCategory, setSavingCategory] = useState(false)
    63→  const [selectedItem, setSelectedItem] = useState<InventoryItem | null>(null)
    64→  const [adjustType, setAdjustType] = useState('ADD')
    65→  const [adjustAmount, setAdjustAmount] = useState('')
    66→  const [adjustReason, setAdjustReason] = useState('')
    67→  const [shipmentItems, setShipmentItems] = useState([{ productId: '', quantity: '', costPrice: '' }])
    68→  const [sortBy, setSortBy] = useState<'name' | 'stock' | 'category'>('name')
    69→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('asc')
    70→  const [productForm, setProductForm] = useState({
    71→    name: '', sku: '', category: 'OTC', price: '', stockQuantity: '',
    72→    minStockLevel: '10', expiryDate: '', barcode: '',
    73→  })
    74→  const [savingProduct, setSavingProduct] = useState(false)
    75→  const addToast = useAppStore((s) => s.addToast)
    76→  const currentUser = useAppStore((s) => s.user)
    77→
    78→  const fetchInventory = useCallback(async () => {
    79→    setLoading(true)
    80→    try {
    81→      const params = new URLSearchParams()
    82→      if (searchQuery) params.set('search', searchQuery)
    83→      if (categoryFilter !== 'ALL') params.set('category', categoryFilter)
    84→      const res = await fetch(`/api/inventory?${params}`)
    85→      if (res.ok) {
    86→        const data = await res.json()
    87→        setItems(data)
    88→      }
    89→    } catch {
    90→      addToast({ title: 'Error', description: 'Failed to load inventory', variant: 'destructive' })
    91→    } finally {
    92→      setLoading(false)
    93→    }
    94→  }, [searchQuery, categoryFilter, addToast])
    95→
    96→  const fetchCategories = useCallback(async () => {
    97→    try {
    98→      const res = await fetch('/api/categories')
    99→      if (res.ok) setCategories(await res.json())
   100→    } catch {
   101→      // silent — categories are optional enhancement
   102→    }
   103→  }, [])
   104→
   105→  useEffect(() => { fetchInventory(); fetchCategories() }, [fetchInventory, fetchCategories])
   106→
   107→  const filteredItems = items.filter((item) => {
   108→    if (stockFilter === 'LOW') return item.quantity <= item.product.reorderPoint
   109→    if (stockFilter === 'OUT') return item.quantity === 0
   110→    if (stockFilter === 'OK') return item.quantity > item.product.reorderPoint
   111→    return true
   112→  }).sort((a, b) => {
   113→    if (sortBy === 'name') return sortDir === 'asc' ? a.product.name.localeCompare(b.product.name) : b.product.name.localeCompare(a.product.name)
   114→    if (sortBy === 'stock') return sortDir === 'asc' ? a.quantity - b.quantity : b.quantity - a.quantity
   115→    return sortDir === 'asc' ? a.product.category.localeCompare(b.product.category) : b.product.category.localeCompare(a.product.category)
   116→  })
   117→
   118→  const lowStockCount = items.filter((i) => i.quantity <= i.product.reorderPoint).length
   119→  const outOfStockCount = items.filter((i) => i.quantity === 0).length
   120→  const totalValue = items.reduce((sum, i) => sum + i.quantity * (i.product.costPrice || i.product.sellingPrice), 0)
   121→
   122→  const handleAdjust = async () => {
   123→    if (!selectedItem || !adjustAmount || !adjustReason) return
   124→    try {
   125→      const adj = adjustType === 'ADD' ? parseInt(adjustAmount) : adjustType === 'REMOVE' ? -parseInt(adjustAmount) : parseInt(adjustAmount)
   126→      await fetch('/api/inventory', {
   127→        method: 'PUT',
   128→        headers: { 'Content-Type': 'application/json' },
   129→        body: JSON.stringify({ productId: selectedItem.productId, adjustment: adj, reason: adjustReason }),
   130→      })
   131→      addToast({ title: 'Stock Updated', description: `${selectedItem.product.name} adjusted`, variant: 'success' })
   132→      setAdjustDialog(false)
   133→      setAdjustAmount('')
   134→      setAdjustReason('')
   135→      fetchInventory()
   136→    } catch {
   137→      addToast({ title: 'Error', description: 'Failed to adjust stock', variant: 'destructive' })
   138→    }
   139→  }
   140→
   141→  const handleDeleteCategory = async (catId: string, catName: string) => {
   142→    try {
   143→      const res = await fetch(`/api/categories/${catId}`, { method: 'DELETE' })
   144→      if (!res.ok) {
   145→        const err = await res.json()
   146→        throw new Error(err.error || 'Failed to delete category')
   147→      }
   148→      addToast({ title: 'Deleted', description: `Category "${catName}" removed`, variant: 'success' })
   149→      fetchCategories()
   150→    } catch (err: any) {
   151→      addToast({ title: 'Error', description: err.message, variant: 'destructive' })
   152→    }
   153→  }
   154→
   155→  const handleAddProduct = async () => {
   156→    if (!productForm.name || !productForm.price) return
   157→    setSavingProduct(true)
   158→    try {
   159→      const res = await fetch('/api/products', {
   160→        method: 'POST',
   161→        headers: {
   162→          'Content-Type': 'application/json',
   163→          'x-user-role': currentUser?.role || 'SUPER_ADMIN',
   164→        },
   165→        body: JSON.stringify({
   166→          name: productForm.name,
   167→          ndc: productForm.sku || null,
   168→          category: productForm.category,
   169→          sellingPrice: parseFloat(productForm.price),
   170→          costPrice: parseFloat(productForm.price) * 0.7,
   171→          reorderPoint: parseInt(productForm.minStockLevel) || 10,
   172→          expiryDate: productForm.expiryDate || null,
   173→          batchNumber: productForm.barcode || null,
   174→        }),
   175→      })
   176→      if (!res.ok) {
   177→        const err = await res.json()
   178→        throw new Error(err.error || 'Failed to create product')
   179→      }
   180→      const newProduct = await res.json()
   181→
   182→      // Set initial stock quantity if specified
   183→      const qty = parseInt(productForm.stockQuantity)
   184→      if (qty > 0) {
   185→        await fetch('/api/inventory', {
   186→          method: 'PUT',
   187→          headers: { 'Content-Type': 'application/json' },
   188→          body: JSON.stringify({
   189→            productId: newProduct.id,
   190→            adjustment: qty,
   191→            reason: 'Initial stock on product creation',
   192→          }),
   193→        })
   194→      }
   195→
   196→      addToast({ title: 'Product Added', description: `${productForm.name} has been added to inventory`, variant: 'success' })
   197→      setAddProductDialog(false)
   198→      setProductForm({ name: '', sku: '', category: 'OTC', price: '', stockQuantity: '', minStockLevel: '10', expiryDate: '', barcode: '' })
   199→      fetchInventory()
   200→    } catch (err: any) {
   201→      addToast({ title: 'Error', description: err.message || 'Failed to add product', variant: 'destructive' })
   202→    } finally {
   203→      setSavingProduct(false)
   204→    }
   205→  }
   206→
   207→  const handleReceiveShipment = async () => {
   208→    const validItems = shipmentItems.filter((i) => i.productId && i.quantity)
   209→    if (validItems.length === 0) return
   210→    try {
   211→      await fetch('/api/inventory', {
   212→        method: 'POST',
   213→        headers: { 'Content-Type': 'application/json' },
   214→        body: JSON.stringify({
   215→          items: validItems.map((i) => ({
   216→            productId: i.productId,
   217→            quantity: parseInt(i.quantity),
   218→            costPrice: parseFloat(i.costPrice) || 0,
   219→          })),
   220→        }),
   221→      })
   222→      addToast({ title: 'Shipment Received', description: `${validItems.length} items received`, variant: 'success' })
   223→      setReceiveDialog(false)
   224→      setShipmentItems([{ productId: '', quantity: '', costPrice: '' }])
   225→      fetchInventory()
   226→    } catch {
   227→      addToast({ title: 'Error', description: 'Failed to receive shipment', variant: 'destructive' })
   228→    }
   229→  }
   230→
   231→  return (
   232→    <div className="space-y-4">
   233→      {/* Stats Cards */}
   234→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   235→        <Card>
   236→          <CardContent className="p-4 flex items-center gap-3">
   237→            <div className="h-10 w-10 rounded-lg bg-emerald-100 flex items-center justify-center">
   238→              <Package className="h-5 w-5 text-emerald-600" />
   239→            </div>
   240→            <div>
   241→              <p className="text-2xl font-bold">{items.length}</p>
   242→              <p className="text-xs text-muted-foreground">Total Products</p>
   243→            </div>
   244→          </CardContent>
   245→        </Card>
   246→        <Card>
   247→          <CardContent className="p-4 flex items-center gap-3">
   248→            <div className="h-10 w-10 rounded-lg bg-red-100 flex items-center justify-center">
   249→              <AlertTriangle className="h-5 w-5 text-red-600" />
   250→            </div>
   251→            <div>
   252→              <p className="text-2xl font-bold text-red-600">{lowStockCount}</p>
   253→              <p className="text-xs text-muted-foreground">Low Stock Alerts</p>
   254→            </div>
   255→          </CardContent>
   256→        </Card>
   257→        <Card>
   258→          <CardContent className="p-4 flex items-center gap-3">
   259→            <div className="h-10 w-10 rounded-lg bg-amber-100 flex items-center justify-center">
   260→              <AlertTriangle className="h-5 w-5 text-amber-600" />
   261→            </div>
   262→            <div>
   263→              <p className="text-2xl font-bold text-amber-600">{outOfStockCount}</p>
   264→              <p className="text-xs text-muted-foreground">Out of Stock</p>
   265→            </div>
   266→          </CardContent>
   267→        </Card>
   268→        <Card>
   269→          <CardContent className="p-4 flex items-center gap-3">
   270→            <div className="h-10 w-10 rounded-lg bg-teal-100 flex items-center justify-center">
   271→              <TrendingUp className="h-5 w-5 text-teal-600" />
   272→            </div>
   273→            <div>
   274→              <p className="text-2xl font-bold">${totalValue.toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 2 })}</p>
   275→              <p className="text-xs text-muted-foreground">Inventory Value</p>
   276→            </div>
   277→          </CardContent>
   278→        </Card>
   279→      </div>
   280→
   281→      {/* Filters & Actions */}
   282→      <Card>
   283→        <CardContent className="p-4">
   284→          <div className="flex flex-col sm:flex-row gap-3">
   285→            <div className="relative flex-1">
   286→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   287→              <Input
   288→                placeholder="Search by product name or NDC..."
   289→                value={searchQuery}
   290→                onChange={(e) => setSearchQuery(e.target.value)}
   291→                className="pl-9"
   292→              />
   293→            </div>
   294→            <Select value={categoryFilter} onValueChange={setCategoryFilter}>
   295→              <SelectTrigger className="w-full sm:w-40">
   296→                <Filter className="h-4 w-4 mr-2" />
   297→                <SelectValue />
   298→              </SelectTrigger>
   299→              <SelectContent>
   300→                <SelectItem value="ALL">All Categories</SelectItem>
   301→                <SelectItem value="OTC">OTC</SelectItem>
   302→                <SelectItem value="PRESCRIPTION">Prescription</SelectItem>
   303→                <SelectItem value="SUPPLEMENT">Supplement</SelectItem>
   304→                <SelectItem value="MEDICAL_DEVICE">Medical Device</SelectItem>
   305→                <SelectItem value="PERSONAL_CARE">Personal Care</SelectItem>
   306→                <SelectItem value="CONSUMABLES">Consumables</SelectItem>
   307→              </SelectContent>
   308→            </Select>
   309→            <Select value={stockFilter} onValueChange={setStockFilter}>
   310→              <SelectTrigger className="w-full sm:w-36">
   311→                <SelectValue />
   312→              </SelectTrigger>
   313→              <SelectContent>
   314→                <SelectItem value="ALL">All Stock</SelectItem>
   315→                <SelectItem value="LOW">Low Stock</SelectItem>
   316→                <SelectItem value="OUT">Out of Stock</SelectItem>
   317→                <SelectItem value="OK">In Stock OK</SelectItem>
   318→              </SelectContent>
   319→            </Select>
   320→            <Button onClick={() => setReceiveDialog(true)} className="bg-emerald-600 hover:bg-emerald-700">
   321→              <Plus className="h-4 w-4 mr-2" />
   322→              Receive Shipment
   323→            </Button>
   324→            <Button onClick={() => setAddProductDialog(true)} className="bg-teal-600 hover:bg-teal-700">
   325→              <PackagePlus className="h-4 w-4 mr-2" />
   326→              Add Product
   327→            </Button>
   328→            <Button onClick={() => setCategoryDialog(true)} variant="outline" className="border-emerald-300 text-emerald-700 hover:bg-emerald-50">
   329→              <Tags className="h-4 w-4 mr-2" />
   330→              Category
   331→            </Button>
   332→          </div>
   333→        </CardContent>
   334→      </Card>
   335→
   336→      {/* Inventory Table */}
   337→      <Card>
   338→        <CardContent className="p-0">
   339→          <Table>
   340→            <TableHeader>
   341→              <TableRow>
   342→                <TableHead className="cursor-pointer" onClick={() => { setSortBy('name'); setSortDir(sortDir === 'asc' ? 'desc' : 'asc') }}>
   343→                  <span className="flex items-center gap-1">Product <ArrowUpDown className="h-3 w-3" /></span>
   344→                </TableHead>
   345→                <TableHead className="hidden md:table-cell">NDC</TableHead>
   346→                <TableHead className="cursor-pointer hidden sm:table-cell" onClick={() => { setSortBy('category'); setSortDir(sortDir === 'asc' ? 'desc' : 'asc') }}>
   347→                  <span className="flex items-center gap-1">Category <ArrowUpDown className="h-3 w-3" /></span>
   348→                </TableHead>
   349→                <TableHead className="cursor-pointer text-right" onClick={() => { setSortBy('stock'); setSortDir(sortDir === 'asc' ? 'desc' : 'asc') }}>
   350→                  <span className="flex items-center justify-end gap-1">In Stock <ArrowUpDown className="h-3 w-3" /></span>
   351→                </TableHead>
   352→                <TableHead className="hidden lg:table-cell text-right">Reorder Point</TableHead>
   353→                <TableHead className="text-right">Status</TableHead>
   354→                <TableHead className="hidden lg:table-cell text-right">Cost</TableHead>
   355→                <TableHead className="hidden lg:table-cell text-right">Retail</TableHead>
   356→                <TableHead className="text-right">Actions</TableHead>
   357→              </TableRow>
   358→            </TableHeader>
   359→            <TableBody>
   360→              {loading ? (
   361→                Array.from({ length: 8 }).map((_, i) => (
   362→                  <TableRow key={i}>
   363→                    {Array.from({ length: 9 }).map((_, j) => (
   364→                      <TableCell key={j}><Skeleton className="h-5 w-full" /></TableCell>
   365→                    ))}
   366→                  </TableRow>
   367→                ))
   368→              ) : filteredItems.length === 0 ? (
   369→                <TableRow>
   370→                  <TableCell colSpan={9} className="text-center py-8 text-muted-foreground">
   371→                    No inventory items found
   372→                  </TableCell>
   373→                </TableRow>
   374→              ) : (
   375→                filteredItems.map((item) => {
   376→                  const isLow = item.quantity <= item.product.reorderPoint
   377→                  const isOut = item.quantity === 0
   378→                  return (
   379→                    <TableRow key={item.id} className={isOut ? 'bg-red-50/50' : isLow ? 'bg-amber-50/50' : ''}>
   380→                      <TableCell>
   381→                        <div>
   382→                          <p className="font-medium text-sm">{item.product.name}</p>
   383→                          <p className="text-xs text-muted-foreground">
   384→                            {item.product.strength} {item.product.dosageForm} · {item.product.unitOfMeasure}
   385→                          </p>
   386→                        </div>
   387→                      </TableCell>
   388→                      <TableCell className="hidden md:table-cell text-xs font-mono">{item.product.ndc || '—'}</TableCell>
   389→                      <TableCell className="hidden sm:table-cell">
   390→                        <Badge variant="outline" className="text-xs">{item.product.category.replace(/_/g, ' ')}</Badge>
   391→                      </TableCell>
   392→                      <TableCell className="text-right font-bold">{item.quantity}</TableCell>
   393→                      <TableCell className="hidden lg:table-cell text-right text-muted-foreground">{item.product.reorderPoint}</TableCell>
   394→                      <TableCell className="text-right">
   395→                        {isOut ? (
   396→                          <Badge className="bg-red-100 text-red-700 border-red-200">Out of Stock</Badge>
   397→                        ) : isLow ? (
   398→                          <Badge className="bg-amber-100 text-amber-700 border-amber-200">Low Stock</Badge>
   399→                        ) : (
   400→                          <Badge className="bg-emerald-100 text-emerald-700 border-emerald-200">In Stock</Badge>
   401→                        )}
   402→                      </TableCell>
   403→                      <TableCell className="hidden lg:table-cell text-right">${item.product.costPrice?.toFixed(2) || '—'}</TableCell>
   404→                      <TableCell className="hidden lg:table-cell text-right">${item.product.sellingPrice.toFixed(2)}</TableCell>
   405→                      <TableCell className="text-right">
   406→                        <Button size="sm" variant="ghost" onClick={() => { setSelectedItem(item); setAdjustDialog(true) }}>
   407→                          <Edit className="h-3.5 w-3.5 mr-1" />
   408→                          Adjust
   409→                        </Button>
   410→                      </TableCell>
   411→                    </TableRow>
   412→                  )
   413→                })
   414→              )}
   415→            </TableBody>
   416→          </Table>
   417→        </CardContent>
   418→      </Card>
   419→
   420→      {/* Add Product Dialog */}
   421→      <Dialog open={addProductDialog} onOpenChange={setAddProductDialog}>
   422→        <DialogContent className="max-w-lg">
   423→          <DialogHeader>
   424→            <DialogTitle className="flex items-center gap-2">
   425→              <PackagePlus className="h-5 w-5 text-teal-600" />
   426→              Add New Product
   427→            </DialogTitle>
   428→          </DialogHeader>
   429→          <div className="grid grid-cols-2 gap-4 max-h-[70vh] overflow-y-auto pr-1">
   430→            {/* Product Name */}
   431→            <div className="col-span-2">
   432→              <Label htmlFor="prod-name">Product Name <span className="text-red-500">*</span></Label>
   433→              <Input
   434→                id="prod-name"
   435→                placeholder="e.g., Amoxicillin 500mg Capsules"
   436→                value={productForm.name}
   437→                onChange={(e) => setProductForm({ ...productForm, name: e.target.value })}
   438→                className="mt-1"
   439→              />
   440→            </div>
   441→
   442→            {/* SKU */}
   443→            <div>
   444→              <Label htmlFor="prod-sku">SKU</Label>
   445→              <Input
   446→                id="prod-sku"
   447→                placeholder="e.g., SKU-00123"
   448→                value={productForm.sku}
   449→                onChange={(e) => setProductForm({ ...productForm, sku: e.target.value })}
   450→                className="mt-1"
   451→              />
   452→            </div>
   453→
   454→            {/* Category */}
   455→            <div>
   456→              <Label htmlFor="prod-category">Category</Label>
   457→              <Select value={productForm.category} onValueChange={(v) => setProductForm({ ...productForm, category: v })}>
   458→                <SelectTrigger className="mt-1">
   459→                  <SelectValue />
   460→                </SelectTrigger>
   461→                <SelectContent>
   462→                  <SelectItem value="OTC">OTC</SelectItem>
   463→                  <SelectItem value="PRESCRIPTION">Prescription</SelectItem>
   464→                  <SelectItem value="SUPPLEMENT">Supplement</SelectItem>
   465→                  <SelectItem value="MEDICAL_DEVICE">Medical Device</SelectItem>
   466→                  <SelectItem value="PERSONAL_CARE">Personal Care</SelectItem>
   467→                  <SelectItem value="CONSUMABLES">Consumables</SelectItem>
   468→                  {categories.filter((c) => !['OTC','PRESCRIPTION','SUPPLEMENT','MEDICAL_DEVICE','PERSONAL_CARE','CONSUMABLES'].includes(c.name)).map((cat) => (
   469→                    <SelectItem key={cat.id} value={cat.name}>{cat.name.replace(/_/g, ' ')}</SelectItem>
   470→                  ))}
   471→                </SelectContent>
   472→              </Select>
   473→            </div>
   474→
   475→            {/* Price */}
   476→            <div>
   477→              <Label htmlFor="prod-price">Price ($) <span className="text-red-500">*</span></Label>
   478→              <Input
   479→                id="prod-price"
   480→                type="number"
   481→                step="0.01"
   482→                min="0"
   483→                placeholder="0.00"
   484→                value={productForm.price}
   485→                onChange={(e) => setProductForm({ ...productForm, price: e.target.value })}
   486→                className="mt-1"
   487→              />
   488→            </div>
   489→
   490→            {/* Stock Quantity */}
   491→            <div>
   492→              <Label htmlFor="prod-stock">Stock Quantity</Label>
   493→              <Input
   494→                id="prod-stock"
   495→                type="number"
   496→                min="0"
