    31→    31→  product: {
    32→    32→    id: string
    33→    33→    name: string
    34→    34→    ndc: string | null
    35→    35→    category: string
    36→    36→    sellingPrice: number

    38→    38→    status: string

    41→    41→    dosageForm: string | null
    42→    42→    strength: string | null

    45→    45→    batchNumber: string | null

    48→    48→
    49→    49→export function InventoryView() {
    50→    50→  const [items, setItems] = useState<InventoryItem[]>([])
    51→    51→  const [loading, setLoading] = useState(true)
    52→    52→  const [searchQuery, setSearchQuery] = useState('')
    53→    53→  const [categoryFilter, setCategoryFilter] = useState('ALL')
    54→    54→  const [stockFilter, setStockFilter] = useState('ALL')
    55→    55→  const [adjustDialog, setAdjustDialog] = useState(false)
    56→    56→  const [receiveDialog, setReceiveDialog] = useState(false)
    57→    57→  const [addProductDialog, setAddProductDialog] = useState(false)
    58→    58→  const [selectedItem, setSelectedItem] = useState<InventoryItem | null>(null)
    59→    59→  const [adjustType, setAdjustType] = useState('ADD')
    60→    60→  const [adjustAmount, setAdjustAmount] = useState('')
    61→    61→  const [adjustReason, setAdjustReason] = useState('')
    62→    62→  const [shipmentItems, setShipmentItems] = useState([{ productId: '', quantity: '', costPrice: '' }])
    63→    63→  const [sortBy, setSortBy] = useState<'name' | 'stock' | 'category'>('name')
    64→    64→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('asc')
    65→    65→  const [productForm, setProductForm] = useState({
    66→    66→    name: '', sku: '', category: 'OTC', price: '', stockQuantity: '',
    67→    67→    minStockLevel: '10', expiryDate: '', barcode: '',
    68→    68→  })
    69→    69→  const [savingProduct, setSavingProduct] = useState(false)
    70→    70→  const addToast = useAppStore((s) => s.addToast)
    71→    71→  const currentUser = useAppStore((s) => s.user)
    72→    72→
    73→    73→  const fetchInventory = useCallback(async () => {
    74→    74→    setLoading(true)
    75→    75→    try {
    76→    76→      const params = new URLSearchParams()
    77→    77→      if (searchQuery) params.set('search', searchQuery)
    78→    78→      if (categoryFilter !== 'ALL') params.set('category', categoryFilter)
    79→    79→      const res = await fetch(`/api/inventory?${params}`)
    80→    80→      if (res.ok) {
    81→    81→        const data = await res.json()
    82→    82→        setItems(data)
    83→    83→      }
    84→    84→    } catch {
    85→    85→      addToast({ title: 'Error', description: 'Failed to load inventory', variant: 'destructive' })
    86→    86→    } finally {
    87→    87→      setLoading(false)
    88→    88→    }
    89→    89→  }, [searchQuery, categoryFilter, addToast])
    90→    90→
    91→    91→  useEffect(() => { fetchInventory() }, [fetchInventory])
    92→    92→
    93→    93→  const filteredItems = items.filter((item) => {
    94→    94→    if (stockFilter === 'LOW') return item.quantity <= item.product.reorderPoint
    95→    95→    if (stockFilter === 'OUT') return item.quantity === 0
    96→    96→    if (stockFilter === 'OK') return item.quantity > item.product.reorderPoint
    97→    97→    return true
    98→    98→  }).sort((a, b) => {
    99→    99→    if (sortBy === 'name') return sortDir === 'asc' ? a.product.name.localeCompare(b.product.name) : b.product.name.localeCompare(a.product.name)
   100→   100→    if (sortBy === 'stock') return sortDir === 'asc' ? a.quantity - b.quantity : b.quantity - a.quantity
   101→   101→    return sortDir === 'asc' ? a.product.category.localeCompare(b.product.category) : b.product.category.localeCompare(a.product.category)
   102→   102→  })
   103→   103→
   104→   104→  const lowStockCount = items.filter((i) => i.quantity <= i.product.reorderPoint).length
   105→   105→  const outOfStockCount = items.filter((i) => i.quantity === 0).length
   106→   106→  const totalValue = items.reduce((sum, i) => sum + i.quantity * (i.product.costPrice || i.product.sellingPrice), 0)
   107→   107→
   108→   108→  const handleAdjust = async () => {
   109→   109→    if (!selectedItem || !adjustAmount || !adjustReason) return
   110→   110→    try {
   111→   111→      const adj = adjustType === 'ADD' ? parseInt(adjustAmount) : adjustType === 'REMOVE' ? -parseInt(adjustAmount) : parseInt(adjustAmount)
   112→   112→      await fetch('/api/inventory', {
   113→   113→        method: 'PUT',
   114→   114→        headers: { 'Content-Type': 'application/json' },
   115→   115→        body: JSON.stringify({ productId: selectedItem.productId, adjustment: adj, reason: adjustReason }),
   116→   116→      })
   117→   117→      addToast({ title: 'Stock Updated', description: `${selectedItem.product.name} adjusted`, variant: 'success' })
   118→   118→      setAdjustDialog(false)
   119→   119→      setAdjustAmount('')
   120→   120→      setAdjustReason('')
   121→   121→      fetchInventory()
   122→   122→    } catch {
   123→   123→      addToast({ title: 'Error', description: 'Failed to adjust stock', variant: 'destructive' })
   124→   124→    }
   125→   125→  }
   126→   126→
   127→   127→  const handleAddProduct = async () => {
   128→   128→    if (!productForm.name || !productForm.price) return
   129→   129→    setSavingProduct(true)
   130→   130→    try {
   131→   131→      const res = await fetch('/api/products', {
   132→   132→        method: 'POST',
   133→   133→        headers: {
   134→   134→          'Content-Type': 'application/json',
   135→   135→          'x-user-role': currentUser?.role || 'SUPER_ADMIN',
   136→   136→        },
   137→   137→        body: JSON.stringify({
   138→   138→          name: productForm.name,
   139→   139→          ndc: productForm.sku || null,
   140→   140→          category: productForm.category,
   141→   141→          sellingPrice: parseFloat(productForm.price),
   142→   142→          costPrice: parseFloat(productForm.price) * 0.7,
   143→   143→          reorderPoint: parseInt(productForm.minStockLevel) || 10,
   144→   144→          expiryDate: productForm.expiryDate || null,
   145→   145→          batchNumber: productForm.barcode || null,
   146→   146→        }),
   147→   147→      })
   148→   148→      if (!res.ok) {
   149→   149→        const err = await res.json()
   150→   150→        throw new Error(err.error || 'Failed to create product')
   151→   151→      }
   152→   152→      const newProduct = await res.json()
   153→   153→
   154→   154→      // Set initial stock quantity if specified
   155→   155→      const qty = parseInt(productForm.stockQuantity)
   156→   156→      if (qty > 0) {
   157→   157→        await fetch('/api/inventory', {
   158→   158→          method: 'PUT',
   159→   159→          headers: { 'Content-Type': 'application/json' },
   160→   160→          body: JSON.stringify({
   161→   161→            productId: newProduct.id,
   162→   162→            adjustment: qty,
   163→   163→            reason: 'Initial stock on product creation',
   164→   164→          }),
   165→   165→        })
   166→   166→      }
   167→   167→
   168→   168→      addToast({ title: 'Product Added', description: `${productForm.name} has been added to inventory`, variant: 'success' })
   169→   169→      setAddProductDialog(false)
   170→   170→      setProductForm({ name: '', sku: '', category: 'OTC', price: '', stockQuantity: '', minStockLevel: '10', expiryDate: '', barcode: '' })
   171→   171→      fetchInventory()
   172→   172→    } catch (err: any) {
   173→   173→      addToast({ title: 'Error', description: err.message || 'Failed to add product', variant: 'destructive' })
   174→   174→    } finally {
   175→   175→      setSavingProduct(false)
   176→   176→    }
   177→   177→  }
   178→   178→
   179→   179→  const handleReceiveShipment = async () => {
   180→   180→    const validItems = shipmentItems.filter((i) => i.productId && i.quantity)
   181→   181→    if (validItems.length === 0) return
   182→   182→    try {
   183→   183→      await fetch('/api/inventory', {
   184→   184→        method: 'POST',
   185→   185→        headers: { 'Content-Type': 'application/json' },
   186→   186→        body: JSON.stringify({
   187→   187→          items: validItems.map((i) => ({
   188→   188→            productId: i.productId,
   189→   189→            quantity: parseInt(i.quantity),
   190→   190→            costPrice: parseFloat(i.costPrice) || 0,
   191→   191→          })),
   192→   192→        }),
   193→   193→      })
   194→   194→      addToast({ title: 'Shipment Received', description: `${validItems.length} items received`, variant: 'success' })
   195→   195→      setReceiveDialog(false)
   196→   196→      setShipmentItems([{ productId: '', quantity: '', costPrice: '' }])
   197→   197→      fetchInventory()
   198→   198→    } catch {
   199→   199→      addToast({ title: 'Error', description: 'Failed to receive shipment', variant: 'destructive' })
   200→   200→    }
   201→   201→  }
   202→   202→
   203→   203→  return (
   204→   204→    <div className="space-y-4">
   205→   205→      {/* Stats Cards */}
   206→   206→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   207→   207→        <Card>
   208→   208→          <CardContent className="p-4 flex items-center gap-3">
   209→   209→            <div className="h-10 w-10 rounded-lg bg-emerald-100 flex items-center justify-center">
   210→   210→              <Package className="h-5 w-5 text-emerald-600" />
   211→   211→            </div>
   212→   212→            <div>
   213→   213→              <p className="text-2xl font-bold">{items.length}</p>
   214→   214→              <p className="text-xs text-muted-foreground">Total Products</p>
   215→   215→            </div>
   216→   216→          </CardContent>
   217→   217→        </Card>
   218→   218→        <Card>
   219→   219→          <CardContent className="p-4 flex items-center gap-3">
   220→   220→            <div className="h-10 w-10 rounded-lg bg-red-100 flex items-center justify-center">
   221→   221→              <AlertTriangle className="h-5 w-5 text-red-600" />
   222→   222→            </div>
   223→   223→            <div>
   224→   224→              <p className="text-2xl font-bold text-red-600">{lowStockCount}</p>
   225→   225→              <p className="text-xs text-muted-foreground">Low Stock Alerts</p>
   226→   226→            </div>
   227→   227→          </CardContent>
   228→   228→        </Card>
   229→   229→        <Card>
   230→   230→          <CardContent className="p-4 flex items-center gap-3">
   231→   231→            <div className="h-10 w-10 rounded-lg bg-amber-100 flex items-center justify-center">
   232→   232→              <AlertTriangle className="h-5 w-5 text-amber-600" />
   233→   233→            </div>
   234→   234→            <div>
   235→   235→              <p className="text-2xl font-bold text-amber-600">{outOfStockCount}</p>
   236→   236→              <p className="text-xs text-muted-foreground">Out of Stock</p>
   237→   237→            </div>
   238→   238→          </CardContent>
   239→   239→        </Card>
   240→   240→        <Card>
   241→   241→          <CardContent className="p-4 flex items-center gap-3">
   242→   242→            <div className="h-10 w-10 rounded-lg bg-teal-100 flex items-center justify-center">
   243→   243→              <TrendingUp className="h-5 w-5 text-teal-600" />
   244→   244→            </div>
   245→   245→            <div>
   246→   246→              <p className="text-2xl font-bold">${totalValue.toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 2 })}</p>
   247→   247→              <p className="text-xs text-muted-foreground">Inventory Value</p>
   248→   248→            </div>
   249→   249→          </CardContent>
   250→   250→        </Card>
   251→   251→      </div>
   252→   252→
   253→   253→      {/* Filters & Actions */}
   254→   254→      <Card>
   255→   255→        <CardContent className="p-4">
   256→   256→          <div className="flex flex-col sm:flex-row gap-3">
   257→   257→            <div className="relative flex-1">
   258→   258→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   259→   259→              <Input
   260→   260→                placeholder="Search by product name or NDC..."
   261→   261→                value={searchQuery}
   262→   262→                onChange={(e) => setSearchQuery(e.target.value)}
   263→   263→                className="pl-9"
   264→   264→              />
   265→   265→            </div>
   266→   266→            <Select value={categoryFilter} onValueChange={setCategoryFilter}>
   267→   267→              <SelectTrigger className="w-full sm:w-40">
   268→   268→                <Filter className="h-4 w-4 mr-2" />
   269→   269→                <SelectValue />
   270→   270→              </SelectTrigger>
   271→   271→              <SelectContent>
   272→   272→                <SelectItem value="ALL">All Categories</SelectItem>
   273→   273→                <SelectItem value="OTC">OTC</SelectItem>
   274→   274→                <SelectItem value="PRESCRIPTION">Prescription</SelectItem>
   275→   275→                <SelectItem value="SUPPLEMENT">Supplement</SelectItem>
   276→   276→                <SelectItem value="MEDICAL_DEVICE">Medical Device</SelectItem>
   277→   277→                <SelectItem value="PERSONAL_CARE">Personal Care</SelectItem>
   278→   278→                <SelectItem value="CONSUMABLES">Consumables</SelectItem>
   279→   279→              </SelectContent>
   280→   280→            </Select>
   281→   281→            <Select value={stockFilter} onValueChange={setStockFilter}>
   282→   282→              <SelectTrigger className="w-full sm:w-36">
   283→   283→                <SelectValue />
   284→   284→              </SelectTrigger>
   285→   285→              <SelectContent>
   286→   286→                <SelectItem value="ALL">All Stock</SelectItem>
   287→   287→                <SelectItem value="LOW">Low Stock</SelectItem>
   288→   288→                <SelectItem value="OUT">Out of Stock</SelectItem>
   289→   289→                <SelectItem value="OK">In Stock OK</SelectItem>
   290→   290→              </SelectContent>
   291→   291→            </Select>
   292→   292→            <Button onClick={() => setReceiveDialog(true)} className="bg-emerald-600 hover:bg-emerald-700">
   293→   293→              <Plus className="h-4 w-4 mr-2" />
   294→   294→              Receive Shipment
   295→   295→            </Button>
   296→   296→            <Button onClick={() => setAddProductDialog(true)} className="bg-teal-600 hover:bg-teal-700">
   297→   297→              <PackagePlus className="h-4 w-4 mr-2" />
   298→   298→              Add Product
   299→   299→            </Button>
   300→   300→          </div>
   301→   301→        </CardContent>
   302→   302→      </Card>
   303→   303→
   304→   304→      {/* Inventory Table */}
   305→   305→      <Card>
   306→   306→        <CardContent className="p-0">
   307→   307→          <Table>
   308→   308→            <TableHeader>
   309→   309→              <TableRow>
   310→   310→                <TableHead className="cursor-pointer" onClick={() => { setSortBy('name'); setSortDir(sortDir === 'asc' ? 'desc' : 'asc') }}>
   311→   311→                  <span className="flex items-center gap-1">Product <ArrowUpDown className="h-3 w-3" /></span>
   312→   312→                </TableHead>
   313→   313→                <TableHead className="hidden md:table-cell">NDC</TableHead>
   314→   314→                <TableHead className="cursor-pointer hidden sm:table-cell" onClick={() => { setSortBy('category'); setSortDir(sortDir === 'asc' ? 'desc' : 'asc') }}>
   315→   315→                  <span className="flex items-center gap-1">Category <ArrowUpDown className="h-3 w-3" /></span>
   316→   316→                </TableHead>
   317→   317→                <TableHead className="cursor-pointer text-right" onClick={() => { setSortBy('stock'); setSortDir(sortDir === 'asc' ? 'desc' : 'asc') }}>
   318→   318→                  <span className="flex items-center justify-end gap-1">In Stock <ArrowUpDown className="h-3 w-3" /></span>
   319→   319→                </TableHead>
   320→   320→                <TableHead className="hidden lg:table-cell text-right">Reorder Point</TableHead>
   321→   321→                <TableHead className="text-right">Status</TableHead>
   322→   322→                <TableHead className="hidden lg:table-cell text-right">Cost</TableHead>
   323→   323→                <TableHead className="hidden lg:table-cell text-right">Retail</TableHead>
   324→   324→                <TableHead className="text-right">Actions</TableHead>
   325→   325→              </TableRow>
   326→   326→            </TableHeader>
   327→   327→            <TableBody>
   328→   328→              {loading ? (
   329→   329→                Array.from({ length: 8 }).map((_, i) => (
   330→   330→                  <TableRow key={i}>
   331→   331→                    {Array.from({ length: 9 }).map((_, j) => (
   332→   332→                      <TableCell key={j}><Skeleton className="h-5 w-full" /></TableCell>
   333→   333→                    ))}
   334→   334→                  </TableRow>
   335→   335→                ))
   336→   336→              ) : filteredItems.length === 0 ? (
   337→   337→                <TableRow>
   338→   338→                  <TableCell colSpan={9} className="text-center py-8 text-muted-foreground">
   339→   339→                    No inventory items found
   340→   340→                  </TableCell>
   341→   341→                </TableRow>
   342→   342→              ) : (
   343→   343→                filteredItems.map((item) => {
   344→   344→                  const isLow = item.quantity <= item.product.reorderPoint
   345→   345→                  const isOut = item.quantity === 0
   346→   346→                  return (
   347→   347→                    <TableRow key={item.id} className={isOut ? 'bg-red-50/50' : isLow ? 'bg-amber-50/50' : ''}>
   348→   348→                      <TableCell>
   349→   349→                        <div>
   350→   350→                          <p className="font-medium text-sm">{item.product.name}</p>
   351→   351→                          <p className="text-xs text-muted-foreground">
   352→   352→                            {item.product.strength} {item.product.dosageForm} · {item.product.unitOfMeasure}
   353→   353→                          </p>
   354→   354→                        </div>
   355→   355→                      </TableCell>
   356→   356→                      <TableCell className="hidden md:table-cell text-xs font-mono">{item.product.ndc || '—'}</TableCell>
   357→   357→                      <TableCell className="hidden sm:table-cell">
   358→   358→                        <Badge variant="outline" className="text-xs">{item.product.category.replace(/_/g, ' ')}</Badge>
   359→   359→                      </TableCell>
   360→   360→                      <TableCell className="text-right font-bold">{item.quantity}</TableCell>
   361→   361→                      <TableCell className="hidden lg:table-cell text-right text-muted-foreground">{item.product.reorderPoint}</TableCell>
   362→   362→                      <TableCell className="text-right">
   363→   363→                        {isOut ? (
   364→   364→                          <Badge className="bg-red-100 text-red-700 border-red-200">Out of Stock</Badge>
   365→   365→                        ) : isLow ? (
   366→   366→                          <Badge className="bg-amber-100 text-amber-700 border-amber-200">Low Stock</Badge>
   367→   367→                        ) : (
   368→   368→                          <Badge className="bg-emerald-100 text-emerald-700 border-emerald-200">In Stock</Badge>
   369→   369→                        )}
   370→   370→                      </TableCell>
   371→   371→                      <TableCell className="hidden lg:table-cell text-right">${item.product.costPrice?.toFixed(2) || '—'}</TableCell>
   372→   372→                      <TableCell className="hidden lg:table-cell text-right">${item.product.sellingPrice.toFixed(2)}</TableCell>
   373→   373→                      <TableCell className="text-right">
   374→   374→                        <Button size="sm" variant="ghost" onClick={() => { setSelectedItem(item); setAdjustDialog(true) }}>
   375→   375→                          <Edit className="h-3.5 w-3.5 mr-1" />
   376→   376→                          Adjust
   377→   377→                        </Button>
   378→   378→                      </TableCell>
   379→   379→                    </TableRow>
   380→   380→                  )
   381→   381→                })
   382→   382→              )}
   383→   383→            </TableBody>
   384→   384→          </Table>
   385→   385→        </CardContent>
   386→   386→      </Card>
   387→   387→
   388→   388→      {/* Add Product Dialog */}
   389→   389→      <Dialog open={addProductDialog} onOpenChange={setAddProductDialog}>
   390→   390→        <DialogContent className="max-w-lg">
   391→   391→          <DialogHeader>
   392→   392→            <DialogTitle className="flex items-center gap-2">
   393→   393→              <PackagePlus className="h-5 w-5 text-teal-600" />
   394→   394→              Add New Product
   395→   395→            </DialogTitle>
   396→   396→          </DialogHeader>
   397→   397→          <div className="grid grid-cols-2 gap-4 max-h-[70vh] overflow-y-auto pr-1">
   398→   398→            {/* Product Name */}
   399→   399→            <div className="col-span-2">
   400→   400→              <Label htmlFor="prod-name">Product Name <span className="text-red-500">*</span></Label>
   401→   401→              <Input
   402→   402→                id="prod-name"
   403→   403→                placeholder="e.g., Amoxicillin 500mg Capsules"
   404→   404→                value={productForm.name}
   405→   405→                onChange={(e) => setProductForm({ ...productForm, name: e.target.value })}
   406→   406→                className="mt-1"
   407→   407→              />
   408→   408→            </div>
   409→   409→
   410→   410→            {/* SKU */}
   411→   411→            <div>
   412→   412→              <Label htmlFor="prod-sku">SKU</Label>
   413→   413→              <Input
   414→   414→                id="prod-sku"
   415→   415→                placeholder="e.g., SKU-00123"
