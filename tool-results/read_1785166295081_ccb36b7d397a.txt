   135→   135→function parsePermissions(raw: string | null): string[] {
   136→   136→  if (!raw) return []
   137→   137→  try {
   138→   138→    const parsed = JSON.parse(raw)
   139→   139→    return Array.isArray(parsed) ? parsed : []
   140→   140→  } catch {
   141→   141→    return []
   142→   142→  }
   143→   143→}

   146→   146→function PermissionGrid({
   147→   147→  permissions,
   148→   148→  onChange,
   149→   149→  disabled,
   150→   150→}: {
   151→   151→  permissions: string[]
   152→   152→  onChange: (updated: string[]) => void
   153→   153→  disabled?: boolean
   154→   154→}) {
   155→   155→  const toggle = (key: string) => {
   156→   156→    if (disabled) return
   157→   157→    if (permissions.includes(key)) {
   158→   158→      onChange(permissions.filter((p) => p !== key))
   159→   159→    } else {
   160→   160→      onChange([...permissions, key])
   161→   161→    }
   162→   162→  }
   163→   163→
   164→   164→  const selectAll = () => {
   165→   165→    if (disabled) return
   166→   166→    onChange(PERMISSIONS.map((p) => p.key))
   167→   167→  }
   168→   168→
   169→   169→  const deselectAll = () => {
   170→   170→    if (disabled) return
   171→   171→    onChange([])
   172→   172→  }
   173→   173→
   174→   174→  const allSelected = ALL_PERMISSIONS.every((p) => permissions.includes(p.key))
   175→   175→
   176→   176→  return (
   177→   177→    <div className="space-y-3">
   178→   178→      <div className="flex items-center justify-between">
   179→   179→        <Label className="text-sm font-semibold">Privileges & Permissions</Label>
   180→   180→        <div className="flex gap-1">
   181→   181→          <Button type="button" variant="outline" size="sm" className="h-6 text-xs" onClick={selectAll} disabled={disabled}>
   182→   182→            Select All
   183→   183→          </Button>
   184→   184→          <Button type="button" variant="outline" size="sm" className="h-6 text-xs" onClick={deselectAll} disabled={disabled}>
   185→   185→            Clear All
   186→   186→          </Button>
   187→   187→        </div>
   188→   188→      </div>
   189→   189→      {PERMISSION_CATEGORIES.map((cat) => {
   190→   190→        const catSelected = cat.permissions.filter(p => permissions.includes(p.key)).length
   191→   191→        return (
   192→   192→          <div key={cat.category} className="border rounded-lg">
   193→   193→            <div className="px-3 py-2 bg-gray-50/50 border-b flex items-center justify-between">
   194→   194→              <span className="text-xs font-semibold text-gray-600 uppercase tracking-wide">{cat.category}</span>
   195→   195→              <Badge variant="outline" className="text-[10px] h-5">{catSelected}/{cat.permissions.length}</Badge>
   196→   196→            </div>
   197→   197→            <div className="divide-y">
   198→   198→              {cat.permissions.map((perm) => {
   199→   199→                const isChecked = permissions.includes(perm.key)
   200→   200→                const Icon = perm.icon
   201→   201→                return (
   202→   202→                  <label
   203→   203→                    key={perm.key}
   204→   204→                    className={`flex items-center gap-3 px-3 py-2.5 cursor-pointer transition-colors ${
   205→   205→                      disabled ? 'opacity-60 cursor-not-allowed' : isChecked ? 'bg-emerald-50/60 hover:bg-emerald-50' : 'hover:bg-muted/50'
   206→   206→                    }`}
   207→   207→                  >
   208→   208→                    <Checkbox
   209→   209→                      checked={isChecked}
   210→   210→                      onCheckedChange={() => toggle(perm.key)}
   211→   211→                      disabled={disabled}
   212→   212→                      className={isChecked ? 'data-[state=checked]:bg-emerald-600 data-[state=checked]:border-emerald-600' : ''}
   213→   213→                    />
   214→   214→                    <Icon className={`h-4 w-4 shrink-0 ${isChecked ? 'text-emerald-600' : 'text-muted-foreground'}`} />
   215→   215→                    <div className="flex-1 min-w-0">
   216→   216→                      <span className={`text-sm font-medium ${isChecked ? 'text-emerald-700' : ''}`}>{perm.label}</span>
   217→   217→                      <span className="text-xs text-muted-foreground ml-2 hidden sm:inline">{perm.description}</span>
   218→   218→                    </div>
   219→   219→                    <Badge variant={isChecked ? 'outline' : 'secondary'} className={`text-[10px] h-5 ${isChecked ? 'border-emerald-300 text-emerald-700' : ''}`}>
   220→   220→                      {isChecked ? 'Granted' : 'Denied'}
   221→   221→                    </Badge>
   222→   222→                  </label>
   223→   223→                )
   224→   224→              })}
   225→   225→            </div>
   226→   226→          </div>
   227→   227→        )
   228→   228→      })}
   229→   229→      <div className="flex items-center justify-between text-xs text-muted-foreground">
   230→   230→        <span>{permissions.length} of {ALL_PERMISSIONS.length} permissions granted</span>
   231→   231→        {!allSelected && permissions.length > 0 && (
   232→   232→          <span className="text-amber-600 font-medium">Custom permissions (different from role defaults)</span>
   233→   233→        )}
   234→   234→      </div>
   235→   235→    </div>
   236→   236→  )
   237→   237→}
