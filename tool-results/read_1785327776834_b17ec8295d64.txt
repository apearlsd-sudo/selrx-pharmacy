    58→function formatDate(dateStr: string): string {
    59→  return new Date(dateStr).toLocaleDateString('en-US', {
    60→    month: 'short',
    61→    day: 'numeric',
    62→    year: 'numeric',
    63→    hour: '2-digit',
    64→    minute: '2-digit',
    65→  })
    66→}

    70→    month: 'short',

    75→// Reason badge colors
    76→function reasonBadgeColor(reason: string): string {
    77→  switch (reason) {
    78→    case 'DEFECTIVE': return 'bg-red-100 text-red-700 border-red-200'
    79→    case 'EXPIRED': return 'bg-orange-100 text-orange-700 border-orange-200'
    80→    case 'WRONG_ITEM': return 'bg-blue-100 text-blue-700 border-blue-200'
    81→    case 'WRONG_QUANTITY': return 'bg-purple-100 text-purple-700 border-purple-200'
    82→    case 'DAMAGED': return 'bg-rose-100 text-rose-700 border-rose-200'
    83→    case 'CUSTOMER_CHANGE_OF_MIND': return 'bg-amber-100 text-amber-700 border-amber-200'
    84→    case 'RECALLED': return 'bg-sky-100 text-sky-700 border-sky-200'
    85→    default: return 'bg-gray-100 text-gray-700 border-gray-200'
    86→  }
    87→}
    88→
    89→function reasonLabel(reason: string): string {
    90→  return reason.replace(/_/g, ' ').replace(/\b\w/g, (c) => c.toUpperCase())
    91→}

    93→// Status badge
    94→function statusBadge(status: string) {
    95→  switch (status) {
    96→    case 'PENDING_APPROVAL':
    97→      return <Badge className="bg-amber-100 text-amber-700 text-xs border-amber-200"><Clock className="h-3 w-3 mr-1" />Pending</Badge>

    99→      return <Badge className="bg-blue-100 text-blue-700 text-xs border-blue-200"><Check className="h-3 w-3 mr-1" />Approved</Badge>
   100→    case 'REJECTED':
   101→      return <Badge className="bg-red-100 text-red-700 text-xs border-red-200"><XCircle className="h-3 w-3 mr-1" />Rejected</Badge>
   102→    case 'COMPLETED':
   103→      return <Badge className="bg-emerald-100 text-emerald-700 text-xs border-emerald-200"><CheckCircle2 className="h-3 w-3 mr-1" />Completed</Badge>

   105→      return <Badge className="bg-gray-100 text-gray-500 text-xs border-gray-200"><Ban className="h-3 w-3 mr-1" />Cancelled</Badge>

   159→export function GoodsReturnView() {
   160→  const user = useAppStore((s) => s.user)
   161→  const addToast = useAppStore((s) => s.addToast)
   162→
   163→  // State for returns list
   164→  const [returns, setReturns] = useState<ReturnRecord[]>([])
   165→  const [summary, setSummary] = useState({
   166→    totalReturns: 0,
   167→    pendingCount: 0,
   168→    completedCount: 0,
   169→    totalRefundAmount: 0,
   170→    topReasons: [] as { reason: string; _count: { reason: number } }[],
   171→  })
   172→  const [loading, setLoading] = useState(true)
   173→  const [page, setPage] = useState(1)
   174→  const [totalPages, setTotalPages] = useState(1)
   175→  const [searchQuery, setSearchQuery] = useState('')
   176→  const [filterStatus, setFilterStatus] = useState('ALL')
   177→  const [filterReason, setFilterReason] = useState('ALL')
   178→
   179→  // State for new return dialog
   180→  const [newReturnOpen, setNewReturnOpen] = useState(false)
   181→  const [txSearchQuery, setTxSearchQuery] = useState('')
   182→  const [foundTx, setFoundTx] = useState<TransactionData | null>(null)
   183→  const [searchingTx, setSearchingTx] = useState(false)
   184→  const [selectedItem, setSelectedItem] = useState<TransactionItem | null>(null)
   185→  const [returnQty, setReturnQty] = useState(1)
   186→  const [returnReason, setReturnReason] = useState('')
   187→  const [returnReasonNote, setReturnReasonNote] = useState('')
   188→  const [returnRefundMethod, setReturnRefundMethod] = useState('CASH')
   189→  const [submitting, setSubmitting] = useState(false)
   190→
   191→  // State for detail dialog
   192→  const [detailOpen, setDetailOpen] = useState(false)
   193→  const [detailReturn, setDetailReturn] = useState<ReturnRecord | null>(null)
   194→  const [actionLoading, setActionLoading] = useState(false)
   195→
   196→  // State for active tab
   197→  const [activeTab, setActiveTab] = useState('all')
   198→
   199→  // Fetch returns list
   200→  const fetchReturns = useCallback(async () => {
   201→    setLoading(true)
   202→    try {
   203→      const params = new URLSearchParams({
   204→        page: String(page),
   205→        limit: '20',
   206→      })
   207→      if (filterStatus !== 'ALL') params.set('status', filterStatus)
   208→      if (filterReason !== 'ALL') params.set('reason', filterReason)
   209→      if (searchQuery) params.set('search', searchQuery)
   210→
   211→      const res = await fetch(`/api/returns?${params}`)
   212→      const data = await res.json()
   213→      if (data.returns && Array.isArray(data.returns)) {
   214→        setReturns(data.returns)
   215→        setTotalPages(data.pagination ? data.pagination.pages : 1)
   216→      }
   217→      if (data.summary) {
   218→        setSummary(data.summary)
   219→      }
   220→    } catch (err) {
   221→      console.error('Failed to fetch returns:', err)
   222→    } finally {
   223→      setLoading(false)
   224→    }
   225→  }, [page, filterStatus, filterReason, searchQuery])
   226→
   227→  useEffect(() => {
   228→    fetchReturns()
   229→  }, [fetchReturns])
   230→
   231→  // Search for transaction by transaction number
   232→  const searchTransaction = async () => {
   233→    if (!txSearchQuery.trim()) return
   234→    setSearchingTx(true)
   235→    setFoundTx(null)
   236→    setSelectedItem(null)
   237→    try {
   238→      const res = await fetch(`/api/sales-history?search=${encodeURIComponent(txSearchQuery.trim())}&limit=5`)
   239→      const data = await res.json()
   240→      if (data.transactions && Array.isArray(data.transactions) && data.transactions.length > 0) {
   241→        const tx = data.transactions[0]
   242→        setFoundTx({
   243→          id: tx.id,
   244→          transactionNo: tx.transactionNo,
   245→          status: tx.status,
   246→          items: tx.items,
   247→          customer: tx.customer,
   248→          createdAt: tx.createdAt,
   249→        })
   250→      } else {
   251→        addToast({ title: 'Transaction not found', description: 'No matching transaction found', variant: 'destructive' })
   252→      }
   253→    } catch (err) {
   254→      console.error('Failed to search transaction:', err)
   255→      addToast({ title: 'Search failed', description: 'Could not search for transaction', variant: 'destructive' })
   256→    } finally {
   257→      setSearchingTx(false)
   258→    }
   259→  }
   260→
   261→  // Submit new return
   262→  const submitReturn = async () => {
   263→    if (!selectedItem || !returnReason || !user || !foundTx) return
   264→    if (returnQty <= 0 || returnQty > selectedItem.quantity) {
   265→      addToast({ title: 'Invalid quantity', description: 'Quantity must be between 1 and the purchased amount', variant: 'destructive' })
   266→      return
   267→    }
   268→
   269→    setSubmitting(true)
   270→    try {
   271→      const customerName = foundTx.customer
   272→        ? `${foundTx.customer.firstName} ${foundTx.customer.lastName}`
   273→        : null
   274→      const customerId = foundTx.customer ? foundTx.customer.id : null
   275→
   276→      const res = await fetch('/api/returns', {
   277→        method: 'POST',
   278→        headers: { 'Content-Type': 'application/json' },
   279→        body: JSON.stringify({
   280→          transactionId: foundTx.id,
   281→          transactionItemId: selectedItem.id,
   282→          productId: selectedItem.productId,
   283→          productName: selectedItem.productName,
   284→          quantity: returnQty,
   285→          unitPrice: selectedItem.unitPrice,
   286→          refundAmount: selectedItem.unitPrice * returnQty,
   287→          reason: returnReason,
   288→          reasonNote: returnReasonNote || null,
   289→          customerId,
   290→          customerName,
   291→          userId: user.id,
   292→          refundMethod: returnRefundMethod,
   293→        }),
   294→      })
   295→      const data = await res.json()
   296→      if (res.ok && data.return) {
   297→        addToast({
   298→          title: 'Return Created',
   299→          description: `Return ticket ${data.return.returnNo} created successfully`,
   300→          variant: 'success',
   301→        })
   302→        // Reset form
   303→        setNewReturnOpen(false)
   304→        setTxSearchQuery('')
   305→        setFoundTx(null)
   306→        setSelectedItem(null)
   307→        setReturnQty(1)
   308→        setReturnReason('')
   309→        setReturnReasonNote('')
   310→        setReturnRefundMethod('CASH')
   311→        fetchReturns()
   312→      } else {
   313→        addToast({ title: 'Failed to create return', description: data.error || 'Unknown error', variant: 'destructive' })
   314→      }
   315→    } catch (err) {
   316→      console.error('Failed to create return:', err)
   317→      addToast({ title: 'Error', description: 'Failed to create return', variant: 'destructive' })
   318→    } finally {
   319→      setSubmitting(false)
   320→    }
   321→  }
   322→
   323→  // Action on return (approve, reject, complete, cancel)
   324→  const performAction = async (returnId: string, action: string) => {
   325→    if (!user) return
   326→    setActionLoading(true)
   327→    try {
   328→      const res = await fetch(`/api/returns/${returnId}`, {
   329→        method: 'PUT',
   330→        headers: { 'Content-Type': 'application/json' },
   331→        body: JSON.stringify({
   332→          action,
   333→          approvedById: user.id,
   334→        }),
   335→      })
   336→      const data = await res.json()
   337→      if (res.ok) {
   338→        const actionLabels: Record<string, string> = {
   339→          approve: 'approved',
   340→          reject: 'rejected',
   341→          complete: 'completed',
   342→          cancel: 'cancelled',
   343→        }
   344→        addToast({
   345→          title: `Return ${actionLabels[action]}`,
   346→          description: `Return has been ${actionLabels[action]}`,
   347→          variant: action === 'reject' ? 'destructive' : 'success',
   348→        })
   349→        fetchReturns()
   350→        // Refresh detail if open
   351→        if (detailReturn && detailReturn.id === returnId) {
   352→          setDetailReturn(data.return)
   353→        }
   354→      } else {
   355→        addToast({ title: 'Action failed', description: data.error || 'Unknown error', variant: 'destructive' })
   356→      }
   357→    } catch (err) {
   358→      console.error('Failed to perform action:', err)
   359→    } finally {
   360→      setActionLoading(false)
   361→    }
   362→  }
   363→
   364→  // Pie chart data for reasons
   365→  const reasonChartData = summary.topReasons.map((r) => ({
   366→    name: reasonLabel(r.reason),
   367→    value: r._count.reason,
   368→  }))
   369→
   370→  // Filter returns for tabs
   371→  const pendingReturns = returns.filter((r) => r.status === 'PENDING_APPROVAL')
   372→
   373→  const displayReturns = activeTab === 'pending' ? pendingReturns : returns
   374→
   375→  return (
   376→    <div className="space-y-6">
   377→      {/* Page Header */}
   378→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   379→        <div>
   380→          <h1 className="text-2xl font-bold text-gray-900">Goods Return</h1>
   381→          <p className="text-sm text-muted-foreground mt-1">Process product returns, restock inventory, and manage return tickets</p>
   382→        </div>
   383→        <Button
   384→          onClick={() => setNewReturnOpen(true)}
   385→          className="bg-emerald-600 hover:bg-emerald-700 text-white"
   386→        >
   387→          <Plus className="h-4 w-4 mr-2" />
   388→          New Return
   389→        </Button>
   390→      </div>
   391→
   392→      {/* Summary Cards */}
   393→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   394→        <Card className="border-none shadow-sm">
   395→          <CardContent className="p-5">
   396→            <div className="flex items-center justify-between">
   397→              <div>
   398→                <p className="text-xs font-medium text-muted-foreground uppercase tracking-wide">Total Returns</p>
   399→                <p className="text-2xl font-bold text-gray-900 mt-1">{summary.totalReturns}</p>
   400→              </div>
   401→              <div className="h-10 w-10 rounded-lg bg-emerald-100 flex items-center justify-center">
   402→                <RotateCcw className="h-5 w-5 text-emerald-600" />
   403→              </div>
   404→            </div>
   405→          </CardContent>
   406→        </Card>
   407→
   408→        <Card className="border-none shadow-sm">
   409→          <CardContent className="p-5">
   410→            <div className="flex items-center justify-between">
   411→              <div>
   412→                <p className="text-xs font-medium text-muted-foreground uppercase tracking-wide">Pending Approval</p>
   413→                <p className="text-2xl font-bold text-amber-600 mt-1">{summary.pendingCount}</p>
   414→              </div>
   415→              <div className="h-10 w-10 rounded-lg bg-amber-100 flex items-center justify-center">
   416→                <Clock className="h-5 w-5 text-amber-600" />
   417→              </div>
   418→            </div>
   419→          </CardContent>
   420→        </Card>
   421→
   422→        <Card className="border-none shadow-sm">
   423→          <CardContent className="p-5">
   424→            <div className="flex items-center justify-between">
   425→              <div>
   426→                <p className="text-xs font-medium text-muted-foreground uppercase tracking-wide">Completed</p>
   427→                <p className="text-2xl font-bold text-emerald-600 mt-1">{summary.completedCount}</p>
   428→              </div>
   429→              <div className="h-10 w-10 rounded-lg bg-emerald-100 flex items-center justify-center">
   430→                <CheckCircle2 className="h-5 w-5 text-emerald-600" />
   431→              </div>
   432→            </div>
   433→          </CardContent>
   434→        </Card>
   435→
   436→        <Card className="border-none shadow-sm">
   437→          <CardContent className="p-5">
   438→            <div className="flex items-center justify-between">
   439→              <div>
   440→                <p className="text-xs font-medium text-muted-foreground uppercase tracking-wide">Total Refunded</p>
   441→                <p className="text-2xl font-bold text-gray-900 mt-1">{formatCurrency(summary.totalRefundAmount)}</p>
   442→              </div>
   443→              <div className="h-10 w-10 rounded-lg bg-rose-100 flex items-center justify-center">
   444→                <DollarSign className="h-5 w-5 text-rose-600" />

   446→            </div>

   448→        </Card>

   455→            <TabsTrigger value="all">All Returns</TabsTrigger>
   456→            <TabsTrigger value="pending" className="relative">
   457→              Pending
   458→              {summary.pendingCount > 0 && (
   459→                <span className="ml-1.5 h-5 w-5 rounded-full bg-amber-500 text-white text-[10px] font-bold flex items-center justify-center inline-flex">
   460→                  {summary.pendingCount}
   461→                </span>
   462→              )}
   463→            </TabsTrigger>
   464→          </TabsList>
   465→
   466→          {/* Filters */}
   467→          <div className="flex items-center gap-2 flex-wrap">
   468→            <div className="relative">
   469→              <Search className="absolute left-2.5 top-1/2 -translate-y-1/2 h-3.5 w-3.5 text-muted-foreground" />
   470→              <Input
   471→                placeholder="Search returns..."
   472→                className="h-9 w-48 pl-8 text-xs"
   473→                value={searchQuery}

   484→                <SelectItem value="APPROVED">Approved</SelectItem>
   485→                <SelectItem value="REJECTED">Rejected</SelectItem>
   486→                <SelectItem value="COMPLETED">Completed</SelectItem>
   487→                <SelectItem value="CANCELLED">Cancelled</SelectItem>
   488→              </SelectContent>
   489→            </Select>
   490→            <Select value={filterReason} onValueChange={(val) => { setFilterReason(val); setPage(1) }}>
   491→              <SelectTrigger className="h-9 w-[140px] text-xs">
   492→                <SelectValue placeholder="Reason" />

   498→                <SelectItem value="WRONG_ITEM">Wrong Item</SelectItem>

   500→                <SelectItem value="DAMAGED">Damaged</SelectItem>

   538→      {/* Reason Breakdown (when there's data) */}
