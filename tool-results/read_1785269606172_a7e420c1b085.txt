    85→    85→  const [customerDropdownOpen, setCustomerDropdownOpen] = useState(false)

   121→   121→        const json = await res.json()
   122→   122→        setProducts(json.products || [])
   123→   123→      }
   124→   124→    } catch {
   125→   125→      addToast({ title: 'Error', description: 'Failed to search products', variant: 'destructive' })
   126→   126→    } finally {
   127→   127→      setSearching(false)
   128→   128→    }
   129→   129→  }, [addToast])
   130→   130→

   138→   138→      const res = await fetch(`/api/customers?search=${encodeURIComponent(query)}&limit=10`)
   139→   139→      if (res.ok) {
   140→   140→        const json = await res.json()
   141→   141→        setCustomerOptions(json.customers || [])
   142→   142→      }
   143→   143→    } catch {
   144→   144→      // Silent fail
   145→   145→    }
   146→   146→  }, [])
   147→   147→
   148→   148→  // Barcode scan
   149→   149→  const handleBarcodeScan = useCallback(async (barcode: string) => {

   152→   152→    try {
   153→   153→      const res = await fetch('/api/hardware?action=barcode', {
   154→   154→        method: 'POST',
   155→   155→        headers: { 'Content-Type': 'application/json' },
   156→   156→        body: JSON.stringify({ barcode: barcode.trim() }),
   157→   157→      })
   158→   158→      if (res.ok) {
   159→   159→        const json = await res.json()
   160→   160→        if (json.product) {
   161→   161→          const product = json.product
   162→   162→          addToCart(
   163→   163→            {
   164→   164→              id: product.id,
   165→   165→              name: product.name,
   166→   166→              ndc: product.ndc,
   167→   167→              sellingPrice: product.sellingPrice,
   168→   168→              requiresPrescription: product.requiresPrescription,
   169→   169→              unitOfMeasure: product.unitOfMeasure,
   170→   170→              strength: product.strength,
   171→   171→              dosageForm: product.dosageForm,
   172→   172→            },
   173→   173→            1
   174→   174→          )
   175→   175→          addToast({
   176→   176→            title: 'Product Added',
   177→   177→            description: `${product.name} added to cart`,
   178→   178→            variant: 'success',
   179→   179→            duration: 2000,
   180→   180→          })
   181→   181→        } else {
   182→   182→          addToast({
   183→   183→            title: 'Not Found',
   184→   184→            description: 'No product found for this barcode',
   185→   185→            variant: 'destructive',
   186→   186→            duration: 3000,
   187→   187→          })
   188→   188→        }
   189→   189→      }
   190→   190→    } catch {
   191→   191→      addToast({
   192→   192→        title: 'Scan Error',
   193→   193→        description: 'Barcode scan failed',
   194→   194→        variant: 'destructive',
   195→   195→      })
   196→   196→    } finally {
   197→   197→      setSearching(false)
   198→   198→      setBarcodeInput('')
   199→   199→      setShowBarcodeInput(false)
   200→   200→    }
   201→   201→  }, [addToCart, addToast])
   202→   202→
   203→   203→  // Debounced product search
   204→   204→  useEffect(() => {
   205→   205→    if (debounceRef.current) clearTimeout(debounceRef.current)
   206→   206→    debounceRef.current = setTimeout(() => {
   207→   207→      searchProducts(searchQuery, activeCategory)
   208→   208→    }, 300)
   209→   209→    return () => {
   210→   210→      if (debounceRef.current) clearTimeout(debounceRef.current)
   211→   211→    }
   212→   212→  }, [searchQuery, activeCategory, searchProducts])
   213→   213→
   214→   214→  // Debounced customer search
   215→   215→  useEffect(() => {
   216→   216→    if (debounceRef.current) clearTimeout(debounceRef.current)
   217→   217→    debounceRef.current = setTimeout(() => {
   218→   218→      searchCustomers(customerSearch)
   219→   219→    }, 300)
   220→   220→    return () => {
   221→   221→      if (debounceRef.current) clearTimeout(debounceRef.current)
   222→   222→    }
   223→   223→  }, [customerSearch, searchCustomers])
   224→   224→
   225→   225→  // Re-search products when inventory changes (stock count, adjust, etc.)
   226→   226→  const prevInventoryVersion = useRef(inventoryVersion)
   227→   227→  useEffect(() => {
   228→   228→    if (prevInventoryVersion.current !== inventoryVersion) {
   229→   229→      prevInventoryVersion.current = inventoryVersion
   230→   230→      searchProducts(searchQuery, activeCategory)
   231→   231→    }
   232→   232→  }, [inventoryVersion, searchQuery, activeCategory, searchProducts])
   233→   233→
   234→   234→  const handleAddToCart = (product: Product) => {
   235→   235→    addToCart(
   236→   236→      {
   237→   237→        id: product.id,
   238→   238→        name: product.name,
   239→   239→        ndc: product.ndc,
   240→   240→        sellingPrice: product.sellingPrice,
   241→   241→        requiresPrescription: product.requiresPrescription,
   242→   242→        unitOfMeasure: product.unitOfMeasure,
   243→   243→        strength: product.strength,
   244→   244→        dosageForm: product.dosageForm,
   245→   245→      },
   246→   246→      1
   247→   247→    )
   248→   248→    addToast({
   249→   249→      title: 'Added',
   250→   250→      description: `${product.name} added to cart`,
   251→   251→      variant: 'success',
   252→   252→      duration: 1500,
   253→   253→    })
   254→   254→  }
   255→   255→
   256→   256→  const handleProcessPayment = async () => {
   257→   257→    if (cart.length === 0) {
   258→   258→      addToast({ title: 'Empty Cart', description: 'Add items before processing', variant: 'destructive' })
   259→   259→      return
   260→   260→    }
   261→   261→
   262→   262→    if (paymentMethod === 'CASH') {
   263→   263→      const tendered = parseFloat(amountTendered)
   264→   264→      if (isNaN(tendered) || tendered < total) {
   265→   265→        addToast({
   266→   266→          title: 'Insufficient Amount',
   267→   267→          description: `Amount tendered must be at least ${formatCurrency(total)}`,
   268→   268→          variant: 'destructive',
   269→   269→        })
   270→   270→        return
   271→   271→      }
   272→   272→    }
   273→   273→
   274→   274→    setIsProcessingPayment(true)
   275→   275→    try {
   276→   276→      const payload = {
   277→   277→        customerId: selectedCustomer?.id || null,
   278→   278→        items: cart.map((item: CartItem) => ({
   279→   279→          productId: item.product.id,
   280→   280→          productName: item.product.name,
   281→   281→          quantity: item.quantity,
   282→   282→          unitPrice: item.product.sellingPrice,
   283→   283→          subtotal: item.product.sellingPrice * item.quantity,
   284→   284→          requiresRx: item.product.requiresPrescription,
   285→   285→        })),
   286→   286→        paymentMethod,
   287→   287→        subtotal,
   288→   288→        tax,
   289→   289→        discount: 0,
   290→   290→        total,
   291→   291→        paymentAmount: paymentMethod === 'CASH' ? parseFloat(amountTendered) || total : total,
   292→   292→      }
   293→   293→
   294→   294→      const res = await fetch('/api/transactions', {
   295→   295→        method: 'POST',
   296→   296→        headers: { 'Content-Type': 'application/json' },
   297→   297→        body: JSON.stringify(payload),
   298→   298→      })
   299→   299→
   300→   300→      if (!res.ok) {
   301→   301→        const err = await res.json()
   302→   302→        throw new Error(err.error || 'Transaction failed')
   303→   303→      }
   304→   304→
   305→   305→      const transaction = await res.json()
   306→   306→      setReceiptTxn(transaction)
   307→   307→      clearCart()
   308→   308→      setAmountTendered('')
   309→   309→      addToast({
   310→   310→        title: 'Payment Successful',
   311→   311→        description: `Transaction ${transaction.transactionNo} completed`,
   312→   312→        variant: 'success',
   313→   313→        duration: 3000,
   314→   314→      })
   315→   315→    } catch (err) {
   316→   316→      addToast({
   317→   317→        title: 'Payment Failed',
   318→   318→        description: err instanceof Error ? err.message : 'Unknown error',
   319→   319→        variant: 'destructive',
   320→   320→        duration: 5000,
   321→   321→      })
   322→   322→    } finally {
   323→   323→      setIsProcessingPayment(false)
   324→   324→    }
   325→   325→  }
   326→   326→
   327→   327→  const handleVoidTransaction = () => {
   328→   328→    if (cart.length === 0) return
   329→   329→    addToast({
   330→   330→      title: 'Transaction Voided',
   331→   331→      description: 'Cart has been cleared',
   332→   332→      variant: 'destructive',
   333→   333→    })
   334→   334→    clearCart()
   335→   335→    setAmountTendered('')
   336→   336→  }
   337→   337→
   338→   338→  const changeAmount =
   339→   339→    paymentMethod === 'CASH' && parseFloat(amountTendered) > 0
   340→   340→      ? parseFloat(amountTendered) - total
   341→   341→      : 0
   342→   342→
   343→   343→  return (
   344→   344→    <>
   345→   345→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
   346→   346→        {/* Left Column - Product Search & List */}
   347→   347→        <div className="lg:col-span-2 space-y-4">
   348→   348→          {/* Search Bar */}
   349→   349→          <Card className="gap-0">
   350→   350→            <CardContent className="p-0">
   351→   351→              <div className="flex items-center gap-2 p-3">
   352→   352→                <div className="relative flex-1">
   353→   353→                  <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   354→   354→                  <Input
   355→   355→                    placeholder="Search products by name, NDC, generic name..."
   356→   356→                    value={searchQuery}
   357→   357→                    onChange={(e) => setSearchQuery(e.target.value)}
   358→   358→                    className="pl-9 h-10"
   359→   359→                  />
   360→   360→                </div>
   361→   361→                <Button
   362→   362→                  variant="outline"
   363→   363→                  size="icon"
   364→   364→                  onClick={() => setShowBarcodeInput(!showBarcodeInput)}
   365→   365→                  title="Barcode Scan"
   366→   366→                >
   367→   367→                  <Camera className="h-4 w-4" />
   368→   368→                </Button>
   369→   369→              </div>
   370→   370→              {/* Barcode Input */}
   371→   371→              {showBarcodeInput && (
   372→   372→                <div className="flex items-center gap-2 px-3 pb-3">
   373→   373→                  <Input
   374→   374→                    placeholder="Scan or type barcode..."
   375→   375→                    value={barcodeInput}
   376→   376→                    onChange={(e) => setBarcodeInput(e.target.value)}
   377→   377→                    onKeyDown={(e) => {
   378→   378→                      if (e.key === 'Enter') handleBarcodeScan(barcodeInput)
   379→   379→                    }}
   380→   380→                    className="h-9"
   381→   381→                    autoFocus
   382→   382→                  />
   383→   383→                  <Button
   384→   384→                    size="sm"
   385→   385→                    onClick={() => handleBarcodeScan(barcodeInput)}
   386→   386→                    disabled={!barcodeInput.trim()}
   387→   387→                  >
   388→   388→                    Scan
   389→   389→                  </Button>
   390→   390→                  <Button
   391→   391→                    size="icon"
   392→   392→                    variant="ghost"
   393→   393→                    onClick={() => {
   394→   394→                      setShowBarcodeInput(false)
   395→   395→                      setBarcodeInput('')
   396→   396→                    }}
   397→   397→                  >
   398→   398→                    <X className="h-4 w-4" />
   399→   399→                  </Button>
   400→   400→                </div>
   401→   401→              )}
   402→   402→              {/* Category Filters */}
   403→   403→              <div className="flex items-center gap-1.5 px-3 pb-3 overflow-x-auto">
   404→   404→                {CATEGORIES.map((cat) => (
   405→   405→                  <Button
   406→   406→                    key={cat.value}
   407→   407→                    variant={activeCategory === cat.value ? 'default' : 'outline'}
   408→   408→                    size="sm"
   409→   409→                    className={`text-xs whitespace-nowrap h-7 ${
   410→   410→                      activeCategory === cat.value ? 'bg-emerald-600 hover:bg-emerald-700' : ''
   411→   411→                    }`}
   412→   412→                    onClick={() => setActiveCategory(cat.value)}
   413→   413→                  >
   414→   414→                    {cat.label}
   415→   415→                  </Button>
   416→   416→                ))}
   417→   417→              </div>
   418→   418→            </CardContent>
   419→   419→          </Card>
   420→   420→
   421→   421→          {/* Product Results */}
   422→   422→          <div className="max-h-[calc(100vh-320px)] overflow-y-auto">
   423→   423→            {searching ? (
   424→   424→              <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   425→   425→                {[...Array(6)].map((_, i) => (
   426→   426→                  <Card key={i}>
   427→   427→                    <CardContent className="p-4">
   428→   428→                      <div className="animate-pulse space-y-2">
   429→   429→                        <div className="h-4 bg-muted rounded w-3/4" />
   430→   430→                        <div className="h-3 bg-muted rounded w-1/2" />
   431→   431→                        <div className="flex justify-between mt-2">
   432→   432→                          <div className="h-4 bg-muted rounded w-20" />
   433→   433→                          <div className="h-8 bg-muted rounded w-8" />
   434→   434→                        </div>
   435→   435→                      </div>
   436→   436→                    </CardContent>
   437→   437→                  </Card>
   438→   438→                ))}
   439→   439→              </div>
   440→   440→            ) : products.length === 0 ? (
   441→   441→              <Card>
   442→   442→                <CardContent className="p-8 text-center">
   443→   443→                  <PackageX className="h-10 w-10 text-muted-foreground mx-auto mb-2" />
   444→   444→                  <p className="text-muted-foreground text-sm">No products found</p>
   445→   445→                </CardContent>
   446→   446→              </Card>
   447→   447→            ) : (
   448→   448→              <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   449→   449→                {products
   450→   450→                  .filter((p) => p.status === 'ACTIVE')
   451→   451→                  .map((product) => {
   452→   452→                    const stock = product.inventory?.[0]?.quantity ?? 0
   453→   453→                    const inCart = cart.find((item: CartItem) => item.product.id === product.id)
   454→   454→                    return (
   455→   455→                      <Card
   456→   456→                        key={product.id}
   457→   457→                        className={`group hover:shadow-md transition-all cursor-pointer ${
   458→   458→                          inCart ? 'ring-2 ring-emerald-500 bg-emerald-50/30' : ''
   459→   459→                        }`}
   460→   460→                        onClick={() => handleAddToCart(product)}
   461→   461→                      >
   462→   462→                        <CardContent className="p-4">
   463→   463→                          <div className="flex items-start justify-between gap-2">
   464→   464→                            <div className="flex-1 min-w-0">
   465→   465→                              <p className="font-medium text-sm leading-tight truncate">
   466→   466→                                {product.name}
   467→   467→                              </p>
   468→   468→                              <p className="text-xs text-muted-foreground mt-0.5">
   469→   469→                                {[product.strength, product.dosageForm, product.unitOfMeasure]
   470→   470→                                  .filter(Boolean)
   471→   471→                                  .join(' · ') || product.category}
   472→   472→                              </p>
   473→   473→                            </div>
   474→   474→                            <div className="text-right shrink-0">
   475→   475→                              <p className="font-bold text-emerald-700 text-sm">
   476→   476→                                {formatCurrency(product.sellingPrice)}
   477→   477→                              </p>
   478→   478→                              <p
   479→   479→                                className={`text-xs mt-0.5 ${
   480→   480→                                  stock <= 5 ? 'text-red-500 font-medium' : 'text-muted-foreground'
   481→   481→                                }`}
   482→   482→                              >
   483→   483→                                Stock: {stock}
   484→   484→                              </p>
   485→   485→                            </div>
   486→   486→                          </div>
   487→   487→                          <div className="flex items-center justify-between mt-3">
   488→   488→                            <div className="flex items-center gap-1.5">
