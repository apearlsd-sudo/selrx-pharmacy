     1→'use client'

     3→import { useState, useEffect, useCallback, useRef } from 'react'
     4→import {
     5→  DollarSign, ShoppingCart, TrendingUp, CalendarDays, Download, FileText,
     6→  Users, UserCircle, ArrowUpRight, ArrowDownRight,
     7→} from 'lucide-react'
     8→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
     9→import { Button } from '@/components/ui/button'
    10→import { Badge } from '@/components/ui/badge'
    11→import { Skeleton } from '@/components/ui/skeleton'
    12→import {
    13→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
    14→} from '@/components/ui/table'
    15→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs'
    16→import { Input } from '@/components/ui/input'
    17→import { Label } from '@/components/ui/label'
    18→import {
    19→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    20→} from '@/components/ui/select'
    21→import {
    22→  BarChart, Bar, LineChart, Line, PieChart, Pie, Cell, XAxis, YAxis, CartesianGrid, Tooltip, ResponsiveContainer, Legend,
    23→} from 'recharts'
    24→import { useAppStore } from '@/store/app-store'
    25→import { authHeaders } from '@/lib/auth-headers'
    26→import { formatCurrency } from '@/lib/currency'
    27→
    28→const CHART_COLORS = ['#059669', '#14b8a6', '#10b981', '#34d399', '#6ee7b7', '#0d9488', '#0f766e', '#a7f3d0', '#0891b2', '#06b6d4']
    29→
    30→interface UserSalesData {
    31→  userId: string
    32→  userName: string
    33→  userEmail: string
    34→  userRole: string
    35→  transactionCount: number
    36→  totalSales: number
    37→  totalSubtotal: number
    38→  totalDiscount: number
    39→  averageSale: number
    40→  totalItemsSold: number
    41→}
    42→
    43→interface SalesSummary {
    44→  totalSales: number
    45→  totalTransactions: number
    46→  totalDiscount: number
    47→  averageTransaction: number
    48→  topSeller: UserSalesData | null
    49→  dateRange: { from: string | null; to: string | null }
    50→}
    51→
    52→interface DailySale {
    53→  date: string
    54→  sales: number
    55→  count: number
    56→}
    57→
    58→interface AllUser {
    59→  id: string
    60→  name: string
    61→  role: string
    62→}
    63→
    64→export function ReportsView() {
    65→  const [activeTab, setActiveTab] = useState('sales')
    66→  const [salesStats, setSalesStats] = useState<any>(null)
    67→  const [inventory, setInventory] = useState<any[]>([])
    68→  const [prescriptions, setPrescriptions] = useState<any[]>([])
    69→  const [transactions, setTransactions] = useState<any[]>([])
    70→  const [loading, setLoading] = useState(true)
    71→  const [dateFrom, setDateFrom] = useState('')
    72→  const [dateTo, setDateTo] = useState('')
    73→  const [completedStockTakes, setCompletedStockTakes] = useState<any[]>([])
    74→  const addToast = useAppStore((s) => s.addToast)
    75→  const inventoryVersion = useAppStore((s) => s.inventoryVersion)
    76→  const user = useAppStore((s) => s.user)
    77→  const isSuperAdmin = user?.role === 'SUPER_ADMIN'
    78→
    79→  // Per-user sales analytics state
    80→  const [userSalesData, setUserSalesData] = useState<UserSalesData[]>([])
    81→  const [salesSummary, setSalesSummary] = useState<SalesSummary | null>(null)
    82→  const [dailySalesTrend, setDailySalesTrend] = useState<DailySale[]>([])
    83→  const [allUsers, setAllUsers] = useState<AllUser[]>([])
    84→  const [selectedUserId, setSelectedUserId] = useState<string>('all')
    85→  const [userSalesLoading, setUserSalesLoading] = useState(false)
    86→
    87→  const fetchSalesData = useCallback(async () => {
    88→    setLoading(true)
    89→    try {
    90→      const [statsRes, txRes, invRes, rxRes, stRes] = await Promise.all([
    91→        fetch('/api/transactions?action=stats', { headers: authHeaders() }),
    92→        fetch('/api/transactions', { headers: authHeaders() }),
    93→        fetch('/api/inventory'),
    94→        fetch('/api/prescriptions'),
    95→        fetch('/api/stock-take', { headers: authHeaders() }),
    96→      ])
    97→      if (statsRes.ok) setSalesStats(await statsRes.json())
    98→      if (txRes.ok) { const d = await txRes.json(); setTransactions(Array.isArray(d) ? d : d.transactions || []) }
    99→      if (invRes.ok) setInventory(await invRes.json())
   100→      if (rxRes.ok) { const d = await rxRes.json(); setPrescriptions(Array.isArray(d) ? d : d.prescriptions || []) }
   101→      if (stRes.ok) {
   102→        const allTakes = await stRes.json()
   103→        setCompletedStockTakes(Array.isArray(allTakes) ? allTakes.filter((st: any) => st.status === 'COMPLETED') : [])
   104→      }
   105→    } catch {
   106→      addToast({ title: 'Error', description: 'Failed to load report data', variant: 'destructive' })
   107→    } finally {
   108→      setLoading(false)
   109→    }
   110→  }, [addToast])
   111→
   112→  // Fetch per-user sales analytics
   113→  const fetchUserSalesAnalytics = useCallback(async (userId?: string) => {
   114→    setUserSalesLoading(true)
   115→    try {
   116→      const params = new URLSearchParams()
   117→      if (userId && userId !== 'all') params.set('userId', userId)
   118→      if (dateFrom) params.set('from', dateFrom)
   119→      if (dateTo) params.set('to', dateTo)
   120→
   121→      const res = await fetch(`/api/sales-history?${params.toString()}`, { headers: authHeaders() })
   122→      if (res.ok) {
   123→        const data = await res.json()
   124→        setSalesSummary(data.summary)
   125→        setUserSalesData(data.salesByUser || [])
   126→        setDailySalesTrend(data.dailySales || [])
   127→        // Fetch all users list for the SUPER_ADMIN dropdown
   128→        if (data.allUsers) setAllUsers(data.allUsers)
   129→      }
   130→    } catch {
   131→      addToast({ title: 'Error', description: 'Failed to load user sales analytics', variant: 'destructive' })
   132→    } finally {
   133→      setUserSalesLoading(false)
   134→    }
   135→  }, [addToast, dateFrom, dateTo])
   136→
   137→  useEffect(() => { fetchSalesData() }, [fetchSalesData])
   138→
   139→  // Fetch user analytics when tab activates or filters change
   140→  useEffect(() => {
   141→    if (activeTab === 'user-sales') {
   142→      fetchUserSalesAnalytics(selectedUserId)
   143→    }
   144→  }, [activeTab, selectedUserId, fetchUserSalesAnalytics])
   145→
   146→  // Re-fetch inventory data in reports when stock changes
   147→  const prevInvVer = useRef(inventoryVersion)
   148→  useEffect(() => {
   149→    if (prevInvVer.current !== inventoryVersion) {
   150→      prevInvVer.current = inventoryVersion
   151→      fetch('/api/inventory').then(r => { if (r.ok) r.json().then(setInventory) }).catch(() => {})
   152→    }
   153→  }, [inventoryVersion])
   154→
   155→  // Prepare chart data
   156→  const salesByCategory = salesStats?.topProducts?.map((p: any, i: number) => ({
   157→    name: p.productName?.split(' ').slice(0, 2).join(' ') || 'Unknown',
   158→    revenue: p._sum?.subtotal || 0,
   159→    units: p._sum?.quantity || 0,
   160→    fill: CHART_COLORS[i % CHART_COLORS.length],
   161→  })) || []
   162→
   163→  const dailySales = transactions
   164→    .filter((t: any) => t.status === 'COMPLETED')
   165→    .reduce((acc: any, t: any) => {
   166→      const date = new Date(t.createdAt).toLocaleDateString('en-US', { month: 'short', day: 'numeric' })
   167→      const existing = acc.find((d: any) => d.date === date)
   168→      if (existing) { existing.sales += t.total; existing.count += 1 }
   169→      else { acc.push({ date, sales: t.total, count: 1 }) }
   170→      return acc
   171→    }, [])
   172→    .slice(-7)
   173→
   174→  const categoryData = inventory.reduce((acc: any, item: any) => {
   175→    const cat = item.product?.category?.replace(/_/g, ' ') || 'Other'
   176→    const existing = acc.find((c: any) => c.category === cat)
   177→    if (existing) existing.count++
   178→    else acc.push({ category: cat, count: 1, fill: CHART_COLORS[acc.length % CHART_COLORS.length] })
   179→    return acc
   180→  }, [])
   181→
   182→  const lowStockItems = inventory.filter((i: any) => i.quantity <= i.product?.reorderPoint)
   183→  const expiringSoon = inventory.filter((i: any) => i.product?.expiryDate && new Date(i.product.expiryDate) <= new Date(Date.now() + 30 * 86400000))
   184→
   185→  const rxByStatus = prescriptions.reduce((acc: any, rx: any) => {
   186→    const existing = acc.find((s: any) => s.status === rx.status)
   187→    if (existing) existing.count++
   188→    else acc.push({ status: rx.status?.replace(/_/g, ' '), count: 1, fill: CHART_COLORS[acc.length % CHART_COLORS.length] })
   189→    return acc
   190→  }, [])
   191→
   192→  // Per-user sales chart data
   193→  const userSalesChartData = userSalesData.map((u, i) => ({
   194→    name: u.userName?.split(' ')[0] || 'Unknown',
   195→    sales: u.totalSales,
   196→    transactions: u.transactionCount,
   197→    items: u.totalItemsSold,
   198→    fill: CHART_COLORS[i % CHART_COLORS.length],
   199→  }))
   200→
   201→  // Daily sales trend for user analytics
   202→  const userDailyChartData = dailySalesTrend.map((d) => ({
   203→    date: d.date,
   204→    sales: d.sales,
   205→    count: d.count,
   206→  }))
   207→
   208→  // CSV export for per-user analytics
   209→  const exportUserSalesCSV = useCallback(() => {
   210→    if (userSalesData.length === 0) {
   211→      addToast({ title: 'No Data', description: 'No data to export', variant: 'destructive' })
   212→      return
   213→    }
   214→    const headers = ['User', 'Email', 'Role', 'Transactions', 'Total Sales', 'Total Discount', 'Avg Sale', 'Items Sold']
   215→    const rows = userSalesData.map(u => [
   216→      u.userName, u.userEmail, u.userRole, u.transactionCount,
   217→      u.totalSales.toFixed(2), u.totalDiscount.toFixed(2), u.averageSale.toFixed(2), u.totalItemsSold,
   218→    ])
   219→    const csv = [headers.join(','), ...rows.map(r => r.join(','))].join('\n')
   220→    const blob = new Blob([csv], { type: 'text/csv' })
   221→    const url = URL.createObjectURL(blob)
   222→    const a = document.createElement('a')
   223→    a.href = url
   224→    a.download = `user-sales-report-${new Date().toISOString().slice(0, 10)}.csv`
   225→    a.click()
   226→    URL.revokeObjectURL(url)
   227→    addToast({ title: 'Exported', description: 'User sales report exported as CSV', variant: 'success' })
   228→  }, [userSalesData, addToast])
   229→
   230→  return (
   231→    <div className="space-y-4">
   232→      {/* Report Type Tabs */}
   233→      <Tabs value={activeTab} onValueChange={setActiveTab}>
   234→        <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-2">
   235→          <TabsList className="flex-wrap">
   236→            <TabsTrigger value="sales">Sales Summary</TabsTrigger>
   237→            <TabsTrigger value="user-sales">User Sales</TabsTrigger>
   238→            <TabsTrigger value="inventory">Inventory</TabsTrigger>
   239→            <TabsTrigger value="prescriptions">Prescriptions</TabsTrigger>
   240→            <TabsTrigger value="stocktake">Stock Take</TabsTrigger>
   241→          </TabsList>
   242→          <div className="flex items-center gap-2 flex-wrap">
   243→            <div className="flex items-center gap-2">
   244→              <Label className="text-xs">From:</Label>
   245→              <Input type="date" value={dateFrom} onChange={(e) => setDateFrom(e.target.value)} className="h-8 w-36 text-xs" />
   246→              <Label className="text-xs">To:</Label>
   247→              <Input type="date" value={dateTo} onChange={(e) => setDateTo(e.target.value)} className="h-8 w-36 text-xs" />
   248→            </div>
   249→            <Button variant="outline" size="sm" onClick={() => addToast({ title: 'Export', description: 'Report exported as CSV' })}>
   250→              <Download className="h-3.5 w-3.5 mr-1" /> Export
   251→            </Button>
   252→          </div>
   253→        </div>
   254→
   255→        {/* Sales Summary Tab */}
   256→        <TabsContent value="sales" className="space-y-4">
   257→          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   258→            <Card>
   259→              <CardContent className="p-4 flex items-center gap-3">
   260→                <div className="h-10 w-10 rounded-lg bg-emerald-100 flex items-center justify-center">
   261→                  <DollarSign className="h-5 w-5 text-emerald-600" />
   262→                </div>
   263→                <div>
   264→                  <div className="text-2xl font-bold">
   265→                    {loading ? <Skeleton className="h-8 w-24" /> : formatCurrency(salesStats?.today?.sales || 0)}
   266→                  </div>
   267→                  <p className="text-xs text-muted-foreground">Today&apos;s Sales</p>
   268→                </div>
   269→              </CardContent>
   270→            </Card>
   271→            <Card>
   272→              <CardContent className="p-4 flex items-center gap-3">
   273→                <div className="h-10 w-10 rounded-lg bg-teal-100 flex items-center justify-center">
   274→                  <ShoppingCart className="h-5 w-5 text-teal-600" />
   275→                </div>
   276→                <div>
   277→                  <div className="text-2xl font-bold">
   278→                    {loading ? <Skeleton className="h-8 w-12" /> : salesStats?.today?.count || 0}
   279→                  </div>
   280→                  <p className="text-xs text-muted-foreground">Transactions Today</p>
   281→                </div>
   282→              </CardContent>
   283→            </Card>
   284→            <Card>
   285→              <CardContent className="p-4 flex items-center gap-3">
   286→                <div className="h-10 w-10 rounded-lg bg-green-100 flex items-center justify-center">
   287→                  <TrendingUp className="h-5 w-5 text-green-600" />
   288→                </div>
   289→                <div>
   290→                  <div className="text-2xl font-bold">
   291→                    {loading ? <Skeleton className="h-8 w-24" /> : formatCurrency(salesStats?.thisWeek?.sales || 0)}
   292→                  </div>
   293→                  <p className="text-xs text-muted-foreground">This Week</p>
   294→                </div>
   295→              </CardContent>
   296→            </Card>
   297→            <Card>
   298→              <CardContent className="p-4 flex items-center gap-3">
   299→                <div className="h-10 w-10 rounded-lg bg-emerald-100 flex items-center justify-center">
   300→                  <CalendarDays className="h-5 w-5 text-emerald-600" />
   301→                </div>
   302→                <div>
   303→                  <div className="text-2xl font-bold">
   304→                    {loading ? <Skeleton className="h-8 w-24" /> : formatCurrency(salesStats?.thisMonth?.sales || 0)}
   305→                  </div>
   306→                  <p className="text-xs text-muted-foreground">This Month</p>
   307→                </div>
   308→              </CardContent>
   309→            </Card>
   310→          </div>
   311→
   312→          <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   313→            {/* Daily Sales Trend */}
   314→            <Card>
   315→              <CardHeader className="pb-2">
   316→                <CardTitle className="text-sm font-semibold">Daily Sales Trend (Last 7 Days)</CardTitle>
   317→              </CardHeader>
   318→              <CardContent>
   319→                {loading ? <Skeleton className="h-64 w-full" /> : (
   320→                  <ResponsiveContainer width="100%" height={260}>
   321→                    <BarChart data={dailySales}>
   322→                      <CartesianGrid strokeDasharray="3 3" stroke="#f0f0f0" />
   323→                      <XAxis dataKey="date" tick={{ fontSize: 11 }} />
   324→                      <YAxis tick={{ fontSize: 11 }} tickFormatter={(v) => `$${v}`} />
   325→                      <Tooltip formatter={(value: any) => formatCurrency(value)} />
   326→                      <Bar dataKey="sales" fill="#059669" radius={[4, 4, 0, 0]} />
   327→                    </BarChart>
   328→                  </ResponsiveContainer>
   329→                )}
   330→              </CardContent>
   331→            </Card>
   332→
   333→            {/* Sales by Category (Pie) */}
   334→            <Card>
   335→              <CardHeader className="pb-2">
   336→                <CardTitle className="text-sm font-semibold">Sales by Product</CardTitle>
   337→              </CardHeader>
   338→              <CardContent>
   339→                {loading ? <Skeleton className="h-64 w-full" /> : (
   340→                  <ResponsiveContainer width="100%" height={260}>
   341→                    <PieChart>
   342→                      <Pie data={salesByCategory} cx="50%" cy="50%" innerRadius={50} outerRadius={90} dataKey="revenue" nameKey="name" label={({ name, percent }) => `${name} ${(percent * 100).toFixed(0)}%`} labelLine={false}>
   343→                        {salesByCategory.map((entry: any, index: number) => (
   344→                          <Cell key={`cell-${index}`} fill={entry.fill} />
   345→                        ))}
   346→                      </Pie>
   347→                      <Tooltip formatter={(value: any) => formatCurrency(value)} />
   348→                    </PieChart>
   349→                  </ResponsiveContainer>
   350→                )}
   351→              </CardContent>
   352→            </Card>
   353→          </div>
   354→
   355→          {/* Top Products Table */}
   356→          <Card>
   357→            <CardHeader className="pb-2">
   358→              <CardTitle className="text-sm font-semibold">Top Selling Products</CardTitle>
   359→            </CardHeader>
   360→            <CardContent className="p-0">
   361→              <Table>
   362→                <TableHeader>
   363→                  <TableRow>
   364→                    <TableHead>#</TableHead>
   365→                    <TableHead>Product</TableHead>
   366→                    <TableHead className="text-right">Units Sold</TableHead>
   367→                    <TableHead className="text-right">Revenue</TableHead>
   368→                  </TableRow>
   369→                </TableHeader>
   370→                <TableBody>
   371→                  {loading ? (
   372→                    Array.from({ length: 5 }).map((_, i) => (
   373→                      <TableRow key={i}>
   374→                        {Array.from({ length: 4 }).map((_, j) => (
   375→                          <TableCell key={j}><Skeleton className="h-5 w-full" /></TableCell>
   376→                        ))}
   377→                      </TableRow>
   378→                    ))
   379→                  ) : (
   380→                    salesStats?.topProducts?.map((p: any, i: number) => (
   381→                      <TableRow key={i}>
   382→                        <TableCell className="font-bold">{i + 1}</TableCell>
   383→                        <TableCell className="font-medium text-sm">{p.productName}</TableCell>
   384→                        <TableCell className="text-right">{p._sum?.quantity || 0}</TableCell>
   385→                        <TableCell className="text-right font-medium">{formatCurrency(p._sum?.subtotal || 0)}</TableCell>
   386→                      </TableRow>
   387→                    ))
   388→                  )}
   389→                </TableBody>
   390→              </Table>
   391→            </CardContent>
   392→          </Card>
   393→        </TabsContent>
   394→
   395→        {/* ========== Per-User Sales Analytics Tab ========== */}
   396→        <TabsContent value="user-sales" className="space-y-4">
   397→          {/* Header with user filter for SUPER_ADMIN */}
   398→          {isSuperAdmin && (
   399→            <div className="flex flex-col sm:flex-row items-start sm:items-center justify-between gap-3">
   400→              <div className="flex items-center gap-2">
   401→                <UserCircle className="h-4 w-4 text-muted-foreground" />
   402→                <Label className="text-sm font-medium">Filter by User:</Label>
   403→                <Select value={selectedUserId} onValueChange={setSelectedUserId}>
   404→                  <SelectTrigger className="w-48 h-8 text-sm">
   405→                    <SelectValue placeholder="All Users" />
   406→                  </SelectTrigger>
   407→                  <SelectContent>
   408→                    <SelectItem value="all">All Users</SelectItem>
   409→                    {allUsers.map((u) => (
   410→                      <SelectItem key={u.id} value={u.id}>{u.name} ({u.role.replace('_', ' ')})</SelectItem>
   411→                    ))}
   412→                  </SelectContent>
   413→                </Select>
   414→              </div>
   415→              <Button variant="outline" size="sm" onClick={exportUserSalesCSV}>
   416→                <Download className="h-3.5 w-3.5 mr-1" /> Export CSV
   417→              </Button>
   418→            </div>
   419→          )}
   420→
   421→          {/* Summary KPI Cards */}
   422→          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   423→            <Card>
   424→              <CardContent className="p-4 flex items-center gap-3">
   425→                <div className="h-10 w-10 rounded-lg bg-emerald-100 flex items-center justify-center">
   426→                  <DollarSign className="h-5 w-5 text-emerald-600" />
   427→                </div>
   428→                <div>
   429→                  <div className="text-2xl font-bold">
   430→                    {userSalesLoading ? <Skeleton className="h-8 w-24" /> : formatCurrency(salesSummary?.totalSales || 0)}
   431→                  </div>
   432→                  <p className="text-xs text-muted-foreground">
   433→                    {isSuperAdmin
   434→                      ? (selectedUserId === 'all' ? 'Total Sales (All Users)' : 'User Total Sales')
   435→                      : 'My Total Sales'}
   436→                  </p>
   437→                </div>
   438→              </CardContent>
   439→            </Card>
   440→            <Card>
   441→              <CardContent className="p-4 flex items-center gap-3">
   442→                <div className="h-10 w-10 rounded-lg bg-teal-100 flex items-center justify-center">
   443→                  <ShoppingCart className="h-5 w-5 text-teal-600" />
   444→                </div>
   445→                <div>
   446→                  <div className="text-2xl font-bold">
   447→                    {userSalesLoading ? <Skeleton className="h-8 w-12" /> : salesSummary?.totalTransactions || 0}
   448→                  </div>
   449→                  <p className="text-xs text-muted-foreground">Total Transactions</p>
   450→                </div>
   451→              </CardContent>
   452→            </Card>
   453→            <Card>
   454→              <CardContent className="p-4 flex items-center gap-3">
   455→                <div className="h-10 w-10 rounded-lg bg-sky-100 flex items-center justify-center">
   456→                  <TrendingUp className="h-5 w-5 text-sky-600" />
   457→                </div>
   458→                <div>
   459→                  <div className="text-2xl font-bold">
   460→                    {userSalesLoading ? <Skeleton className="h-8 w-24" /> : formatCurrency(salesSummary?.averageTransaction || 0)}
   461→                  </div>
   462→                  <p className="text-xs text-muted-foreground">Avg Transaction</p>
   463→                </div>
   464→              </CardContent>
   465→            </Card>
   466→            <Card>
   467→              <CardContent className="p-4 flex items-center gap-3">
   468→                <div className="h-10 w-10 rounded-lg bg-amber-100 flex items-center justify-center">
   469→                  <Users className="h-5 w-5 text-amber-600" />
   470→                </div>
   471→                <div>
   472→                  <div className="text-2xl font-bold">
   473→                    {userSalesLoading ? <Skeleton className="h-8 w-12" /> : userSalesData.length}
   474→                  </div>
   475→                  <p className="text-xs text-muted-foreground">
   476→                    {isSuperAdmin ? 'Active Users' : 'Data Source'}
   477→                  </p>
   478→                </div>
   479→              </CardContent>
   480→            </Card>
   481→          </div>
