   192→   192→
   193→   193→  return (

   211→   211→              />
   212→   212→            </div>

   214→   214→              <Label className="text-xs">Description</Label>

   216→   216→                placeholder="Brief description of this category"
   217→   217→                value={newDesc}

   232→   232→          </div>

   234→   234→      </Card>
   235→   235→
   236→   236→      {/* Search */}

   238→   238→        <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   239→   239→        <Input placeholder="Search categories..." value={search} onChange={(e) => setSearch(e.target.value)} className="pl-9" />
   240→   240→      </div>
   241→   241→
   242→   242→      {/* Categories Table */}
   243→   243→      <Card>
   244→   244→        <CardContent className="p-0">
   245→   245→          <Table>
   246→   246→            <TableHeader>
   247→   247→              <TableRow>
   248→   248→                <TableHead>Category Name</TableHead>
   249→   249→                <TableHead className="hidden sm:table-cell">Description</TableHead>
   250→   250→                <TableHead className="text-center">Products</TableHead>
   251→   251→                <TableHead className="text-center">Type</TableHead>
   252→   252→                <TableHead className="text-right">Actions</TableHead>
   253→   253→              </TableRow>
   254→   254→            </TableHeader>
   255→   255→            <TableBody>

   257→   257→                Array.from({ length: 5 }).map((_, i) => (
   258→   258→                  <TableRow key={i}>
   259→   259→                    {Array.from({ length: 5 }).map((_, j) => (

   279→   279→                        <Badge variant="secondary" className="text-xs">{prodCount}</Badge>

   290→   290→                          size="sm" variant="ghost"
   291→   291→                          className="text-red-400 hover:text-red-600 hover:bg-red-50"
   292→   292→                          onClick={() => handleDelete(cat)}
   293→   293→                          disabled={isBuiltIn || prodCount > 0}
   294→   294→                          title={isBuiltIn ? 'System category' : prodCount > 0 ? `${prodCount} products linked` : 'Delete'}
   295→   295→                        >
   296→   296→                          <Trash2 className="h-3.5 w-3.5" />
   297→   297→                        </Button>

   304→   304→          </Table>

   315→   315→  const [categories, setCategories] = useState<Category[]>([])
   316→   316→  const [vendors, setVendors] = useState<Vendor[]>([])
   317→   317→  const [loading, setLoading] = useState(true)
   318→   318→  const [saving, setSaving] = useState(false)
   319→   319→  const [search, setSearch] = useState('')
   320→   320→  const addToast = useAppStore((s) => s.addToast)
   321→   321→
   322→   322→  const [form, setForm] = useState({
   323→   323→    name: '', sku: '', category: 'OTC', dosageForm: '', manufacturer: '', costPrice: '', sellingPrice: '',
   324→   324→    stockQuantity: '0', minStockLevel: '10', expiryDate: '', barcode: '', vendorId: '',
   325→   325→  })
   326→   326→  const [showNewManufacturer, setShowNewManufacturer] = useState(false)
   327→   327→  const [newManufacturerName, setNewManufacturerName] = useState('')

   330→   330→  const [showNewDosageForm, setShowNewDosageForm] = useState(false)

   335→   335→  const fetchData = useCallback(async () => {
   336→   336→    setLoading(true)

   338→   338→      const [prodRes, catRes, venRes] = await Promise.all([
   339→   339→        fetch('/api/products'),
   340→   340→        fetch('/api/categories'),
   341→   341→        fetch('/api/vendors'),
   342→   342→      ])

   345→   345→        const list = Array.isArray(data) ? data : data.products || []
   346→   346→        setDrugs(list)
   347→   347→      }
   348→   348→      if (catRes.ok) setCategories(await catRes.json())
   349→   349→      if (venRes.ok) setVendors(await venRes.json())
   350→   350→    } catch {
   351→   351→      addToast({ title: 'Error', description: 'Failed to load data', variant: 'destructive' })
   352→   352→    } finally {
   353→   353→      setLoading(false)
   354→   354→    }
   355→   355→  }, [addToast])
   356→   356→
   357→   357→  useEffect(() => { fetchData() }, [fetchData])
   358→   358→
   359→   359→  const handleCreate = async () => {
   360→   360→    if (!form.name || !form.sellingPrice) return
   361→   361→    setSaving(true)
   362→   362→    try {
   363→   363→      const res = await fetch('/api/products', {
   364→   364→        method: 'POST',
   365→   365→        headers: {
   366→   366→          'Content-Type': 'application/json',
   367→   367→          'x-user-role': 'SUPER_ADMIN',
   368→   368→        },
   369→   369→        body: JSON.stringify({
   370→   370→          name: form.name,
   371→   371→          ndc: form.sku || null,
   372→   372→          category: form.category,
   373→   373→          dosageForm: form.dosageForm || null,
   374→   374→          manufacturer: form.manufacturer || null,
   375→   375→          costPrice: form.costPrice ? parseFloat(form.costPrice) : null,
   376→   376→          sellingPrice: parseFloat(form.sellingPrice),
   377→   377→          reorderPoint: parseInt(form.minStockLevel) || 10,
   378→   378→          expiryDate: form.expiryDate || null,
   379→   379→          batchNumber: form.barcode || null,
   380→   380→          vendorId: form.vendorId || null,
   381→   381→        }),
   382→   382→      })
   383→   383→      if (!res.ok) {
   384→   384→        const err = await res.json()
   385→   385→        throw new Error(err.error || 'Failed to create product')
   386→   386→      }
   387→   387→      const newProduct = await res.json()
   388→   388→
   389→   389→      // Set initial stock
   390→   390→      const qty = parseInt(form.stockQuantity) || 0
   391→   391→      if (qty > 0) {
   392→   392→        await fetch('/api/inventory', {
   393→   393→          method: 'PUT',
   394→   394→          headers: { 'Content-Type': 'application/json' },
   395→   395→          body: JSON.stringify({ productId: newProduct.id, adjustment: qty, reason: 'Initial stock on creation' }),
   396→   396→        })
   397→   397→      }
   398→   398→
   399→   399→      addToast({ title: 'Drug Added', description: `"${form.name}" registered in inventory`, variant: 'success' })
   400→   400→      setForm({
   401→   401→        name: '', sku: '', category: 'OTC', dosageForm: '', manufacturer: '', costPrice: '', sellingPrice: '',
   402→   402→        stockQuantity: '0', minStockLevel: '10', expiryDate: '', barcode: '', vendorId: '',
   403→   403→      })
   404→   404→      setShowNewManufacturer(false)
   405→   405→      setNewManufacturerName('')
   406→   406→      fetchData()
   407→   407→    } catch (err: any) {
   408→   408→      addToast({ title: 'Error', description: err.message || 'Failed to add drug', variant: 'destructive' })
   409→   409→    } finally {
   410→   410→      setSaving(false)
   411→   411→    }
   412→   412→  }
   413→   413→
   414→   414→  const filtered = drugs.filter((d) =>
   415→   415→    d.name.toLowerCase().includes(search.toLowerCase()) ||
   416→   416→    (d.ndc && d.ndc.toLowerCase().includes(search.toLowerCase()))
   417→   417→  )
   418→   418→
   419→   419→  // Extract unique manufacturers from existing drugs
   420→   420→  const existingManufacturers = [...new Set(drugs.map((d) => d.manufacturer).filter(Boolean))] as string[]
   421→   421→
   422→   422→  const handleAddNewManufacturer = () => {
   423→   423→    const trimmed = newManufacturerName.trim()
   424→   424→    if (!trimmed) return
   425→   425→    setForm({ ...form, manufacturer: trimmed })
   426→   426→    setNewManufacturerName('')
   427→   427→    setShowNewManufacturer(false)
   428→   428→  }
   429→   429→
   430→   430→  const handleAddNewVendor = async () => {
   431→   431→    const trimmed = newVendorName.trim()
   432→   432→    if (!trimmed) return
   433→   433→    try {
   434→   434→      const res = await fetch('/api/vendors', {
   435→   435→        method: 'POST',
   436→   436→        headers: { 'Content-Type': 'application/json' },
   437→   437→        body: JSON.stringify({ name: trimmed }),
   438→   438→      })
   439→   439→      if (res.ok) {
   440→   440→        const vendor = await res.json()
   441→   441→        setForm({ ...form, vendorId: vendor.id })
   442→   442→        setVendors((prev) => [...prev, { id: vendor.id, name: trimmed, contactPerson: null, email: null, phone: null, address: null, notes: null }])
   443→   443→        setNewVendorName('')
   444→   444→        setShowNewVendor(false)
