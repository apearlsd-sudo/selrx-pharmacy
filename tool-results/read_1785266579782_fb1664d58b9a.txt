    92→    92→      if (res.ok) {

    95→    95→      }
    96→    96→    } catch {

    98→    98→    } finally {
    99→    99→      setLoading(false)
   100→   100→    }
   101→   101→  }, [searchQuery, categoryFilter, addToast])
   102→   102→
   103→   103→  const fetchCategories = useCallback(async () => {
   104→   104→    try {
   105→   105→      const res = await fetch('/api/categories')
   106→   106→      if (res.ok) setCategories(await res.json())

   109→   109→    }
   110→   110→  }, [])

   112→   112→  useEffect(() => { fetchInventory(); fetchCategories() }, [fetchInventory, fetchCategories])
   113→   113→
   114→   114→  const filteredItems = items.filter((item) => {
   115→   115→    const q = Number(item.quantity) || 0
   116→   116→    const r = Number(item.product.reorderPoint) || 10
   117→   117→    if (stockFilter === 'LOW') return q > 0 && q <= r
   118→   118→    if (stockFilter === 'OUT') return q === 0
   119→   119→    if (stockFilter === 'OK') return q > r
   120→   120→    return true
   121→   121→  }).sort((a, b) => {
   122→   122→    if (sortBy === 'name') return sortDir === 'asc' ? a.product.name.localeCompare(b.product.name) : b.product.name.localeCompare(a.product.name)
   123→   123→    if (sortBy === 'stock') return sortDir === 'asc' ? (Number(a.quantity)||0) - (Number(b.quantity)||0) : (Number(b.quantity)||0) - (Number(a.quantity)||0)
   124→   124→    return sortDir === 'asc' ? a.product.category.localeCompare(b.product.category) : b.product.category.localeCompare(a.product.category)
   125→   125→  })

   127→   127→  const lowStockCount = items.filter((i) => { const q = Number(i.quantity) || 0; const r = Number(i.product.reorderPoint) || 10; return q > 0 && q <= r }).length
   128→   128→  const outOfStockCount = items.filter((i) => (Number(i.quantity) || 0) === 0).length

   134→   134→      const isSet = adjustType === 'SET'
   135→   135→      const adj = adjustAmount ? (adjustType === 'ADD' ? parseInt(adjustAmount) : adjustType === 'REMOVE' ? -parseInt(adjustAmount) : parseInt(adjustAmount)) : 0
   136→   136→      const body: Record<string, any> = {
   137→   137→        productId: selectedItem.productId,
   138→   138→        adjustmentType: adjustType,
   139→   139→        reason: adjustReason,
   140→   140→      }
   141→   141→      if (isSet) {
   142→   142→        body.setQuantity = adj
   143→   143→        body.adjustment = 0
   144→   144→      } else {
   145→   145→        body.adjustment = adj
   146→   146→      }
   147→   147→      if (adjustCostPrice !== '') body.costPrice = parseFloat(adjustCostPrice)
   148→   148→      if (adjustSellingPrice !== '') body.sellingPrice = parseFloat(adjustSellingPrice)

   151→   151→        method: 'PUT',

   153→   153→        body: JSON.stringify(body),
   154→   154→      })

   156→   156→        const err = await res.json()

   166→   166→      bumpInventoryVersion()
   167→   167→    } catch (err: any) {
   168→   168→      addToast({ title: 'Error', description: err.message || 'Failed to adjust stock', variant: 'destructive' })
   169→   169→    }
   170→   170→  }

   177→   177→        method: 'POST',

   179→   179→          'Content-Type': 'application/json',
   180→   180→          'x-user-role': currentUser?.role || 'SUPER_ADMIN',

   183→   183→          name: productForm.name,

   185→   185→          category: productForm.category,

   188→   188→          reorderPoint: parseInt(productForm.minStockLevel) || 10,
   189→   189→          expiryDate: productForm.expiryDate || null,
   190→   190→          batchNumber: productForm.barcode || null,
   191→   191→        }),
   192→   192→      })

   195→   195→        throw new Error(err.error || 'Failed to create product')
   196→   196→      }
   197→   197→      const newProduct = await res.json()
   198→   198→
   199→   199→      // Set initial stock quantity if specified
   200→   200→      const qty = parseInt(productForm.stockQuantity)
   201→   201→      if (qty > 0) {
   202→   202→        await fetch('/api/inventory', {
   203→   203→          method: 'PUT',
   204→   204→          headers: { 'Content-Type': 'application/json' },
   205→   205→          body: JSON.stringify({
   206→   206→            productId: newProduct.id,
   207→   207→            adjustment: qty,
   208→   208→            reason: 'Initial stock on product creation',
   209→   209→          }),
   210→   210→        })
   211→   211→      }
   212→   212→
   213→   213→      addToast({ title: 'Product Added', description: `${productForm.name} has been added to inventory`, variant: 'success' })
   214→   214→      setAddProductDialog(false)
   215→   215→      setProductForm({ name: '', sku: '', category: 'OTC', price: '', stockQuantity: '', minStockLevel: '10', expiryDate: '', barcode: '' })
   216→   216→      fetchInventory()
   217→   217→    } catch (err: any) {
   218→   218→      addToast({ title: 'Error', description: err.message || 'Failed to add product', variant: 'destructive' })

   220→   220→      setSavingProduct(false)
   221→   221→    }
   222→   222→  }

   229→   229→      const params = new URLSearchParams({ search: query, limit: '15' })
   230→   230→      const res = await fetch(`/api/products?${params}`)
   231→   231→      if (res.ok) {
   232→   232→        const data = await res.json()
   233→   233→        const productList = data.products || data
   234→   234→        // Merge current inventory quantities
   235→   235→        setStockSearchResults(
   236→   236→          (Array.isArray(productList) ? productList : []).map((p: any) => {
   237→   237→            const inv = items.find(i => i.productId === p.id)

   239→   239→              id: p.id,
   240→   240→              name: p.name,
   241→   241→              ndc: p.ndc || null,
   242→   242→              unitOfMeasure: p.unitOfMeasure || 'EA',
   243→   243→              currentQty: inv ? (Number(inv.quantity) || 0) : (p.inventory ? (Number(p.inventory.quantity) || 0) : 0),
   244→   244→              costPrice: p.costPrice,
   245→   245→              sellingPrice: p.sellingPrice,
   246→   246→            }
   247→   247→          })
   248→   248→        )
   249→   249→      }
   250→   250→    } catch { /* silent */ } finally {
   251→   251→      setStockSearching(false)
   252→   252→    }
   253→   253→  }, [items])
   254→   254→
   255→   255→  // Debounced search
   256→   256→  useEffect(() => {

   259→   259→    return () => clearTimeout(timer)
   260→   260→  }, [stockSearch, stockCountDialog, searchStockProducts])
   261→   261→

   263→   263→    if (stockEntries.find(e => e.productId === product.id)) return
   264→   264→    setStockEntries(prev => [...prev, {
   265→   265→      productId: product.id,
   266→   266→      name: product.name,
   267→   267→      ndc: product.ndc,
   268→   268→      currentQty: product.currentQty,
   269→   269→      physicalQty: String(product.currentQty),
   270→   270→      unit: product.unitOfMeasure,
   271→   271→      costPrice: product.costPrice ? String(product.costPrice) : '',
   272→   272→      sellingPrice: product.sellingPrice ? String(product.sellingPrice) : '',
   273→   273→    }])
   274→   274→    setStockSearch('')
   275→   275→    setStockSearchResults([])
   276→   276→  }
   277→   277→
   278→   278→  const removeStockEntry = (productId: string) => {
   279→   279→    setStockEntries(prev => prev.filter(e => e.productId !== productId))
   280→   280→  }
   281→   281→
   282→   282→  const updateStockEntry = (productId: string, field: 'physicalQty' | 'costPrice' | 'sellingPrice', value: string) => {

   284→   284→  }
   285→   285→
   286→   286→  const handleStockCountSave = async () => {
   287→   287→    if (stockEntries.length === 0) return
   288→   288→    setStockSaving(true)
   289→   289→    try {
   290→   290→      let updated = 0
   291→   291→      let failed = 0
   292→   292→      const results: { name: string; requested: number; confirmed: number }[] = []
   293→   293→      for (const entry of stockEntries) {
   294→   294→        const physicalQty = parseInt(entry.physicalQty)
   295→   295→        if (isNaN(physicalQty)) continue
   296→   296→        const body: Record<string, any> = {
   297→   297→          productId: entry.productId,
   298→   298→          adjustmentType: 'SET',
   299→   299→          setQuantity: physicalQty,
   300→   300→          adjustment: 0,
   301→   301→          reason: 'Physical stock count',
   302→   302→        }
   303→   303→        // Include price updates if changed
   304→   304→        const cp = parseFloat(entry.costPrice)
   305→   305→        if (!isNaN(cp) && cp >= 0) body.costPrice = cp
   306→   306→        const sp = parseFloat(entry.sellingPrice)
   307→   307→        if (!isNaN(sp) && sp >= 0) body.sellingPrice = sp
   308→   308→        try {
   309→   309→          const res = await fetch('/api/inventory', {
   310→   310→            method: 'PUT',
   311→   311→            headers: { 'Content-Type': 'application/json', 'x-user-role': currentUser?.role || 'SUPER_ADMIN' },
   312→   312→            body: JSON.stringify(body),
   313→   313→          })

   315→   315→            const json = await res.json()
   316→   316→            const confirmed = Number(json.newQuantity) || Number(json.inventory?.quantity) || 0
   317→   317→            results.push({ name: entry.name, requested: physicalQty, confirmed })
   318→   318→            if (confirmed === physicalQty) {
   319→   319→              updated++
   320→   320→            } else {
   321→   321→              console.error(`Stock count mismatch for ${entry.name}: requested ${physicalQty}, DB confirmed ${confirmed}`)
   322→   322→              failed++
   323→   323→            }
   324→   324→          } else {
   325→   325→            const err = await res.json().catch(() => ({ error: 'Unknown error' }))
   326→   326→            console.error(`Stock count failed for ${entry.name}:`, err)
   327→   327→            failed++
   328→   328→          }
   329→   329→        } catch (e) {
   330→   330→          console.error(`Stock count network error for ${entry.name}:`, e)
   331→   331→          failed++
   332→   332→        }
   333→   333→      }
   334→   334→      console.log('[Stock Count Save] Results:', results)
   335→   335→      if (updated > 0) {
   336→   336→        addToast({
   337→   337→          title: 'Stock Count Saved',
   338→   338→          description: `${updated} product${updated !== 1 ? 's' : ''} updated to physical count${failed > 0 ? ` (${failed} failed)` : ''}`,
   339→   339→          variant: updated === stockEntries.length ? 'success' : 'default',
   340→   340→        })
   341→   341→      } else {
   342→   342→        addToast({ title: 'Save Failed', description: `All ${failed} product updates failed. Check console for details.`, variant: 'destructive' })
   343→   343→      }
   344→   344→      setStockCountDialog(false)
   345→   345→      setStockEntries([])
   346→   346→      setStockSearch('')
   347→   347→      setStockSearchResults([])
   348→   348→      // Wait for Turso read replica to catch up, then force-refresh

   350→   350→      await fetchInventory(true)
   351→   351→      // Notify all other views (POS, Dashboard, Master Data, Reports) to re-fetch
   352→   352→      bumpInventoryVersion()
   353→   353→    } catch (err: any) {
   354→   354→      addToast({ title: 'Error', description: err.message || 'Failed to save stock count', variant: 'destructive' })
   355→   355→    } finally {
   356→   356→      setStockSaving(false)
   357→   357→    }
   358→   358→  }
   359→   359→
   360→   360→  return (
   361→   361→    <div className="space-y-4">
   362→   362→      {/* Stats Cards */}
   363→   363→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   364→   364→        <Card>
   365→   365→          <CardContent className="p-4 flex items-center gap-3">
   366→   366→            <div className="h-10 w-10 rounded-lg bg-emerald-100 flex items-center justify-center">
   367→   367→              <Package className="h-5 w-5 text-emerald-600" />
   368→   368→            </div>
   369→   369→            <div>
   370→   370→              <p className="text-2xl font-bold">{items.length}</p>
   371→   371→              <p className="text-xs text-muted-foreground">Total Products</p>
   372→   372→            </div>
   373→   373→          </CardContent>
   374→   374→        </Card>
   375→   375→        <Card>
   376→   376→          <CardContent className="p-4 flex items-center gap-3">
   377→   377→            <div className="h-10 w-10 rounded-lg bg-red-100 flex items-center justify-center">
   378→   378→              <AlertTriangle className="h-5 w-5 text-red-600" />
   379→   379→            </div>
   380→   380→            <div>
   381→   381→              <p className="text-2xl font-bold text-red-600">{lowStockCount}</p>
   382→   382→              <p className="text-xs text-muted-foreground">Low Stock Alerts</p>
   383→   383→            </div>
   384→   384→          </CardContent>
   385→   385→        </Card>
   386→   386→        <Card>
   387→   387→          <CardContent className="p-4 flex items-center gap-3">
   388→   388→            <div className="h-10 w-10 rounded-lg bg-amber-100 flex items-center justify-center">
   389→   389→              <AlertTriangle className="h-5 w-5 text-amber-600" />
   390→   390→            </div>
   391→   391→            <div>
   392→   392→              <p className="text-2xl font-bold text-amber-600">{outOfStockCount}</p>
   393→   393→              <p className="text-xs text-muted-foreground">Out of Stock</p>
   394→   394→            </div>
   395→   395→          </CardContent>
   396→   396→        </Card>
   397→   397→        <Card>
   398→   398→          <CardContent className="p-4 flex items-center gap-3">
   399→   399→            <div className="h-10 w-10 rounded-lg bg-teal-100 flex items-center justify-center">
   400→   400→              <TrendingUp className="h-5 w-5 text-teal-600" />
   401→   401→            </div>
   402→   402→            <div>
   403→   403→              <p className="text-2xl font-bold">${totalValue.toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 2 })}</p>
   404→   404→              <p className="text-xs text-muted-foreground">Inventory Value</p>
   405→   405→            </div>
   406→   406→          </CardContent>
   407→   407→        </Card>
   408→   408→      </div>
   409→   409→
   410→   410→      {/* Filters & Actions */}
   411→   411→      <Card>
   412→   412→        <CardContent className="p-4">
   413→   413→          <div className="flex flex-col sm:flex-row gap-3">
   414→   414→            <div className="relative flex-1">
   415→   415→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   416→   416→              <Input
   417→   417→                placeholder="Search by product name or NDC..."
   418→   418→                value={searchQuery}
   419→   419→                onChange={(e) => setSearchQuery(e.target.value)}
   420→   420→                className="pl-9"
   421→   421→              />
   422→   422→            </div>
