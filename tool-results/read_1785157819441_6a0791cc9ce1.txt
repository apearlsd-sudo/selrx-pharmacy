    68→function parsePermissions(raw: string | null): string[] {
    69→  if (!raw) return []
    70→  try {
    71→    const parsed = JSON.parse(raw)
    72→    return Array.isArray(parsed) ? parsed : []
    73→  } catch {
    74→    return []
    75→  }
    76→}

    78→// ── Permission Checkboxes Component ────────────────────────────────────
    79→function PermissionGrid({
    80→  permissions,
    81→  onChange,
    82→  disabled,
    83→}: {
    84→  permissions: string[]
    85→  onChange: (updated: string[]) => void
    86→  disabled?: boolean
    87→}) {
    88→  const toggle = (key: string) => {
    89→    if (disabled) return
    90→    if (permissions.includes(key)) {
    91→      onChange(permissions.filter((p) => p !== key))
    92→    } else {
    93→      onChange([...permissions, key])
    94→    }
    95→  }
    96→
    97→  const selectAll = () => {
    98→    if (disabled) return
    99→    onChange(PERMISSIONS.map((p) => p.key))
   100→  }
   101→
   102→  const deselectAll = () => {
   103→    if (disabled) return
   104→    onChange([])
   105→  }
   106→
   107→  const allSelected = PERMISSIONS.every((p) => permissions.includes(p.key))
   108→
   109→  return (
   110→    <div className="space-y-3">
   111→      <div className="flex items-center justify-between">
   112→        <Label className="text-sm font-semibold">Privileges & Permissions</Label>
   113→        <div className="flex gap-1">
   114→          <Button type="button" variant="outline" size="sm" className="h-6 text-xs" onClick={selectAll} disabled={disabled}>
   115→            Select All
   116→          </Button>
   117→          <Button type="button" variant="outline" size="sm" className="h-6 text-xs" onClick={deselectAll} disabled={disabled}>
   118→            Clear All
   119→          </Button>
   120→        </div>
   121→      </div>
   122→      <div className="border rounded-lg divide-y">
   123→        {PERMISSIONS.map((perm) => {
   124→          const isChecked = permissions.includes(perm.key)
   125→          const Icon = perm.icon
   126→          return (
   127→            <label
   128→              key={perm.key}
   129→              className={`flex items-center gap-3 px-3 py-2.5 cursor-pointer transition-colors ${
   130→                disabled ? 'opacity-60 cursor-not-allowed' : isChecked ? 'bg-emerald-50/60 hover:bg-emerald-50' : 'hover:bg-muted/50'
   131→              }`}
   132→            >
   133→              <Checkbox
   134→                checked={isChecked}
   135→                onCheckedChange={() => toggle(perm.key)}
   136→                disabled={disabled}
   137→                className={isChecked ? 'data-[state=checked]:bg-emerald-600 data-[state=checked]:border-emerald-600' : ''}
   138→              />
   139→              <Icon className={`h-4 w-4 shrink-0 ${isChecked ? 'text-emerald-600' : 'text-muted-foreground'}`} />
   140→              <div className="flex-1 min-w-0">
   141→                <span className={`text-sm font-medium ${isChecked ? 'text-emerald-700' : ''}`}>{perm.label}</span>
   142→                <span className="text-xs text-muted-foreground ml-2 hidden sm:inline">{perm.description}</span>
   143→              </div>
   144→              <Badge variant={isChecked ? 'outline' : 'secondary'} className={`text-[10px] h-5 ${isChecked ? 'border-emerald-300 text-emerald-700' : ''}`}>
   145→                {isChecked ? 'Granted' : 'Denied'}
   146→              </Badge>
   147→            </label>
   148→          )
   149→        })}
   150→      </div>
   151→      <div className="flex items-center justify-between text-xs text-muted-foreground">
   152→        <span>{permissions.length} of {PERMISSIONS.length} permissions granted</span>
   153→        {!allSelected && permissions.length > 0 && (
   154→          <span className="text-amber-600 font-medium">Custom permissions (different from role defaults)</span>
   155→        )}
   156→      </div>
   157→    </div>
   158→  )
   159→}

   161→// ── Main View ──────────────────────────────────────────────────────────
   162→export function UsersView() {
   163→  const [users, setUsers] = useState<UserItem[]>([])
   164→  const [loading, setLoading] = useState(true)
   165→  const [createDialog, setCreateDialog] = useState(false)
   166→  const [editDialog, setEditDialog] = useState(false)
   167→  const [detailDialog, setDetailDialog] = useState(false)
   168→  const [selectedUser, setSelectedUser] = useState<UserItem | null>(null)
   169→  const [form, setForm] = useState({
   170→    name: '', email: '', password: '', role: 'CLERK', phone: '', licenseNumber: '',
   171→  })
   172→  const [formPermissions, setFormPermissions] = useState<string[]>([])
   173→  const [editForm, setEditForm] = useState({
   174→    role: '', phone: '', licenseNumber: '',
   175→  })
   176→  const [editPermissions, setEditPermissions] = useState<string[]>([])
   177→  const [saving, setSaving] = useState(false)
   178→  const addToast = useAppStore((s) => s.addToast)
   179→  const currentUser = useAppStore((s) => s.user)
   180→
   181→  const fetchUsers = useCallback(async () => {
   182→    setLoading(true)
   183→    try {
   184→      const res = await fetch('/api/users')
   185→      if (res.ok) setUsers(await res.json())
   186→    } catch {
   187→      addToast({ title: 'Error', description: 'Failed to load users', variant: 'destructive' })
   188→    } finally {
   189→      setLoading(false)
   190→    }
   191→  }, [addToast])
   192→
   193→  useEffect(() => { fetchUsers() }, [fetchUsers])
   194→
   195→  // When role changes in create dialog, auto-fill default permissions
   196→  const handleCreateRoleChange = (newRole: string) => {
   197→    setForm({ ...form, role: newRole })
   198→    setFormPermissions([...DEFAULT_ROLE_PERMISSIONS[newRole] || []])
   199→  }
   200→
   201→  // When role changes in edit dialog, auto-fill default permissions
   202→  const handleEditRoleChange = (newRole: string) => {
   203→    setEditForm({ ...editForm, role: newRole })
   204→    setEditPermissions([...DEFAULT_ROLE_PERMISSIONS[newRole] || []])
   205→  }
   206→
   207→  const openEditDialog = (user: UserItem) => {
   208→    setSelectedUser(user)
   209→    const parsed = parsePermissions(user.permissions)
   210→    const perms = parsed.length > 0 ? parsed : [...DEFAULT_ROLE_PERMISSIONS[user.role] || []]
   211→    setEditForm({ role: user.role, phone: user.phone || '', licenseNumber: user.licenseNumber || '' })
   212→    setEditPermissions(perms)
   213→    setEditDialog(true)
   214→  }
   215→
   216→  const handleCreate = async () => {
   217→    if (!form.name || !form.email || !form.password) return
   218→    setSaving(true)
   219→    try {
   220→      const res = await fetch('/api/users', {
   221→        method: 'POST',
   222→        headers: { 'Content-Type': 'application/json', 'x-user-role': 'SUPER_ADMIN' },
   223→        body: JSON.stringify({ ...form, userRole: form.role, permissions: formPermissions }),
   224→      })
   225→      if (!res.ok) {
   226→        const err = await res.json()
   227→        throw new Error(err.error || 'Failed to create user')
   228→      }
   229→      addToast({ title: 'Created', description: 'User created successfully', variant: 'success' })
   230→      setCreateDialog(false)
   231→      setForm({ name: '', email: '', password: '', role: 'CLERK', phone: '', licenseNumber: '' })
   232→      setFormPermissions([])
   233→      fetchUsers()
   234→    } catch (err: any) {
   235→      addToast({ title: 'Error', description: err.message || 'Failed to create user', variant: 'destructive' })
   236→    } finally {
   237→      setSaving(false)
   238→    }
   239→  }
   240→
   241→  const handleUpdate = async () => {
   242→    if (!selectedUser) return
   243→    setSaving(true)
   244→    try {
   245→      const res = await fetch(`/api/users?id=${selectedUser.id}`, {
   246→        method: 'PUT',
   247→        headers: { 'Content-Type': 'application/json', 'x-user-role': 'SUPER_ADMIN' },
   248→        body: JSON.stringify({ userRole: editForm.role, phone: editForm.phone, licenseNumber: editForm.licenseNumber, permissions: editPermissions }),
   249→      })
   250→      if (!res.ok) {
   251→        const err = await res.json()
   252→        throw new Error(err.error || 'Failed to update user')
   253→      }
   254→      addToast({ title: 'Updated', description: `${selectedUser.name} has been updated`, variant: 'success' })
   255→      setEditDialog(false)
   256→      setSelectedUser(null)
   257→      fetchUsers()
   258→    } catch (err: any) {
   259→      addToast({ title: 'Error', description: err.message || 'Failed to update user', variant: 'destructive' })
   260→    } finally {
   261→      setSaving(false)
   262→    }
   263→  }
   264→
   265→  const handleToggleActive = async (user: UserItem) => {
   266→    try {
   267→      await fetch(`/api/users?id=${user.id}`, {
   268→        method: 'PUT',
   269→        headers: { 'Content-Type': 'application/json', 'x-user-role': 'SUPER_ADMIN' },
   270→        body: JSON.stringify({ active: !user.active }),
   271→      })
   272→      addToast({ title: user.active ? 'Deactivated' : 'Activated', description: `${user.name} ${user.active ? 'deactivated' : 'activated'}`, variant: 'success' })
   273→      fetchUsers()
   274→    } catch {
   275→      addToast({ title: 'Error', description: 'Failed to update user', variant: 'destructive' })
   276→    }
   277→  }
   278→
   279→  const getUserPermissions = (user: UserItem): string[] => {
   280→    const parsed = parsePermissions(user.permissions)
   281→    return parsed.length > 0 ? parsed : DEFAULT_ROLE_PERMISSIONS[user.role] || []
   282→  }
   283→
   284→  return (
   285→    <div className="space-y-4">
   286→      {/* Summary */}
   287→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   288→        <Card>
   289→          <CardContent className="p-4 flex items-center gap-3">
   290→            <div className="h-10 w-10 rounded-lg bg-emerald-100 flex items-center justify-center">
   291→              <UserCheck className="h-5 w-5 text-emerald-600" />
   292→            </div>
   293→            <div>
   294→              <p className="text-2xl font-bold">{users.length}</p>
   295→              <p className="text-xs text-muted-foreground">Total Users</p>
   296→            </div>
   297→          </CardContent>
   298→        </Card>
   299→        <Card>
   300→          <CardContent className="p-4 flex items-center gap-3">
   301→            <div className="h-10 w-10 rounded-lg bg-green-100 flex items-center justify-center">
   302→              <CheckCircle className="h-5 w-5 text-green-600" />
   303→            </div>
   304→            <div>
   305→              <p className="text-2xl font-bold text-green-600">{users.filter((u) => u.active).length}</p>
   306→              <p className="text-xs text-muted-foreground">Active</p>
   307→            </div>
   308→          </CardContent>
   309→        </Card>
   310→        <Card>
   311→          <CardContent className="p-4 flex items-center gap-3">
   312→            <div className="h-10 w-10 rounded-lg bg-gray-100 flex items-center justify-center">
   313→              <Ban className="h-5 w-5 text-gray-600" />
   314→            </div>
   315→            <div>
   316→              <p className="text-2xl font-bold text-gray-600">{users.filter((u) => !u.active).length}</p>
   317→              <p className="text-xs text-muted-foreground">Inactive</p>
   318→            </div>
   319→          </CardContent>
   320→        </Card>
   321→        <Card>
   322→          <CardContent className="p-4 flex items-center gap-3">
   323→            <div className="h-10 w-10 rounded-lg bg-red-100 flex items-center justify-center">
   324→              <Shield className="h-5 w-5 text-red-600" />
   325→            </div>
   326→            <div>
   327→              <p className="text-2xl font-bold text-red-600">{users.filter((u) => u.role === 'SUPER_ADMIN').length}</p>
   328→              <p className="text-xs text-muted-foreground">Admins</p>
   329→            </div>
   330→          </CardContent>
   331→        </Card>
   332→      </div>
   333→
   334→      {/* Header */}
   335→      <div className="flex items-center justify-between">
   336→        <div>
   337→          <h2 className="text-lg font-semibold">User Management</h2>
   338→          <p className="text-sm text-muted-foreground">Manage staff accounts and role permissions</p>
   339→        </div>
   340→        {(currentUser?.role === 'SUPER_ADMIN') && (
   341→          <Button onClick={() => {
   342→            setForm({ name: '', email: '', password: '', role: 'CLERK', phone: '', licenseNumber: '' })
   343→            setFormPermissions([...DEFAULT_ROLE_PERMISSIONS.CLERK])
   344→            setCreateDialog(true)
   345→          }} className="bg-emerald-600 hover:bg-emerald-700">
   346→            <Plus className="h-4 w-4 mr-2" />
   347→            Add User
   348→          </Button>
   349→        )}
   350→      </div>
   351→
   352→      {/* Users Table */}
   353→      <Card>
   354→        <CardContent className="p-0">
   355→          <Table>
   356→            <TableHeader>
   357→              <TableRow>
   358→                <TableHead>Name</TableHead>
   359→                <TableHead className="hidden sm:table-cell">Email</TableHead>
   360→                <TableHead>Role</TableHead>
   361→                <TableHead className="hidden md:table-cell">Privileges</TableHead>
   362→                <TableHead>Status</TableHead>
   363→                <TableHead className="hidden lg:table-cell">Last Login</TableHead>
   364→                <TableHead className="text-right">Actions</TableHead>
   365→              </TableRow>
   366→            </TableHeader>
   367→            <TableBody>
   368→              {loading ? (
   369→                Array.from({ length: 5 }).map((_, i) => (
   370→                  <TableRow key={i}>
   371→                    {Array.from({ length: 7 }).map((_, j) => (
   372→                      <TableCell key={j}><Skeleton className="h-5 w-full" /></TableCell>
   373→                    ))}
   374→                  </TableRow>
   375→                ))
   376→              ) : (
   377→                users.map((userItem) => {
   378→                  const roleCfg = ROLE_CONFIG[userItem.role] || ROLE_CONFIG.CLERK
   379→                  const perms = getUserPermissions(userItem)
   380→                  const isCustomPerms = parsePermissions(userItem.permissions).length > 0
   381→                  return (
   382→                    <TableRow key={userItem.id} className={!userItem.active ? 'opacity-50' : ''}>
   383→                      <TableCell>
   384→                        <div className="flex items-center gap-3">
   385→                          <div className="h-8 w-8 rounded-full bg-emerald-100 flex items-center justify-center text-xs font-bold text-emerald-700">
   386→                            {userItem.name.split(' ').map((n) => n[0]).join('')}
   387→                          </div>
   388→                          <p className="font-medium text-sm">{userItem.name}</p>
   389→                        </div>
   390→                      </TableCell>
   391→                      <TableCell className="hidden sm:table-cell text-sm text-muted-foreground">{userItem.email}</TableCell>
   392→                      <TableCell>
   393→                        <Badge className={`text-xs ${roleCfg.color}`}>{roleCfg.label}</Badge>
   394→                      </TableCell>
   395→                      <TableCell className="hidden md:table-cell">
   396→                        <div className="flex items-center gap-1 flex-wrap max-w-[200px]">
   397→                          <Badge variant="outline" className="text-[10px] px-1.5 py-0">
   398→                            {perms.length}/{PERMISSIONS.length}
   399→                          </Badge>
   400→                          {isCustomPerms && (
   401→                            <Badge className="text-[10px] px-1.5 py-0 bg-amber-100 text-amber-700 border-amber-200">
   402→                              Custom
   403→                            </Badge>
   404→                          )}
   405→                        </div>
   406→                      </TableCell>
   407→                      <TableCell>
   408→                        {userItem.active ? (
   409→                          <Badge className="bg-green-100 text-green-700 border-green-200 text-xs">Active</Badge>
   410→                        ) : (
   411→                          <Badge className="bg-red-100 text-red-700 border-red-200 text-xs">Inactive</Badge>
   412→                        )}
   413→                      </TableCell>
   414→                      <TableCell className="hidden lg:table-cell text-xs text-muted-foreground">
   415→                        {userItem.lastLogin ? (
   416→                          <div className="flex items-center gap-1">
   417→                            <Clock className="h-3 w-3" />
   418→                            {new Date(userItem.lastLogin).toLocaleString()}
   419→                          </div>
   420→                        ) : (
   421→                          'Never'
   422→                        )}
   423→                      </TableCell>
   424→                      <TableCell className="text-right">
   425→                        <div className="flex items-center justify-end gap-1">
   426→                          <Button size="sm" variant="ghost" onClick={() => { setSelectedUser(userItem); setDetailDialog(true) }}>
   427→                            <Eye className="h-3.5 w-3.5" />
   428→                          </Button>
   429→                          {currentUser?.role === 'SUPER_ADMIN' && userItem.id !== currentUser.id && (
   430→                            <>
   431→                              <Button size="sm" variant="ghost" onClick={() => openEditDialog(userItem)}>
   432→                                <Edit className="h-3.5 w-3.5" />
   433→                              </Button>
   434→                              <Button
   435→                                size="sm"
   436→                                variant="ghost"
   437→                                className={userItem.active ? 'text-red-500' : 'text-emerald-600'}
   438→                                onClick={() => handleToggleActive(userItem)}
   439→                              >
   440→                                {userItem.active ? <Ban className="h-3.5 w-3.5" /> : <CheckCircle className="h-3.5 w-3.5" />}
   441→                              </Button>
   442→                            </>
   443→                          )}
   444→                        </div>
   445→                      </TableCell>
   446→                    </TableRow>
   447→                  )
   448→                })
   449→              )}
   450→            </TableBody>
   451→          </Table>
   452→        </CardContent>
   453→      </Card>
   454→
   455→      {/* Role Permission Reference Matrix */}
   456→      <Card>
   457→        <CardContent className="p-4">
   458→          <h3 className="text-sm font-semibold mb-3">Default Role Permissions</h3>
   459→          <div className="overflow-x-auto">
   460→            <Table>
   461→              <TableHeader>
   462→                <TableRow>
   463→                  <TableHead>Permission</TableHead>
   464→                  <TableHead className="text-center">Super Admin</TableHead>
   465→                  <TableHead className="text-center">Pharmacist</TableHead>
   466→                  <TableHead className="text-center">Technician</TableHead>
   467→                  <TableHead className="text-center">Cashier</TableHead>
   468→                  <TableHead className="text-center">Clerk</TableHead>
   469→                </TableRow>
   470→              </TableHeader>
   471→              <TableBody>
   472→                {PERMISSIONS.map((perm) => (
   473→                  <TableRow key={perm.key}>
   474→                    <TableCell className="font-medium text-sm">{perm.label}</TableCell>
   475→                    {Object.keys(ROLE_CONFIG).map((role) => {
   476→                      const hasPerm = DEFAULT_ROLE_PERMISSIONS[role]?.includes(perm.key)
   477→                      return (
   478→                        <TableCell key={role} className="text-center">
   479→                          {hasPerm ? <CheckCircle className="h-4 w-4 text-emerald-600 mx-auto" /> : <XCircle className="h-4 w-4 text-gray-300 mx-auto" />}
   480→                        </TableCell>
   481→                      )
   482→                    })}
   483→                  </TableRow>
   484→                ))}
   485→              </TableBody>
   486→            </Table>
   487→          </div>
   488→        </CardContent>
   489→      </Card>
   490→
   491→      {/* ── Create User Dialog ─────────────────────────────────────────── */}
   492→      <Dialog open={createDialog} onOpenChange={setCreateDialog}>
   493→        <DialogContent className="max-w-2xl max-h-[90vh]">
   494→          <DialogHeader>
   495→            <DialogTitle className="flex items-center gap-2">
   496→              <Plus className="h-5 w-5 text-emerald-600" />
