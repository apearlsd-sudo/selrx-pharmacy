    47→function formatDate(dateStr: string): string {
    48→  return new Date(dateStr).toLocaleDateString('en-US', {
    49→    month: 'short',
    50→    day: 'numeric',
    51→    year: 'numeric',
    52→    hour: '2-digit',
    53→    minute: '2-digit',
    54→  })
    55→}
    56→
    57→function formatTime(dateStr: string): string {
    58→  return new Date(dateStr).toLocaleTimeString('en-US', {
    59→    hour: '2-digit',
    60→    minute: '2-digit',
    61→  })
    62→}

    71→// Status badge color helper
    72→function statusBadge(status: string) {
    73→  switch (status) {
    74→    case 'COMPLETED':
    75→      return <Badge className="bg-emerald-100 text-emerald-700 text-xs border-emerald-200">Completed</Badge>
    76→    case 'PENDING':
    77→      return <Badge className="bg-amber-100 text-amber-700 text-xs border-amber-200">Pending</Badge>
    78→    case 'VOIDED':
    79→      return <Badge className="bg-red-100 text-red-700 text-xs border-red-200">Voided</Badge>
    80→    case 'REFUNDED':
    81→      return <Badge className="bg-gray-100 text-gray-700 text-xs border-gray-200">Refunded</Badge>
    82→    default:
    83→      return <Badge variant="outline" className="text-xs">{status}</Badge>
    84→  }
    85→}

    88→function paymentBadge(method: string) {

    90→    CASH: 'bg-emerald-100 text-emerald-700 border-emerald-200',
    91→    CREDIT_CARD: 'bg-blue-100 text-blue-700 border-blue-200',

    94→    FSA_HSA: 'bg-amber-100 text-amber-700 border-amber-200',
    95→    SPLIT: 'bg-indigo-100 text-indigo-700 border-indigo-200',

   102→}
