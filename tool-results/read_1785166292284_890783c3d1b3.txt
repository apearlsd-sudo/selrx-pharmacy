   135→function parsePermissions(raw: string | null): string[] {
   136→  if (!raw) return []
   137→  try {
   138→    const parsed = JSON.parse(raw)
   139→    return Array.isArray(parsed) ? parsed : []
   140→  } catch {
   141→    return []
   142→  }
   143→}

   145→// ── Permission Checkboxes Component ────────────────────────────────────
   146→function PermissionGrid({
   147→  permissions,
   148→  onChange,
   149→  disabled,
   150→}: {
   151→  permissions: string[]
   152→  onChange: (updated: string[]) => void
   153→  disabled?: boolean
   154→}) {
   155→  const toggle = (key: string) => {
   156→    if (disabled) return
   157→    if (permissions.includes(key)) {
   158→      onChange(permissions.filter((p) => p !== key))
   159→    } else {
   160→      onChange([...permissions, key])
   161→    }
   162→  }
   163→
   164→  const selectAll = () => {
   165→    if (disabled) return
   166→    onChange(PERMISSIONS.map((p) => p.key))
   167→  }
   168→
   169→  const deselectAll = () => {
   170→    if (disabled) return
   171→    onChange([])
   172→  }
   173→
   174→  const allSelected = ALL_PERMISSIONS.every((p) => permissions.includes(p.key))
   175→
   176→  return (
   177→    <div className="space-y-3">
   178→      <div className="flex items-center justify-between">
   179→        <Label className="text-sm font-semibold">Privileges & Permissions</Label>
   180→        <div className="flex gap-1">
   181→          <Button type="button" variant="outline" size="sm" className="h-6 text-xs" onClick={selectAll} disabled={disabled}>
   182→            Select All
   183→          </Button>
   184→          <Button type="button" variant="outline" size="sm" className="h-6 text-xs" onClick={deselectAll} disabled={disabled}>
   185→            Clear All
   186→          </Button>
   187→        </div>
   188→      </div>
   189→      {PERMISSION_CATEGORIES.map((cat) => {
   190→        const catSelected = cat.permissions.filter(p => permissions.includes(p.key)).length
   191→        return (
   192→          <div key={cat.category} className="border rounded-lg">
   193→            <div className="px-3 py-2 bg-gray-50/50 border-b flex items-center justify-between">
   194→              <span className="text-xs font-semibold text-gray-600 uppercase tracking-wide">{cat.category}</span>
   195→              <Badge variant="outline" className="text-[10px] h-5">{catSelected}/{cat.permissions.length}</Badge>
   196→            </div>
   197→            <div className="divide-y">
   198→              {cat.permissions.map((perm) => {
   199→                const isChecked = permissions.includes(perm.key)
   200→                const Icon = perm.icon
   201→                return (
   202→                  <label
   203→                    key={perm.key}
   204→                    className={`flex items-center gap-3 px-3 py-2.5 cursor-pointer transition-colors ${
   205→                      disabled ? 'opacity-60 cursor-not-allowed' : isChecked ? 'bg-emerald-50/60 hover:bg-emerald-50' : 'hover:bg-muted/50'
   206→                    }`}
   207→                  >
   208→                    <Checkbox
   209→                      checked={isChecked}
   210→                      onCheckedChange={() => toggle(perm.key)}
   211→                      disabled={disabled}
   212→                      className={isChecked ? 'data-[state=checked]:bg-emerald-600 data-[state=checked]:border-emerald-600' : ''}
   213→                    />
   214→                    <Icon className={`h-4 w-4 shrink-0 ${isChecked ? 'text-emerald-600' : 'text-muted-foreground'}`} />
   215→                    <div className="flex-1 min-w-0">
   216→                      <span className={`text-sm font-medium ${isChecked ? 'text-emerald-700' : ''}`}>{perm.label}</span>
   217→                      <span className="text-xs text-muted-foreground ml-2 hidden sm:inline">{perm.description}</span>
   218→                    </div>
   219→                    <Badge variant={isChecked ? 'outline' : 'secondary'} className={`text-[10px] h-5 ${isChecked ? 'border-emerald-300 text-emerald-700' : ''}`}>
   220→                      {isChecked ? 'Granted' : 'Denied'}
   221→                    </Badge>
   222→                  </label>
   223→                )
   224→              })}
   225→            </div>
   226→          </div>
   227→        )
   228→      })}
   229→      <div className="flex items-center justify-between text-xs text-muted-foreground">
   230→        <span>{permissions.length} of {ALL_PERMISSIONS.length} permissions granted</span>
   231→        {!allSelected && permissions.length > 0 && (
   232→          <span className="text-amber-600 font-medium">Custom permissions (different from role defaults)</span>
   233→        )}
   234→      </div>
   235→    </div>
   236→  )
   237→}

   251→  const [editForm, setEditForm] = useState({
   252→    role: '', phone: '', licenseNumber: '',
   253→  })
   254→  const [editPermissions, setEditPermissions] = useState<string[]>([])
   255→  const [saving, setSaving] = useState(false)
   256→  const addToast = useAppStore((s) => s.addToast)
   257→  const currentUser = useAppStore((s) => s.user)
   258→
   259→  // ── Roles Management State ──
   260→  const [activeTab, setActiveTab] = useState<'users' | 'roles'>('users')
   261→  const [roles, setRoles] = useState<SystemRoleItem[]>([])
   262→  const [createRoleDialog, setCreateRoleDialog] = useState(false)
   263→  const [editRoleDialog, setEditRoleDialog] = useState(false)
   264→  const [selectedRole, setSelectedRole] = useState<SystemRoleItem | null>(null)
   265→  const [roleForm, setRoleForm] = useState({ name: '', label: '', description: '' })
   266→  const [roleFormPermissions, setRoleFormPermissions] = useState<string[]>([])
   267→  const [editRoleForm, setEditRoleForm] = useState({ label: '', description: '', color: '' })
   268→  const [editRolePermissions, setEditRolePermissions] = useState<string[]>([])
   269→
   270→  const fetchUsers = useCallback(async () => {
   271→    setLoading(true)
   272→    try {
   273→      const res = await fetch('/api/users')
   274→      if (res.ok) setUsers(await res.json())
   275→    } catch {
   276→      addToast({ title: 'Error', description: 'Failed to load users', variant: 'destructive' })
   277→    } finally {
   278→      setLoading(false)
   279→    }
   280→  }, [addToast])
   281→
   282→  useEffect(() => { fetchUsers() }, [fetchUsers])
   283→
   284→  // ── Roles CRUD ──
   285→  const fetchRoles = useCallback(async () => {
   286→    try {
   287→      const res = await fetch('/api/roles', { headers: { 'x-user-role': 'SUPER_ADMIN' } })
   288→      if (res.ok) {
   289→        const data = await res.json()
   290→        setRoles(data)
   291→        // Update DEFAULT_ROLE_PERMISSIONS from DB roles
   292→        data.forEach((r: SystemRoleItem) => {
   293→          try {
   294→            const perms = JSON.parse(r.permissions)
   295→            if (Array.isArray(perms)) DEFAULT_ROLE_PERMISSIONS[r.name] = perms
   296→          } catch { /* skip */ }
   297→        })
   298→      }
   299→    } catch { /* silent */ }
   300→  }, [])
   301→
   302→  useEffect(() => { fetchRoles() }, [fetchRoles])
   303→
   304→  const handleCreateRole = async () => {
   305→    if (!roleForm.name || !roleForm.label) return
   306→    setSaving(true)
   307→    try {
   308→      const res = await fetch('/api/roles', {
   309→        method: 'POST',
   310→        headers: { 'Content-Type': 'application/json', 'x-user-role': 'SUPER_ADMIN' },
   311→        body: JSON.stringify({ ...roleForm, permissions: roleFormPermissions }),
   312→      })
   313→      if (!res.ok) {
   314→        const err = await res.json()
   315→        throw new Error(err.error || 'Failed to create role')
   316→      }
   317→      addToast({ title: 'Role Created', description: `"${roleForm.label}" role created successfully`, variant: 'success' })
   318→      setCreateRoleDialog(false)
   319→      setRoleForm({ name: '', label: '', description: '' })
   320→      setRoleFormPermissions([])
   321→      fetchRoles()
   322→    } catch (err: any) {
   323→      addToast({ title: 'Error', description: err.message || 'Failed to create role', variant: 'destructive' })
   324→    } finally {
   325→      setSaving(false)
   326→    }
   327→  }
   328→
   329→  const openEditRoleDialog = (role: SystemRoleItem) => {
   330→    setSelectedRole(role)
   331→    setEditRoleForm({ label: role.label, description: role.description || '', color: role.color })
   332→    try { setEditRolePermissions(JSON.parse(role.permissions)) } catch { setEditRolePermissions([]) }
   333→    setEditRoleDialog(true)
   334→  }
   335→
   336→  const handleUpdateRole = async () => {
   337→    if (!selectedRole) return
   338→    setSaving(true)
   339→    try {
   340→      const res = await fetch(`/api/roles?id=${selectedRole.id}`, {
   341→        method: 'PUT',
   342→        headers: { 'Content-Type': 'application/json', 'x-user-role': 'SUPER_ADMIN' },
   343→        body: JSON.stringify({ ...editRoleForm, permissions: editRolePermissions }),
   344→      })
   345→      if (!res.ok) {
   346→        const err = await res.json()
   347→        throw new Error(err.error || 'Failed to update role')
   348→      }
   349→      addToast({ title: 'Role Updated', description: `"${editRoleForm.label}" updated successfully`, variant: 'success' })
   350→      setEditRoleDialog(false)
   351→      setSelectedRole(null)
   352→      fetchRoles()
   353→    } catch (err: any) {
   354→      addToast({ title: 'Error', description: err.message || 'Failed to update role', variant: 'destructive' })
   355→    } finally {
   356→      setSaving(false)
   357→    }
   358→  }
   359→
   360→  const handleDeleteRole = async (role: SystemRoleItem) => {
   361→    if (role.isSystem || role._count.users > 0) {
   362→      addToast({ title: 'Cannot Delete', description: role.isSystem ? 'System roles cannot be deleted' : `Reassign ${role._count.users} user(s) first`, variant: 'destructive' })
   363→      return
   364→    }
   365→    if (!confirm(`Delete the "${role.label}" role? This cannot be undone.`)) return

   367→      const res = await fetch(`/api/roles?id=${role.id}`, {
   368→        method: 'DELETE',
   369→        headers: { 'x-user-role': 'SUPER_ADMIN' },
   370→      })
   371→      if (res.ok) {
   372→        addToast({ title: 'Deleted', description: `"${role.label}" role deleted`, variant: 'success' })
   373→        fetchRoles()
   374→      }
   375→    } catch {
   376→      addToast({ title: 'Error', description: 'Failed to delete role', variant: 'destructive' })
   377→    }
   378→  }

   398→    setEditDialog(true)
   399→  }
   400→
   401→  const handleCreate = async () => {
   402→    if (!form.name || !form.email || !form.password) return
   403→    setSaving(true)
   404→    try {
   405→      const res = await fetch('/api/users', {
   406→        method: 'POST',
   407→        headers: { 'Content-Type': 'application/json', 'x-user-role': 'SUPER_ADMIN' },
   408→        body: JSON.stringify({ ...form, userRole: form.role, permissions: formPermissions }),
   409→      })
   410→      if (!res.ok) {
   411→        const err = await res.json()
   412→        throw new Error(err.error || 'Failed to create user')

   414→      addToast({ title: 'Created', description: 'User created successfully', variant: 'success' })
   415→      setCreateDialog(false)
   416→      setForm({ name: '', email: '', password: '', role: 'CLERK', phone: '', licenseNumber: '' })
   417→      setFormPermissions([])

   420→      addToast({ title: 'Error', description: err.message || 'Failed to create user', variant: 'destructive' })

   422→      setSaving(false)
   423→    }
   424→  }
   425→
   426→  const handleUpdate = async () => {
   427→    if (!selectedUser) return
   428→    setSaving(true)
   429→    try {
   430→      const res = await fetch(`/api/users?id=${selectedUser.id}`, {
   431→        method: 'PUT',
   432→        headers: { 'Content-Type': 'application/json', 'x-user-role': 'SUPER_ADMIN' },
   433→        body: JSON.stringify({ userRole: editForm.role, phone: editForm.phone, licenseNumber: editForm.licenseNumber, permissions: editPermissions }),

   436→        const err = await res.json()

   440→      setEditDialog(false)
   441→      setSelectedUser(null)
   442→      fetchUsers()
   443→    } catch (err: any) {
   444→      addToast({ title: 'Error', description: err.message || 'Failed to update user', variant: 'destructive' })
   445→    } finally {
   446→      setSaving(false)
