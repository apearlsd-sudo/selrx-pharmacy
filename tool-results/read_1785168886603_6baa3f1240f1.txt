    51→    51→  const [loading, setLoading] = useState(true)

    53→    53→  const [categoryFilter, setCategoryFilter] = useState('ALL')
    54→    54→  const [stockFilter, setStockFilter] = useState('ALL')
    55→    55→  const [adjustDialog, setAdjustDialog] = useState(false)

    58→    58→  const [categoryDialog, setCategoryDialog] = useState(false)
    59→    59→  const [categories, setCategories] = useState<{ id: string; name: string; description: string | null; _count?: { products: number } }[]>([])
    60→    60→  const [newCategoryName, setNewCategoryName] = useState('')
    61→    61→  const [newCategoryDesc, setNewCategoryDesc] = useState('')
    62→    62→  const [savingCategory, setSavingCategory] = useState(false)
    63→    63→  const [selectedItem, setSelectedItem] = useState<InventoryItem | null>(null)
    64→    64→  const [adjustType, setAdjustType] = useState('ADD')
    65→    65→  const [adjustAmount, setAdjustAmount] = useState('')
    66→    66→  const [adjustReason, setAdjustReason] = useState('')
    67→    67→  const [shipmentItems, setShipmentItems] = useState([{ productId: '', quantity: '', costPrice: '' }])
    68→    68→  const [sortBy, setSortBy] = useState<'name' | 'stock' | 'category'>('name')
    69→    69→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('asc')
    70→    70→  const [productForm, setProductForm] = useState({
    71→    71→    name: '', sku: '', category: 'OTC', price: '', stockQuantity: '',
    72→    72→    minStockLevel: '10', expiryDate: '', barcode: '',
    73→    73→  })
    74→    74→  const [savingProduct, setSavingProduct] = useState(false)
    75→    75→  const addToast = useAppStore((s) => s.addToast)
    76→    76→  const currentUser = useAppStore((s) => s.user)
    77→    77→
    78→    78→  const fetchInventory = useCallback(async () => {
    79→    79→    setLoading(true)
    80→    80→    try {
    81→    81→      const params = new URLSearchParams()
    82→    82→      if (searchQuery) params.set('search', searchQuery)
    83→    83→      if (categoryFilter !== 'ALL') params.set('category', categoryFilter)
    84→    84→      const res = await fetch(`/api/inventory?${params}`)
    85→    85→      if (res.ok) {

    87→    87→        setItems(data)
    88→    88→      }
    89→    89→    } catch {
    90→    90→      addToast({ title: 'Error', description: 'Failed to load inventory', variant: 'destructive' })
    91→    91→    } finally {
    92→    92→      setLoading(false)
    93→    93→    }
    94→    94→  }, [searchQuery, categoryFilter, addToast])
    95→    95→
    96→    96→  const fetchCategories = useCallback(async () => {
    97→    97→    try {
    98→    98→      const res = await fetch('/api/categories')
    99→    99→      if (res.ok) setCategories(await res.json())
   100→   100→    } catch {
   101→   101→      // silent — categories are optional enhancement
   102→   102→    }
   103→   103→  }, [])
   104→   104→
   105→   105→  useEffect(() => { fetchInventory(); fetchCategories() }, [fetchInventory, fetchCategories])
   106→   106→
   107→   107→  const filteredItems = items.filter((item) => {
   108→   108→    if (stockFilter === 'LOW') return item.quantity <= item.product.reorderPoint
   109→   109→    if (stockFilter === 'OUT') return item.quantity === 0
   110→   110→    if (stockFilter === 'OK') return item.quantity > item.product.reorderPoint
   111→   111→    return true
   112→   112→  }).sort((a, b) => {
   113→   113→    if (sortBy === 'name') return sortDir === 'asc' ? a.product.name.localeCompare(b.product.name) : b.product.name.localeCompare(a.product.name)
   114→   114→    if (sortBy === 'stock') return sortDir === 'asc' ? a.quantity - b.quantity : b.quantity - a.quantity
   115→   115→    return sortDir === 'asc' ? a.product.category.localeCompare(b.product.category) : b.product.category.localeCompare(a.product.category)
   116→   116→  })
   117→   117→
   118→   118→  const lowStockCount = items.filter((i) => i.quantity <= i.product.reorderPoint).length
   119→   119→  const outOfStockCount = items.filter((i) => i.quantity === 0).length
   120→   120→  const totalValue = items.reduce((sum, i) => sum + i.quantity * (i.product.costPrice || i.product.sellingPrice), 0)
   121→   121→
   122→   122→  const handleAdjust = async () => {
   123→   123→    if (!selectedItem || !adjustAmount || !adjustReason) return
   124→   124→    try {
   125→   125→      const adj = adjustType === 'ADD' ? parseInt(adjustAmount) : adjustType === 'REMOVE' ? -parseInt(adjustAmount) : parseInt(adjustAmount)
   126→   126→      await fetch('/api/inventory', {
   127→   127→        method: 'PUT',
   128→   128→        headers: { 'Content-Type': 'application/json' },
   129→   129→        body: JSON.stringify({ productId: selectedItem.productId, adjustment: adj, reason: adjustReason }),
   130→   130→      })
   131→   131→      addToast({ title: 'Stock Updated', description: `${selectedItem.product.name} adjusted`, variant: 'success' })
   132→   132→      setAdjustDialog(false)
   133→   133→      setAdjustAmount('')
   134→   134→      setAdjustReason('')
   135→   135→      fetchInventory()
   136→   136→    } catch {
   137→   137→      addToast({ title: 'Error', description: 'Failed to adjust stock', variant: 'destructive' })
   138→   138→    }
   139→   139→  }
   140→   140→
   141→   141→  const handleDeleteCategory = async (catId: string, catName: string) => {
   142→   142→    try {
   143→   143→      const res = await fetch(`/api/categories/${catId}`, { method: 'DELETE' })
   144→   144→      if (!res.ok) {
   145→   145→        const err = await res.json()
   146→   146→        throw new Error(err.error || 'Failed to delete category')
   147→   147→      }
   148→   148→      addToast({ title: 'Deleted', description: `Category "${catName}" removed`, variant: 'success' })
   149→   149→      fetchCategories()
   150→   150→    } catch (err: any) {
   151→   151→      addToast({ title: 'Error', description: err.message, variant: 'destructive' })
   152→   152→    }
   153→   153→  }
   154→   154→
   155→   155→  const handleAddProduct = async () => {
   156→   156→    if (!productForm.name || !productForm.price) return
   157→   157→    setSavingProduct(true)
   158→   158→    try {
   159→   159→      const res = await fetch('/api/products', {
   160→   160→        method: 'POST',
   161→   161→        headers: {
   162→   162→          'Content-Type': 'application/json',
   163→   163→          'x-user-role': currentUser?.role || 'SUPER_ADMIN',
   164→   164→        },
   165→   165→        body: JSON.stringify({
   166→   166→          name: productForm.name,
   167→   167→          ndc: productForm.sku || null,
   168→   168→          category: productForm.category,
   169→   169→          sellingPrice: parseFloat(productForm.price),
   170→   170→          costPrice: parseFloat(productForm.price) * 0.7,
   171→   171→          reorderPoint: parseInt(productForm.minStockLevel) || 10,
   172→   172→          expiryDate: productForm.expiryDate || null,
   173→   173→          batchNumber: productForm.barcode || null,
   174→   174→        }),
   175→   175→      })
   176→   176→      if (!res.ok) {
   177→   177→        const err = await res.json()
   178→   178→        throw new Error(err.error || 'Failed to create product')
   179→   179→      }
   180→   180→      const newProduct = await res.json()
   181→   181→
   182→   182→      // Set initial stock quantity if specified
   183→   183→      const qty = parseInt(productForm.stockQuantity)
   184→   184→      if (qty > 0) {
   185→   185→        await fetch('/api/inventory', {
   186→   186→          method: 'PUT',
   187→   187→          headers: { 'Content-Type': 'application/json' },
   188→   188→          body: JSON.stringify({
   189→   189→            productId: newProduct.id,
   190→   190→            adjustment: qty,
   191→   191→            reason: 'Initial stock on product creation',
   192→   192→          }),
   193→   193→        })
   194→   194→      }
   195→   195→
   196→   196→      addToast({ title: 'Product Added', description: `${productForm.name} has been added to inventory`, variant: 'success' })
   197→   197→      setAddProductDialog(false)
   198→   198→      setProductForm({ name: '', sku: '', category: 'OTC', price: '', stockQuantity: '', minStockLevel: '10', expiryDate: '', barcode: '' })
   199→   199→      fetchInventory()
   200→   200→    } catch (err: any) {
   201→   201→      addToast({ title: 'Error', description: err.message || 'Failed to add product', variant: 'destructive' })
   202→   202→    } finally {
   203→   203→      setSavingProduct(false)
   204→   204→    }
   205→   205→  }
   206→   206→
   207→   207→  const handleReceiveShipment = async () => {
   208→   208→    const validItems = shipmentItems.filter((i) => i.productId && i.quantity)
   209→   209→    if (validItems.length === 0) return
   210→   210→    try {
   211→   211→      await fetch('/api/inventory', {
   212→   212→        method: 'POST',
   213→   213→        headers: { 'Content-Type': 'application/json' },
   214→   214→        body: JSON.stringify({
   215→   215→          items: validItems.map((i) => ({
   216→   216→            productId: i.productId,
   217→   217→            quantity: parseInt(i.quantity),
   218→   218→            costPrice: parseFloat(i.costPrice) || 0,
   219→   219→          })),
   220→   220→        }),
   221→   221→      })
   222→   222→      addToast({ title: 'Shipment Received', description: `${validItems.length} items received`, variant: 'success' })
   223→   223→      setReceiveDialog(false)
   224→   224→      setShipmentItems([{ productId: '', quantity: '', costPrice: '' }])
   225→   225→      fetchInventory()
   226→   226→    } catch {
   227→   227→      addToast({ title: 'Error', description: 'Failed to receive shipment', variant: 'destructive' })
   228→   228→    }
   229→   229→  }
   230→   230→
   231→   231→  return (
   232→   232→    <div className="space-y-4">
   233→   233→      {/* Stats Cards */}
   234→   234→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   235→   235→        <Card>
   236→   236→          <CardContent className="p-4 flex items-center gap-3">
   237→   237→            <div className="h-10 w-10 rounded-lg bg-emerald-100 flex items-center justify-center">
   238→   238→              <Package className="h-5 w-5 text-emerald-600" />
   239→   239→            </div>
   240→   240→            <div>
   241→   241→              <p className="text-2xl font-bold">{items.length}</p>
   242→   242→              <p className="text-xs text-muted-foreground">Total Products</p>
   243→   243→            </div>
   244→   244→          </CardContent>
   245→   245→        </Card>
   246→   246→        <Card>
   247→   247→          <CardContent className="p-4 flex items-center gap-3">
   248→   248→            <div className="h-10 w-10 rounded-lg bg-red-100 flex items-center justify-center">
   249→   249→              <AlertTriangle className="h-5 w-5 text-red-600" />
   250→   250→            </div>
   251→   251→            <div>
   252→   252→              <p className="text-2xl font-bold text-red-600">{lowStockCount}</p>
   253→   253→              <p className="text-xs text-muted-foreground">Low Stock Alerts</p>
   254→   254→            </div>
   255→   255→          </CardContent>
   256→   256→        </Card>
   257→   257→        <Card>
   258→   258→          <CardContent className="p-4 flex items-center gap-3">
   259→   259→            <div className="h-10 w-10 rounded-lg bg-amber-100 flex items-center justify-center">

   261→   261→            </div>
   262→   262→            <div>

   264→   264→              <p className="text-xs text-muted-foreground">Out of Stock</p>
   265→   265→            </div>
   266→   266→          </CardContent>
   267→   267→        </Card>
   268→   268→        <Card>
   269→   269→          <CardContent className="p-4 flex items-center gap-3">
   270→   270→            <div className="h-10 w-10 rounded-lg bg-teal-100 flex items-center justify-center">
   271→   271→              <TrendingUp className="h-5 w-5 text-teal-600" />
   272→   272→            </div>
   273→   273→            <div>
   274→   274→              <p className="text-2xl font-bold">${totalValue.toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 2 })}</p>
   275→   275→              <p className="text-xs text-muted-foreground">Inventory Value</p>
   276→   276→            </div>
   277→   277→          </CardContent>
   278→   278→        </Card>
   279→   279→      </div>
   280→   280→
   281→   281→      {/* Filters & Actions */}
   282→   282→      <Card>
   283→   283→        <CardContent className="p-4">
   284→   284→          <div className="flex flex-col sm:flex-row gap-3">
   285→   285→            <div className="relative flex-1">
   286→   286→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   287→   287→              <Input
   288→   288→                placeholder="Search by product name or NDC..."
   289→   289→                value={searchQuery}
   290→   290→                onChange={(e) => setSearchQuery(e.target.value)}

   292→   292→              />
   293→   293→            </div>
   294→   294→            <Select value={categoryFilter} onValueChange={setCategoryFilter}>
   295→   295→              <SelectTrigger className="w-full sm:w-40">

   298→   298→              </SelectTrigger>

   302→   302→                <SelectItem value="PRESCRIPTION">Prescription</SelectItem>
   303→   303→                <SelectItem value="SUPPLEMENT">Supplement</SelectItem>

   311→   311→                <SelectValue />

   320→   320→            <Button onClick={() => setReceiveDialog(true)} className="bg-emerald-600 hover:bg-emerald-700">

   322→   322→              Receive Shipment

   325→   325→              <PackagePlus className="h-4 w-4 mr-2" />
   326→   326→              Add Product

   328→   328→            <Button onClick={() => setCategoryDialog(true)} variant="outline" className="border-emerald-300 text-emerald-700 hover:bg-emerald-50">
   329→   329→              <Tags className="h-4 w-4 mr-2" />
   330→   330→              Category

   332→   332→          </div>
   333→   333→        </CardContent>

   335→   335→
   336→   336→      {/* Inventory Table */}
   337→   337→      <Card>
   338→   338→        <CardContent className="p-0">
   339→   339→          <Table>
   340→   340→            <TableHeader>
   341→   341→              <TableRow>
   342→   342→                <TableHead className="cursor-pointer" onClick={() => { setSortBy('name'); setSortDir(sortDir === 'asc' ? 'desc' : 'asc') }}>
   343→   343→                  <span className="flex items-center gap-1">Product <ArrowUpDown className="h-3 w-3" /></span>
   344→   344→                </TableHead>
   345→   345→                <TableHead className="hidden md:table-cell">NDC</TableHead>

   347→   347→                  <span className="flex items-center gap-1">Category <ArrowUpDown className="h-3 w-3" /></span>
   348→   348→                </TableHead>
   349→   349→                <TableHead className="cursor-pointer text-right" onClick={() => { setSortBy('stock'); setSortDir(sortDir === 'asc' ? 'desc' : 'asc') }}>
   350→   350→                  <span className="flex items-center justify-end gap-1">In Stock <ArrowUpDown className="h-3 w-3" /></span>
   351→   351→                </TableHead>
   352→   352→                <TableHead className="hidden lg:table-cell text-right">Reorder Point</TableHead>
   353→   353→                <TableHead className="text-right">Status</TableHead>
   354→   354→                <TableHead className="hidden lg:table-cell text-right">Cost</TableHead>
   355→   355→                <TableHead className="hidden lg:table-cell text-right">Retail</TableHead>
   356→   356→                <TableHead className="text-right">Actions</TableHead>
   357→   357→              </TableRow>
   358→   358→            </TableHeader>
   359→   359→            <TableBody>
   360→   360→              {loading ? (
   361→   361→                Array.from({ length: 8 }).map((_, i) => (
   362→   362→                  <TableRow key={i}>
   363→   363→                    {Array.from({ length: 9 }).map((_, j) => (

   366→   366→                  </TableRow>

   368→   368→              ) : filteredItems.length === 0 ? (
   369→   369→                <TableRow>
   370→   370→                  <TableCell colSpan={9} className="text-center py-8 text-muted-foreground">
   371→   371→                    No inventory items found
   372→   372→                  </TableCell>
   373→   373→                </TableRow>
   374→   374→              ) : (
   375→   375→                filteredItems.map((item) => {
   376→   376→                  const isLow = item.quantity <= item.product.reorderPoint
   377→   377→                  const isOut = item.quantity === 0
   378→   378→                  return (
   379→   379→                    <TableRow key={item.id} className={isOut ? 'bg-red-50/50' : isLow ? 'bg-amber-50/50' : ''}>

   383→   383→                          <p className="text-xs text-muted-foreground">
   384→   384→                            {item.product.strength} {item.product.dosageForm} · {item.product.unitOfMeasure}
   385→   385→                          </p>
   386→   386→                        </div>
   387→   387→                      </TableCell>
   388→   388→                      <TableCell className="hidden md:table-cell text-xs font-mono">{item.product.ndc || '—'}</TableCell>
   389→   389→                      <TableCell className="hidden sm:table-cell">

   392→   392→                      <TableCell className="text-right font-bold">{item.quantity}</TableCell>

   402→   402→                      </TableCell>
   403→   403→                      <TableCell className="hidden lg:table-cell text-right">${item.product.costPrice?.toFixed(2) || '—'}</TableCell>
   404→   404→                      <TableCell className="hidden lg:table-cell text-right">${item.product.sellingPrice.toFixed(2)}</TableCell>
   405→   405→                      <TableCell className="text-right">
   406→   406→                        <Button size="sm" variant="ghost" onClick={() => { setSelectedItem(item); setAdjustDialog(true) }}>
   407→   407→                          <Edit className="h-3.5 w-3.5 mr-1" />
   408→   408→                          Adjust
   409→   409→                        </Button>
