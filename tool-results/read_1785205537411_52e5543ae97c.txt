     1→'use client'
     2→
     3→import { useState, useEffect, useCallback } from 'react'
     4→import {
     5→  Package, Search, AlertTriangle, Edit, ArrowUpDown,
     6→  Download, Filter, TrendingUp, PackagePlus, ClipboardCheck, X
     7→} from 'lucide-react'
     8→import { Button } from '@/components/ui/button'
     9→import { Input } from '@/components/ui/input'
    10→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
    11→import { Badge } from '@/components/ui/badge'
    12→import { Skeleton } from '@/components/ui/skeleton'
    13→import {
    14→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
    15→} from '@/components/ui/table'
    16→import {
    17→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription, DialogFooter,
    18→} from '@/components/ui/dialog'
    19→import { Label } from '@/components/ui/label'
    20→import { Textarea } from '@/components/ui/textarea'
    21→import {
    22→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    23→} from '@/components/ui/select'
    24→import { useAppStore } from '@/store/app-store'
    25→
    26→interface InventoryItem {
    27→  id: string
    28→  productId: string
    29→  quantity: number
    30→  lastCounted: string | null
    31→  product: {
    32→    id: string
    33→    name: string
    34→    ndc: string | null
    35→    category: string
    36→    sellingPrice: number
    37→    costPrice: number | null
    38→    status: string
    39→    reorderPoint: number
    40→    maxStock: number | null
    41→    dosageForm: string | null
    42→    strength: string | null
    43→    unitOfMeasure: string
    44→    expiryDate: string | null
    45→    batchNumber: string | null
    46→  }
    47→}
    48→
    49→export function InventoryView() {
    50→  const [items, setItems] = useState<InventoryItem[]>([])
    51→  const [loading, setLoading] = useState(true)
    52→  const [searchQuery, setSearchQuery] = useState('')
    53→  const [categoryFilter, setCategoryFilter] = useState('ALL')
    54→  const [stockFilter, setStockFilter] = useState('ALL')
    55→  const [adjustDialog, setAdjustDialog] = useState(false)
    56→  const [addProductDialog, setAddProductDialog] = useState(false)
    57→  const [categories, setCategories] = useState<{ id: string; name: string; description: string | null; _count?: { products: number } }[]>([])
    58→  const [selectedItem, setSelectedItem] = useState<InventoryItem | null>(null)
    59→  const [adjustType, setAdjustType] = useState('ADD')
    60→  const [adjustAmount, setAdjustAmount] = useState('')
    61→  const [adjustReason, setAdjustReason] = useState('')
    62→  const [adjustCostPrice, setAdjustCostPrice] = useState('')
    63→  const [adjustSellingPrice, setAdjustSellingPrice] = useState('')
    64→  const [stockCountDialog, setStockCountDialog] = useState(false)
    65→  const [stockSearch, setStockSearch] = useState('')
    66→  const [stockEntries, setStockEntries] = useState<{ productId: string; name: string; currentQty: number; physicalQty: string; unit: string }[]>([])
    67→  const [stockSaving, setStockSaving] = useState(false)
    68→  const [sortBy, setSortBy] = useState<'name' | 'stock' | 'category'>('name')
    69→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('asc')
    70→  const [productForm, setProductForm] = useState({
    71→    name: '', sku: '', category: 'OTC', price: '', stockQuantity: '',
    72→    minStockLevel: '10', expiryDate: '', barcode: '',
    73→  })
    74→  const [savingProduct, setSavingProduct] = useState(false)
    75→  const addToast = useAppStore((s) => s.addToast)
    76→  const currentUser = useAppStore((s) => s.user)
    77→
    78→  const fetchInventory = useCallback(async () => {
    79→    setLoading(true)
    80→    try {
    81→      const params = new URLSearchParams()
    82→      if (searchQuery) params.set('search', searchQuery)
    83→      if (categoryFilter !== 'ALL') params.set('category', categoryFilter)
    84→      const res = await fetch(`/api/inventory?${params}`)
    85→      if (res.ok) {
    86→        const data = await res.json()
    87→        setItems(data)
    88→      }
    89→    } catch {
    90→      addToast({ title: 'Error', description: 'Failed to load inventory', variant: 'destructive' })
    91→    } finally {
    92→      setLoading(false)
    93→    }
    94→  }, [searchQuery, categoryFilter, addToast])
    95→
    96→  const fetchCategories = useCallback(async () => {
    97→    try {
    98→      const res = await fetch('/api/categories')
    99→      if (res.ok) setCategories(await res.json())
   100→    } catch {
   101→      // silent — categories are optional enhancement
   102→    }
   103→  }, [])
   104→
   105→  useEffect(() => { fetchInventory(); fetchCategories() }, [fetchInventory, fetchCategories])
   106→
   107→  const filteredItems = items.filter((item) => {
   108→    const q = Number(item.quantity) || 0
   109→    const r = Number(item.product.reorderPoint) || 10
   110→    if (stockFilter === 'LOW') return q > 0 && q <= r
   111→    if (stockFilter === 'OUT') return q === 0
   112→    if (stockFilter === 'OK') return q > r
   113→    return true
   114→  }).sort((a, b) => {
   115→    if (sortBy === 'name') return sortDir === 'asc' ? a.product.name.localeCompare(b.product.name) : b.product.name.localeCompare(a.product.name)
   116→    if (sortBy === 'stock') return sortDir === 'asc' ? (Number(a.quantity)||0) - (Number(b.quantity)||0) : (Number(b.quantity)||0) - (Number(a.quantity)||0)
   117→    return sortDir === 'asc' ? a.product.category.localeCompare(b.product.category) : b.product.category.localeCompare(a.product.category)
   118→  })
   119→
   120→  const lowStockCount = items.filter((i) => { const q = Number(i.quantity) || 0; const r = Number(i.product.reorderPoint) || 10; return q > 0 && q <= r }).length
   121→  const outOfStockCount = items.filter((i) => (Number(i.quantity) || 0) === 0).length
   122→  const totalValue = items.reduce((sum, i) => sum + (Number(i.quantity) || 0) * (i.product.costPrice || i.product.sellingPrice), 0)
   123→
   124→  const handleAdjust = async () => {
   125→    if (!selectedItem || (!adjustAmount && !adjustCostPrice && !adjustSellingPrice) || !adjustReason) return
   126→    try {
   127→      const isSet = adjustType === 'SET'
   128→      const adj = adjustAmount ? (adjustType === 'ADD' ? parseInt(adjustAmount) : adjustType === 'REMOVE' ? -parseInt(adjustAmount) : parseInt(adjustAmount)) : 0
   129→      const body: Record<string, any> = {
   130→        productId: selectedItem.productId,
   131→        adjustmentType: adjustType,
   132→        reason: adjustReason,
   133→      }
   134→      if (isSet) {
   135→        body.setQuantity = adj
   136→        body.adjustment = 0
   137→      } else {
   138→        body.adjustment = adj
   139→      }
   140→      if (adjustCostPrice !== '') body.costPrice = parseFloat(adjustCostPrice)
   141→      if (adjustSellingPrice !== '') body.sellingPrice = parseFloat(adjustSellingPrice)
   142→
   143→      const res = await fetch('/api/inventory', {
   144→        method: 'PUT',
   145→        headers: { 'Content-Type': 'application/json', 'x-user-role': currentUser?.role || 'SUPER_ADMIN' },
   146→        body: JSON.stringify(body),
   147→      })
   148→      if (!res.ok) {
   149→        const err = await res.json()
   150→        throw new Error(err.error || 'Failed to adjust')
   151→      }
   152→      addToast({ title: 'Product Updated', description: `${selectedItem.product.name} adjusted successfully`, variant: 'success' })
   153→      setAdjustDialog(false)
   154→      setAdjustAmount('')
   155→      setAdjustReason('')
   156→      setAdjustCostPrice('')
   157→      setAdjustSellingPrice('')
   158→      fetchInventory()
   159→    } catch (err: any) {
   160→      addToast({ title: 'Error', description: err.message || 'Failed to adjust stock', variant: 'destructive' })
   161→    }
   162→  }
   163→
   164→  const handleAddProduct = async () => {
   165→    if (!productForm.name || !productForm.price) return
   166→    setSavingProduct(true)
   167→    try {
   168→      const res = await fetch('/api/products', {
   169→        method: 'POST',
   170→        headers: {
   171→          'Content-Type': 'application/json',
   172→          'x-user-role': currentUser?.role || 'SUPER_ADMIN',
   173→        },
   174→        body: JSON.stringify({
   175→          name: productForm.name,
   176→          ndc: productForm.sku || null,
   177→          category: productForm.category,
   178→          sellingPrice: parseFloat(productForm.price),
   179→          costPrice: parseFloat(productForm.price) * 0.7,
   180→          reorderPoint: parseInt(productForm.minStockLevel) || 10,
   181→          expiryDate: productForm.expiryDate || null,
   182→          batchNumber: productForm.barcode || null,
   183→        }),
   184→      })
   185→      if (!res.ok) {
   186→        const err = await res.json()
   187→        throw new Error(err.error || 'Failed to create product')
   188→      }
   189→      const newProduct = await res.json()
   190→
   191→      // Set initial stock quantity if specified
   192→      const qty = parseInt(productForm.stockQuantity)
   193→      if (qty > 0) {
   194→        await fetch('/api/inventory', {
   195→          method: 'PUT',
   196→          headers: { 'Content-Type': 'application/json' },
   197→          body: JSON.stringify({
   198→            productId: newProduct.id,
   199→            adjustment: qty,
   200→            reason: 'Initial stock on product creation',
   201→          }),
   202→        })
   203→      }
   204→
   205→      addToast({ title: 'Product Added', description: `${productForm.name} has been added to inventory`, variant: 'success' })
   206→      setAddProductDialog(false)
   207→      setProductForm({ name: '', sku: '', category: 'OTC', price: '', stockQuantity: '', minStockLevel: '10', expiryDate: '', barcode: '' })
   208→      fetchInventory()
   209→    } catch (err: any) {
   210→      addToast({ title: 'Error', description: err.message || 'Failed to add product', variant: 'destructive' })
   211→    } finally {
   212→      setSavingProduct(false)
   213→    }
   214→  }
   215→
   216→  // ── Stock Count: search + set physical quantities ──────────────
   217→  const filteredStockProducts = stockSearch.length > 0
   218→    ? items.filter(i => i.product.name.toLowerCase().includes(stockSearch.toLowerCase()) || i.product.ndc?.toLowerCase().includes(stockSearch.toLowerCase()))
   219→    : []
   220→
   221→  const addStockEntry = (item: InventoryItem) => {
   222→    if (stockEntries.find(e => e.productId === item.productId)) return
   223→    setStockEntries(prev => [...prev, {
   224→      productId: item.productId,
   225→      name: item.product.name,
   226→      currentQty: Number(item.quantity) || 0,
   227→      physicalQty: String(Number(item.quantity) || 0),
   228→      unit: item.product.unitOfMeasure,
   229→    }])
   230→    setStockSearch('')
   231→  }
   232→
   233→  const removeStockEntry = (productId: string) => {
   234→    setStockEntries(prev => prev.filter(e => e.productId !== productId))
   235→  }
   236→
   237→  const updateStockEntry = (productId: string, physicalQty: string) => {
   238→    setStockEntries(prev => prev.map(e => e.productId === productId ? { ...e, physicalQty } : e))
   239→  }
   240→
   241→  const handleStockCountSave = async () => {
   242→    if (stockEntries.length === 0) return
   243→    setStockSaving(true)
   244→    try {
   245→      let updated = 0
   246→      for (const entry of stockEntries) {
   247→        const physicalQty = parseInt(entry.physicalQty)
   248→        if (isNaN(physicalQty)) continue
   249→        if (physicalQty === entry.currentQty) continue // no change needed
   250→        await fetch('/api/inventory', {
   251→          method: 'PUT',
   252→          headers: { 'Content-Type': 'application/json', 'x-user-role': currentUser?.role || 'SUPER_ADMIN' },
   253→          body: JSON.stringify({
   254→            productId: entry.productId,
   255→            adjustmentType: 'SET',
   256→            setQuantity: physicalQty,
   257→            adjustment: 0,
   258→            reason: 'Physical stock count',
   259→          }),
   260→        })
   261→        updated++
   262→      }
   263→      addToast({ title: 'Stock Count Saved', description: `${updated} product${updated !== 1 ? 's' : ''} updated to physical count`, variant: 'success' })
   264→      setStockCountDialog(false)
   265→      setStockEntries([])
   266→      fetchInventory()
   267→    } catch (err: any) {
   268→      addToast({ title: 'Error', description: err.message || 'Failed to save stock count', variant: 'destructive' })
   269→    } finally {
   270→      setStockSaving(false)
   271→    }
   272→  }
   273→
   274→  return (
   275→    <div className="space-y-4">
   276→      {/* Stats Cards */}
   277→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   278→        <Card>
   279→          <CardContent className="p-4 flex items-center gap-3">
   280→            <div className="h-10 w-10 rounded-lg bg-emerald-100 flex items-center justify-center">
   281→              <Package className="h-5 w-5 text-emerald-600" />
   282→            </div>
   283→            <div>
   284→              <p className="text-2xl font-bold">{items.length}</p>
   285→              <p className="text-xs text-muted-foreground">Total Products</p>
   286→            </div>
   287→          </CardContent>
   288→        </Card>
   289→        <Card>
   290→          <CardContent className="p-4 flex items-center gap-3">
   291→            <div className="h-10 w-10 rounded-lg bg-red-100 flex items-center justify-center">
   292→              <AlertTriangle className="h-5 w-5 text-red-600" />
   293→            </div>
   294→            <div>
   295→              <p className="text-2xl font-bold text-red-600">{lowStockCount}</p>
   296→              <p className="text-xs text-muted-foreground">Low Stock Alerts</p>
   297→            </div>
   298→          </CardContent>
   299→        </Card>
   300→        <Card>
   301→          <CardContent className="p-4 flex items-center gap-3">
   302→            <div className="h-10 w-10 rounded-lg bg-amber-100 flex items-center justify-center">
   303→              <AlertTriangle className="h-5 w-5 text-amber-600" />
   304→            </div>
   305→            <div>
   306→              <p className="text-2xl font-bold text-amber-600">{outOfStockCount}</p>
   307→              <p className="text-xs text-muted-foreground">Out of Stock</p>
   308→            </div>
   309→          </CardContent>
   310→        </Card>
   311→        <Card>
   312→          <CardContent className="p-4 flex items-center gap-3">
   313→            <div className="h-10 w-10 rounded-lg bg-teal-100 flex items-center justify-center">
   314→              <TrendingUp className="h-5 w-5 text-teal-600" />
   315→            </div>
   316→            <div>
   317→              <p className="text-2xl font-bold">${totalValue.toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 2 })}</p>
   318→              <p className="text-xs text-muted-foreground">Inventory Value</p>
   319→            </div>
   320→          </CardContent>
   321→        </Card>
   322→      </div>
   323→
   324→      {/* Filters & Actions */}
   325→      <Card>
   326→        <CardContent className="p-4">
   327→          <div className="flex flex-col sm:flex-row gap-3">
   328→            <div className="relative flex-1">
   329→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   330→              <Input
   331→                placeholder="Search by product name or NDC..."
   332→                value={searchQuery}
   333→                onChange={(e) => setSearchQuery(e.target.value)}
   334→                className="pl-9"
   335→              />
   336→            </div>
   337→            <Select value={categoryFilter} onValueChange={setCategoryFilter}>
   338→              <SelectTrigger className="w-full sm:w-40">
   339→                <Filter className="h-4 w-4 mr-2" />
   340→                <SelectValue />
   341→              </SelectTrigger>
   342→              <SelectContent>
   343→                <SelectItem value="ALL">All Categories</SelectItem>
   344→                <SelectItem value="OTC">OTC</SelectItem>
   345→                <SelectItem value="PRESCRIPTION">Prescription</SelectItem>
   346→                <SelectItem value="SUPPLEMENT">Supplement</SelectItem>
   347→                <SelectItem value="MEDICAL_DEVICE">Medical Device</SelectItem>
   348→                <SelectItem value="PERSONAL_CARE">Personal Care</SelectItem>
   349→                <SelectItem value="CONSUMABLES">Consumables</SelectItem>
   350→              </SelectContent>
   351→            </Select>
   352→            <Select value={stockFilter} onValueChange={setStockFilter}>
   353→              <SelectTrigger className="w-full sm:w-36">
   354→                <SelectValue />
   355→              </SelectTrigger>
   356→              <SelectContent>
   357→                <SelectItem value="ALL">All Stock</SelectItem>
   358→                <SelectItem value="LOW">Low Stock</SelectItem>
   359→                <SelectItem value="OUT">Out of Stock</SelectItem>
   360→                <SelectItem value="OK">In Stock OK</SelectItem>
   361→              </SelectContent>
   362→            </Select>
   363→            <Button onClick={() => setAddProductDialog(true)} className="bg-teal-600 hover:bg-teal-700">
   364→              <PackagePlus className="h-4 w-4 mr-2" />
   365→              Add Product
   366→            </Button>
   367→            <Button onClick={() => setStockCountDialog(true)} className="bg-indigo-600 hover:bg-indigo-700">
   368→              <ClipboardCheck className="h-4 w-4 mr-2" />
   369→              Stock Count
   370→            </Button>
   371→          </div>
   372→        </CardContent>
   373→      </Card>
   374→
   375→      {/* Inventory Table */}
   376→      <Card>
   377→        <CardContent className="p-0">
   378→          <Table>
   379→            <TableHeader>
   380→              <TableRow>
   381→                <TableHead className="cursor-pointer" onClick={() => { setSortBy('name'); setSortDir(sortDir === 'asc' ? 'desc' : 'asc') }}>
   382→                  <span className="flex items-center gap-1">Product <ArrowUpDown className="h-3 w-3" /></span>
   383→                </TableHead>
   384→                <TableHead className="hidden md:table-cell">NDC</TableHead>
   385→                <TableHead className="cursor-pointer hidden sm:table-cell" onClick={() => { setSortBy('category'); setSortDir(sortDir === 'asc' ? 'desc' : 'asc') }}>
   386→                  <span className="flex items-center gap-1">Category <ArrowUpDown className="h-3 w-3" /></span>
   387→                </TableHead>
   388→                <TableHead className="cursor-pointer text-right" onClick={() => { setSortBy('stock'); setSortDir(sortDir === 'asc' ? 'desc' : 'asc') }}>
   389→                  <span className="flex items-center justify-end gap-1">In Stock <ArrowUpDown className="h-3 w-3" /></span>
   390→                </TableHead>
   391→                <TableHead className="hidden lg:table-cell text-right">Reorder Point</TableHead>
   392→                <TableHead className="text-right">Status</TableHead>
   393→                <TableHead className="hidden lg:table-cell text-right">Cost</TableHead>
   394→                <TableHead className="hidden lg:table-cell text-right">Retail</TableHead>
   395→                <TableHead className="text-right">Actions</TableHead>
   396→              </TableRow>
   397→            </TableHeader>
   398→            <TableBody>
   399→              {loading ? (
   400→                Array.from({ length: 8 }).map((_, i) => (
   401→                  <TableRow key={i}>
   402→                    {Array.from({ length: 9 }).map((_, j) => (
   403→                      <TableCell key={j}><Skeleton className="h-5 w-full" /></TableCell>
   404→                    ))}
   405→                  </TableRow>
   406→                ))
   407→              ) : filteredItems.length === 0 ? (
   408→                <TableRow>
   409→                  <TableCell colSpan={9} className="text-center py-8 text-muted-foreground">
   410→                    No inventory items found
   411→                  </TableCell>
   412→                </TableRow>
   413→              ) : (
   414→                filteredItems.map((item) => {
   415→                  const qty = Number(item.quantity) || 0
   416→                  const reorder = Number(item.product.reorderPoint) || 10
   417→                  const isOut = qty === 0
   418→                  const isLow = qty > 0 && qty <= reorder
   419→                  return (
   420→                    <TableRow key={item.id} className={isOut ? 'bg-red-50/50' : isLow ? 'bg-amber-50/50' : ''}>
   421→                      <TableCell>
   422→                        <div>
   423→                          <p className="font-medium text-sm">{item.product.name}</p>
   424→                          <p className="text-xs text-muted-foreground">
   425→                            {item.product.strength} {item.product.dosageForm} · {item.product.unitOfMeasure}
   426→                          </p>
   427→                        </div>
   428→                      </TableCell>
   429→                      <TableCell className="hidden md:table-cell text-xs font-mono">{item.product.ndc || '—'}</TableCell>
   430→                      <TableCell className="hidden sm:table-cell">
   431→                        <Badge variant="outline" className="text-xs">{item.product.category.replace(/_/g, ' ')}</Badge>
   432→                      </TableCell>
   433→                      <TableCell className="text-right font-bold">{qty}</TableCell>
   434→                      <TableCell className="hidden lg:table-cell text-right text-muted-foreground">{reorder}</TableCell>
   435→                      <TableCell className="text-right">
   436→                        {isOut ? (
   437→                          <Badge className="bg-red-100 text-red-700 border-red-200">Out of Stock</Badge>
   438→                        ) : isLow ? (
   439→                          <Badge className="bg-amber-100 text-amber-700 border-amber-200">Low Stock</Badge>
   440→                        ) : (
   441→                          <Badge className="bg-emerald-100 text-emerald-700 border-emerald-200">In Stock</Badge>
   442→                        )}
   443→                      </TableCell>
   444→                      <TableCell className="hidden lg:table-cell text-right">${item.product.costPrice?.toFixed(2) || '—'}</TableCell>
   445→                      <TableCell className="hidden lg:table-cell text-right">${item.product.sellingPrice.toFixed(2)}</TableCell>
   446→                      <TableCell className="text-right">
   447→                        <Button size="sm" variant="ghost" onClick={() => { setSelectedItem(item); setAdjustDialog(true) }}>
   448→                          <Edit className="h-3.5 w-3.5 mr-1" />
   449→                          Adjust
   450→                        </Button>
   451→                      </TableCell>
   452→                    </TableRow>
   453→                  )
   454→                })
   455→              )}
   456→            </TableBody>
   457→          </Table>
   458→        </CardContent>
   459→      </Card>
   460→
   461→      {/* Add Product Dialog */}
   462→      <Dialog open={addProductDialog} onOpenChange={setAddProductDialog}>
   463→        <DialogContent className="max-w-lg">
   464→          <DialogHeader>
   465→            <DialogTitle className="flex items-center gap-2">
   466→              <PackagePlus className="h-5 w-5 text-teal-600" />
   467→              Add New Product
   468→            </DialogTitle>
   469→          </DialogHeader>
   470→          <div className="grid grid-cols-2 gap-4 max-h-[70vh] overflow-y-auto pr-1">
   471→            {/* Product Name */}
   472→            <div className="col-span-2">
   473→              <Label htmlFor="prod-name">Product Name <span className="text-red-500">*</span></Label>
   474→              <Input
   475→                id="prod-name"
   476→                placeholder="e.g., Amoxicillin 500mg Capsules"
   477→                value={productForm.name}
   478→                onChange={(e) => setProductForm({ ...productForm, name: e.target.value })}
   479→                className="mt-1"
   480→              />
   481→            </div>
   482→
   483→            {/* SKU */}
   484→            <div>
   485→              <Label htmlFor="prod-sku">SKU</Label>
   486→              <Input
   487→                id="prod-sku"
   488→                placeholder="e.g., SKU-00123"
   489→                value={productForm.sku}
   490→                onChange={(e) => setProductForm({ ...productForm, sku: e.target.value })}
   491→                className="mt-1"

   520→                id="prod-price"
   521→                type="number"

   524→                placeholder="0.00"

   531→            {/* Stock Quantity */}

   535→                id="prod-stock"
   536→                type="number"
   537→                min="0"

   539→                value={productForm.stockQuantity}

   545→            {/* Min Stock Level */}

   547→              <Label htmlFor="prod-minstock">Min Stock Level</Label>

   549→                id="prod-minstock"
   550→                type="number"

   552→                placeholder="10"

   584→          <DialogFooter className="mt-4">

   587→              onClick={handleAddProduct}
   588→              className="bg-teal-600 hover:bg-teal-700"

   591→              {savingProduct ? 'Adding...' : 'Add Product'}

   605→          {selectedItem && (
   606→            <div className="space-y-4">
   607→              <div className="bg-muted rounded-lg p-3">
   608→                <p className="font-medium">{selectedItem.product.name}</p>

   610→                  Current Stock: {Number(selectedItem.quantity) || 0} {selectedItem.product.unitOfMeasure}

   616→                <div>
   617→                  <Label>Adjustment Type</Label>

   669→
   670→          {/* Product Search */}

   674→              placeholder="Search product by name or NDC..."

   677→              className="pl-9"
   678→            />
   679→            {filteredStockProducts.length > 0 && (

   698→          </div>
   699→
   700→          {/* Selected Products List */}
   701→          <div className="border rounded-lg divide-y max-h-64 overflow-y-auto">
   702→            {stockEntries.length === 0 ? (

   705→              </div>
   706→            ) : (
   707→              stockEntries.map((entry) => {
   708→                const diff = parseInt(entry.physicalQty || '0') - entry.currentQty

   710→                  <div key={entry.productId} className="flex items-center gap-3 px-3 py-2">
   711→                    <div className="flex-1 min-w-0">
   712→                      <p className="font-medium text-sm truncate">{entry.name}</p>

   716→                      <Input
   717→                        type="number"
   718→                        min="0"
   719→                        value={entry.physicalQty}

   722→                        placeholder="Qty"
   723→                      />
   724→                      <span className="text-xs text-muted-foreground">{entry.unit}</span>
   725→                      {diff !== 0 && (

   730→                      <Button size="sm" variant="ghost" onClick={() => removeStockEntry(entry.productId)} className="h-7 w-7 p-0 text-muted-foreground hover:text-red-600">
   731→                        <X className="h-3.5 w-3.5" />

   734→                  </div>

   739→
   740→          <DialogFooter>

   745→              onClick={handleStockCountSave}
