     1→'use client'
     2→
     3→import { useState, useEffect, useCallback } from 'react'
     4→import {
     5→  Package, Search, AlertTriangle, Edit, ArrowUpDown,
     6→  Download, Filter, TrendingUp, PackagePlus
     7→} from 'lucide-react'
     8→import { Button } from '@/components/ui/button'
     9→import { Input } from '@/components/ui/input'
    10→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
    11→import { Badge } from '@/components/ui/badge'
    12→import { Skeleton } from '@/components/ui/skeleton'
    13→import {
    14→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
    15→} from '@/components/ui/table'
    16→import {
    17→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter,
    18→} from '@/components/ui/dialog'
    19→import { Label } from '@/components/ui/label'
    20→import { Textarea } from '@/components/ui/textarea'
    21→import {
    22→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    23→} from '@/components/ui/select'
    24→import { useAppStore } from '@/store/app-store'
    25→
    26→interface InventoryItem {
    27→  id: string
    28→  productId: string
    29→  quantity: number
    30→  lastCounted: string | null
    31→  product: {
    32→    id: string
    33→    name: string
    34→    ndc: string | null
    35→    category: string
    36→    sellingPrice: number
    37→    costPrice: number | null
    38→    status: string
    39→    reorderPoint: number
    40→    maxStock: number | null
    41→    dosageForm: string | null
    42→    strength: string | null
    43→    unitOfMeasure: string
    44→    expiryDate: string | null
    45→    batchNumber: string | null
    46→  }
    47→}
    48→
    49→export function InventoryView() {
    50→  const [items, setItems] = useState<InventoryItem[]>([])
    51→  const [loading, setLoading] = useState(true)
    52→  const [searchQuery, setSearchQuery] = useState('')
    53→  const [categoryFilter, setCategoryFilter] = useState('ALL')
    54→  const [stockFilter, setStockFilter] = useState('ALL')
    55→  const [adjustDialog, setAdjustDialog] = useState(false)
    56→  const [addProductDialog, setAddProductDialog] = useState(false)
    57→  const [categories, setCategories] = useState<{ id: string; name: string; description: string | null; _count?: { products: number } }[]>([])
    58→  const [selectedItem, setSelectedItem] = useState<InventoryItem | null>(null)
    59→  const [adjustType, setAdjustType] = useState('ADD')
    60→  const [adjustAmount, setAdjustAmount] = useState('')
    61→  const [adjustReason, setAdjustReason] = useState('')
    62→  const [adjustCostPrice, setAdjustCostPrice] = useState('')
    63→  const [adjustSellingPrice, setAdjustSellingPrice] = useState('')
    64→  const [sortBy, setSortBy] = useState<'name' | 'stock' | 'category'>('name')
    65→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('asc')
    66→  const [productForm, setProductForm] = useState({
    67→    name: '', sku: '', category: 'OTC', price: '', stockQuantity: '',
    68→    minStockLevel: '10', expiryDate: '', barcode: '',
    69→  })
    70→  const [savingProduct, setSavingProduct] = useState(false)
    71→  const addToast = useAppStore((s) => s.addToast)
    72→  const currentUser = useAppStore((s) => s.user)
    73→
    74→  const fetchInventory = useCallback(async () => {
    75→    setLoading(true)
    76→    try {
    77→      const params = new URLSearchParams()
    78→      if (searchQuery) params.set('search', searchQuery)
    79→      if (categoryFilter !== 'ALL') params.set('category', categoryFilter)
    80→      const res = await fetch(`/api/inventory?${params}`)
    81→      if (res.ok) {
    82→        const data = await res.json()
    83→        setItems(data)
    84→      }
    85→    } catch {
    86→      addToast({ title: 'Error', description: 'Failed to load inventory', variant: 'destructive' })
    87→    } finally {
    88→      setLoading(false)
    89→    }
    90→  }, [searchQuery, categoryFilter, addToast])
    91→
    92→  const fetchCategories = useCallback(async () => {
    93→    try {
    94→      const res = await fetch('/api/categories')
    95→      if (res.ok) setCategories(await res.json())
    96→    } catch {
    97→      // silent — categories are optional enhancement
    98→    }
    99→  }, [])
   100→
   101→  useEffect(() => { fetchInventory(); fetchCategories() }, [fetchInventory, fetchCategories])
   102→
   103→  const filteredItems = items.filter((item) => {
   104→    const q = Number(item.quantity) || 0
   105→    const r = Number(item.product.reorderPoint) || 10
   106→    if (stockFilter === 'LOW') return q > 0 && q <= r
   107→    if (stockFilter === 'OUT') return q === 0
   108→    if (stockFilter === 'OK') return q > r
   109→    return true
   110→  }).sort((a, b) => {
   111→    if (sortBy === 'name') return sortDir === 'asc' ? a.product.name.localeCompare(b.product.name) : b.product.name.localeCompare(a.product.name)
   112→    if (sortBy === 'stock') return sortDir === 'asc' ? (Number(a.quantity)||0) - (Number(b.quantity)||0) : (Number(b.quantity)||0) - (Number(a.quantity)||0)
   113→    return sortDir === 'asc' ? a.product.category.localeCompare(b.product.category) : b.product.category.localeCompare(a.product.category)
   114→  })
   115→
   116→  const lowStockCount = items.filter((i) => { const q = Number(i.quantity) || 0; const r = Number(i.product.reorderPoint) || 10; return q > 0 && q <= r }).length
   117→  const outOfStockCount = items.filter((i) => (Number(i.quantity) || 0) === 0).length
   118→  const totalValue = items.reduce((sum, i) => sum + (Number(i.quantity) || 0) * (i.product.costPrice || i.product.sellingPrice), 0)
   119→
   120→  const handleAdjust = async () => {
   121→    if (!selectedItem || (!adjustAmount && !adjustCostPrice && !adjustSellingPrice) || !adjustReason) return
   122→    try {
   123→      const adj = adjustAmount ? (adjustType === 'ADD' ? parseInt(adjustAmount) : adjustType === 'REMOVE' ? -parseInt(adjustAmount) : parseInt(adjustAmount)) : 0
   124→      const body: Record<string, any> = {
   125→        productId: selectedItem.productId,
   126→        adjustment: adj,
   127→        reason: adjustReason,
   128→      }
   129→      if (adjustCostPrice !== '') body.costPrice = parseFloat(adjustCostPrice)
   130→      if (adjustSellingPrice !== '') body.sellingPrice = parseFloat(adjustSellingPrice)
   131→
   132→      const res = await fetch('/api/inventory', {
   133→        method: 'PUT',
   134→        headers: { 'Content-Type': 'application/json' },
   135→        body: JSON.stringify(body),
   136→      })
   137→      if (!res.ok) {
   138→        const err = await res.json()
   139→        throw new Error(err.error || 'Failed to adjust')
   140→      }
   141→      addToast({ title: 'Product Updated', description: `${selectedItem.product.name} adjusted successfully`, variant: 'success' })
   142→      setAdjustDialog(false)
   143→      setAdjustAmount('')
   144→      setAdjustReason('')
   145→      setAdjustCostPrice('')
   146→      setAdjustSellingPrice('')
   147→      fetchInventory()
   148→    } catch (err: any) {
   149→      addToast({ title: 'Error', description: err.message || 'Failed to adjust stock', variant: 'destructive' })
   150→    }
   151→  }
   152→
   153→  const handleAddProduct = async () => {
   154→    if (!productForm.name || !productForm.price) return
   155→    setSavingProduct(true)
   156→    try {
   157→      const res = await fetch('/api/products', {
   158→        method: 'POST',
   159→        headers: {
   160→          'Content-Type': 'application/json',
   161→          'x-user-role': currentUser?.role || 'SUPER_ADMIN',
   162→        },
   163→        body: JSON.stringify({
   164→          name: productForm.name,
   165→          ndc: productForm.sku || null,
   166→          category: productForm.category,
   167→          sellingPrice: parseFloat(productForm.price),
   168→          costPrice: parseFloat(productForm.price) * 0.7,
   169→          reorderPoint: parseInt(productForm.minStockLevel) || 10,
   170→          expiryDate: productForm.expiryDate || null,
   171→          batchNumber: productForm.barcode || null,
   172→        }),
   173→      })
   174→      if (!res.ok) {
   175→        const err = await res.json()
   176→        throw new Error(err.error || 'Failed to create product')
   177→      }
   178→      const newProduct = await res.json()
   179→
   180→      // Set initial stock quantity if specified
   181→      const qty = parseInt(productForm.stockQuantity)
   182→      if (qty > 0) {
   183→        await fetch('/api/inventory', {
   184→          method: 'PUT',
   185→          headers: { 'Content-Type': 'application/json' },
   186→          body: JSON.stringify({
   187→            productId: newProduct.id,
   188→            adjustment: qty,
   189→            reason: 'Initial stock on product creation',
   190→          }),
   191→        })
   192→      }
   193→
   194→      addToast({ title: 'Product Added', description: `${productForm.name} has been added to inventory`, variant: 'success' })
   195→      setAddProductDialog(false)
   196→      setProductForm({ name: '', sku: '', category: 'OTC', price: '', stockQuantity: '', minStockLevel: '10', expiryDate: '', barcode: '' })
   197→      fetchInventory()
   198→    } catch (err: any) {
   199→      addToast({ title: 'Error', description: err.message || 'Failed to add product', variant: 'destructive' })
   200→    } finally {
   201→      setSavingProduct(false)
   202→    }
   203→  }
   204→
   205→  return (
   206→    <div className="space-y-4">
   207→      {/* Stats Cards */}
   208→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   209→        <Card>
   210→          <CardContent className="p-4 flex items-center gap-3">
   211→            <div className="h-10 w-10 rounded-lg bg-emerald-100 flex items-center justify-center">
   212→              <Package className="h-5 w-5 text-emerald-600" />
   213→            </div>
   214→            <div>
   215→              <p className="text-2xl font-bold">{items.length}</p>
   216→              <p className="text-xs text-muted-foreground">Total Products</p>
   217→            </div>
   218→          </CardContent>
   219→        </Card>
   220→        <Card>
   221→          <CardContent className="p-4 flex items-center gap-3">
   222→            <div className="h-10 w-10 rounded-lg bg-red-100 flex items-center justify-center">
   223→              <AlertTriangle className="h-5 w-5 text-red-600" />
   224→            </div>
   225→            <div>
   226→              <p className="text-2xl font-bold text-red-600">{lowStockCount}</p>
   227→              <p className="text-xs text-muted-foreground">Low Stock Alerts</p>
   228→            </div>
   229→          </CardContent>
   230→        </Card>
   231→        <Card>
   232→          <CardContent className="p-4 flex items-center gap-3">
   233→            <div className="h-10 w-10 rounded-lg bg-amber-100 flex items-center justify-center">
   234→              <AlertTriangle className="h-5 w-5 text-amber-600" />
   235→            </div>
   236→            <div>
   237→              <p className="text-2xl font-bold text-amber-600">{outOfStockCount}</p>
   238→              <p className="text-xs text-muted-foreground">Out of Stock</p>
   239→            </div>
   240→          </CardContent>
   241→        </Card>
   242→        <Card>
   243→          <CardContent className="p-4 flex items-center gap-3">
   244→            <div className="h-10 w-10 rounded-lg bg-teal-100 flex items-center justify-center">
   245→              <TrendingUp className="h-5 w-5 text-teal-600" />
   246→            </div>
   247→            <div>
   248→              <p className="text-2xl font-bold">${totalValue.toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 2 })}</p>
   249→              <p className="text-xs text-muted-foreground">Inventory Value</p>
   250→            </div>
   251→          </CardContent>
   252→        </Card>
   253→      </div>
   254→
   255→      {/* Filters & Actions */}
   256→      <Card>
   257→        <CardContent className="p-4">
   258→          <div className="flex flex-col sm:flex-row gap-3">
   259→            <div className="relative flex-1">
   260→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   261→              <Input
   262→                placeholder="Search by product name or NDC..."
   263→                value={searchQuery}
   264→                onChange={(e) => setSearchQuery(e.target.value)}
   265→                className="pl-9"
   266→              />
   267→            </div>
   268→            <Select value={categoryFilter} onValueChange={setCategoryFilter}>
   269→              <SelectTrigger className="w-full sm:w-40">
   270→                <Filter className="h-4 w-4 mr-2" />
   271→                <SelectValue />
   272→              </SelectTrigger>
   273→              <SelectContent>
   274→                <SelectItem value="ALL">All Categories</SelectItem>
   275→                <SelectItem value="OTC">OTC</SelectItem>
   276→                <SelectItem value="PRESCRIPTION">Prescription</SelectItem>
   277→                <SelectItem value="SUPPLEMENT">Supplement</SelectItem>
   278→                <SelectItem value="MEDICAL_DEVICE">Medical Device</SelectItem>
   279→                <SelectItem value="PERSONAL_CARE">Personal Care</SelectItem>
   280→                <SelectItem value="CONSUMABLES">Consumables</SelectItem>
   281→              </SelectContent>
   282→            </Select>
   283→            <Select value={stockFilter} onValueChange={setStockFilter}>
   284→              <SelectTrigger className="w-full sm:w-36">
   285→                <SelectValue />
   286→              </SelectTrigger>
   287→              <SelectContent>
   288→                <SelectItem value="ALL">All Stock</SelectItem>
   289→                <SelectItem value="LOW">Low Stock</SelectItem>
   290→                <SelectItem value="OUT">Out of Stock</SelectItem>
   291→                <SelectItem value="OK">In Stock OK</SelectItem>
   292→              </SelectContent>
   293→            </Select>
   294→            <Button onClick={() => setAddProductDialog(true)} className="bg-teal-600 hover:bg-teal-700">
   295→              <PackagePlus className="h-4 w-4 mr-2" />
   296→              Add Product
   297→            </Button>
   298→          </div>
   299→        </CardContent>
   300→      </Card>
   301→
   302→      {/* Inventory Table */}
   303→      <Card>
   304→        <CardContent className="p-0">
   305→          <Table>
   306→            <TableHeader>
   307→              <TableRow>
   308→                <TableHead className="cursor-pointer" onClick={() => { setSortBy('name'); setSortDir(sortDir === 'asc' ? 'desc' : 'asc') }}>
   309→                  <span className="flex items-center gap-1">Product <ArrowUpDown className="h-3 w-3" /></span>
   310→                </TableHead>
   311→                <TableHead className="hidden md:table-cell">NDC</TableHead>
   312→                <TableHead className="cursor-pointer hidden sm:table-cell" onClick={() => { setSortBy('category'); setSortDir(sortDir === 'asc' ? 'desc' : 'asc') }}>
   313→                  <span className="flex items-center gap-1">Category <ArrowUpDown className="h-3 w-3" /></span>
   314→                </TableHead>
   315→                <TableHead className="cursor-pointer text-right" onClick={() => { setSortBy('stock'); setSortDir(sortDir === 'asc' ? 'desc' : 'asc') }}>
   316→                  <span className="flex items-center justify-end gap-1">In Stock <ArrowUpDown className="h-3 w-3" /></span>
   317→                </TableHead>
   318→                <TableHead className="hidden lg:table-cell text-right">Reorder Point</TableHead>
   319→                <TableHead className="text-right">Status</TableHead>
   320→                <TableHead className="hidden lg:table-cell text-right">Cost</TableHead>
   321→                <TableHead className="hidden lg:table-cell text-right">Retail</TableHead>
   322→                <TableHead className="text-right">Actions</TableHead>
   323→              </TableRow>
   324→            </TableHeader>
   325→            <TableBody>
   326→              {loading ? (
   327→                Array.from({ length: 8 }).map((_, i) => (
   328→                  <TableRow key={i}>
   329→                    {Array.from({ length: 9 }).map((_, j) => (
   330→                      <TableCell key={j}><Skeleton className="h-5 w-full" /></TableCell>
   331→                    ))}
   332→                  </TableRow>
   333→                ))
   334→              ) : filteredItems.length === 0 ? (
   335→                <TableRow>
   336→                  <TableCell colSpan={9} className="text-center py-8 text-muted-foreground">
   337→                    No inventory items found
   338→                  </TableCell>
   339→                </TableRow>
   340→              ) : (
   341→                filteredItems.map((item) => {
   342→                  const qty = Number(item.quantity) || 0
   343→                  const reorder = Number(item.product.reorderPoint) || 10
   344→                  const isOut = qty === 0
   345→                  const isLow = qty > 0 && qty <= reorder
   346→                  return (
   347→                    <TableRow key={item.id} className={isOut ? 'bg-red-50/50' : isLow ? 'bg-amber-50/50' : ''}>
   348→                      <TableCell>
   349→                        <div>
   350→                          <p className="font-medium text-sm">{item.product.name}</p>
   351→                          <p className="text-xs text-muted-foreground">
   352→                            {item.product.strength} {item.product.dosageForm} · {item.product.unitOfMeasure}
   353→                          </p>
   354→                        </div>
   355→                      </TableCell>
   356→                      <TableCell className="hidden md:table-cell text-xs font-mono">{item.product.ndc || '—'}</TableCell>
   357→                      <TableCell className="hidden sm:table-cell">
   358→                        <Badge variant="outline" className="text-xs">{item.product.category.replace(/_/g, ' ')}</Badge>
   359→                      </TableCell>
   360→                      <TableCell className="text-right font-bold">{qty}</TableCell>
   361→                      <TableCell className="hidden lg:table-cell text-right text-muted-foreground">{reorder}</TableCell>
   362→                      <TableCell className="text-right">
   363→                        {isOut ? (
   364→                          <Badge className="bg-red-100 text-red-700 border-red-200">Out of Stock</Badge>
   365→                        ) : isLow ? (
   366→                          <Badge className="bg-amber-100 text-amber-700 border-amber-200">Low Stock</Badge>
   367→                        ) : (
   368→                          <Badge className="bg-emerald-100 text-emerald-700 border-emerald-200">In Stock</Badge>
   369→                        )}
   370→                      </TableCell>
   371→                      <TableCell className="hidden lg:table-cell text-right">${item.product.costPrice?.toFixed(2) || '—'}</TableCell>
   372→                      <TableCell className="hidden lg:table-cell text-right">${item.product.sellingPrice.toFixed(2)}</TableCell>
   373→                      <TableCell className="text-right">
   374→                        <Button size="sm" variant="ghost" onClick={() => { setSelectedItem(item); setAdjustDialog(true) }}>
   375→                          <Edit className="h-3.5 w-3.5 mr-1" />
   376→                          Adjust
   377→                        </Button>
   378→                      </TableCell>
   379→                    </TableRow>
   380→                  )
   381→                })
   382→              )}
   383→            </TableBody>
   384→          </Table>
   385→        </CardContent>
   386→      </Card>
   387→
   388→      {/* Add Product Dialog */}
   389→      <Dialog open={addProductDialog} onOpenChange={setAddProductDialog}>
   390→        <DialogContent className="max-w-lg">
   391→          <DialogHeader>
   392→            <DialogTitle className="flex items-center gap-2">
   393→              <PackagePlus className="h-5 w-5 text-teal-600" />
   394→              Add New Product
   395→            </DialogTitle>
   396→          </DialogHeader>
   397→          <div className="grid grid-cols-2 gap-4 max-h-[70vh] overflow-y-auto pr-1">
   398→            {/* Product Name */}
   399→            <div className="col-span-2">
   400→              <Label htmlFor="prod-name">Product Name <span className="text-red-500">*</span></Label>
   401→              <Input
   402→                id="prod-name"
   403→                placeholder="e.g., Amoxicillin 500mg Capsules"
   404→                value={productForm.name}
   405→                onChange={(e) => setProductForm({ ...productForm, name: e.target.value })}
   406→                className="mt-1"
   407→              />
   408→            </div>
   409→
   410→            {/* SKU */}
   411→            <div>
   412→              <Label htmlFor="prod-sku">SKU</Label>
   413→              <Input
   414→                id="prod-sku"
   415→                placeholder="e.g., SKU-00123"
   416→                value={productForm.sku}
   417→                onChange={(e) => setProductForm({ ...productForm, sku: e.target.value })}
   418→                className="mt-1"
   419→              />
   420→            </div>
   421→
   422→            {/* Category */}
   423→            <div>
   424→              <Label htmlFor="prod-category">Category</Label>
   425→              <Select value={productForm.category} onValueChange={(v) => setProductForm({ ...productForm, category: v })}>
   426→                <SelectTrigger className="mt-1">
   427→                  <SelectValue />
   428→                </SelectTrigger>
   429→                <SelectContent>
   430→                  <SelectItem value="OTC">OTC</SelectItem>
   431→                  <SelectItem value="PRESCRIPTION">Prescription</SelectItem>
   432→                  <SelectItem value="SUPPLEMENT">Supplement</SelectItem>
   433→                  <SelectItem value="MEDICAL_DEVICE">Medical Device</SelectItem>
   434→                  <SelectItem value="PERSONAL_CARE">Personal Care</SelectItem>
   435→                  <SelectItem value="CONSUMABLES">Consumables</SelectItem>
   436→                  {categories.filter((c) => !['OTC','PRESCRIPTION','SUPPLEMENT','MEDICAL_DEVICE','PERSONAL_CARE','CONSUMABLES'].includes(c.name)).map((cat) => (
   437→                    <SelectItem key={cat.id} value={cat.name}>{cat.name.replace(/_/g, ' ')}</SelectItem>
   438→                  ))}
   439→                </SelectContent>
   440→              </Select>
   441→            </div>
   442→
   443→            {/* Price */}
   444→            <div>
   445→              <Label htmlFor="prod-price">Price ($) <span className="text-red-500">*</span></Label>
   446→              <Input
   447→                id="prod-price"
   448→                type="number"
   449→                step="0.01"
   450→                min="0"
   451→                placeholder="0.00"
   452→                value={productForm.price}
   453→                onChange={(e) => setProductForm({ ...productForm, price: e.target.value })}
   454→                className="mt-1"
   455→              />
   456→            </div>
   457→
   458→            {/* Stock Quantity */}
   459→            <div>
   460→              <Label htmlFor="prod-stock">Stock Quantity</Label>
   461→              <Input
   462→                id="prod-stock"
   463→                type="number"
   464→                min="0"
   465→                placeholder="0"
   466→                value={productForm.stockQuantity}
   467→                onChange={(e) => setProductForm({ ...productForm, stockQuantity: e.target.value })}
   468→                className="mt-1"
   469→              />
   470→            </div>
   471→
   472→            {/* Min Stock Level */}
   473→            <div>
   474→              <Label htmlFor="prod-minstock">Min Stock Level</Label>
   475→              <Input
   476→                id="prod-minstock"
   477→                type="number"
   478→                min="0"
   479→                placeholder="10"
   480→                value={productForm.minStockLevel}
   481→                onChange={(e) => setProductForm({ ...productForm, minStockLevel: e.target.value })}
   482→                className="mt-1"
   483→              />
   484→            </div>
   485→
   486→            {/* Expiry Date */}
   487→            <div>
   488→              <Label htmlFor="prod-expiry">Expiry Date</Label>
   489→              <Input
   490→                id="prod-expiry"
   491→                type="date"
   492→                value={productForm.expiryDate}
   493→                onChange={(e) => setProductForm({ ...productForm, expiryDate: e.target.value })}
   494→                className="mt-1"
   495→              />
   496→            </div>
   497→
   498→            {/* Barcode */}
   499→            <div>
   500→              <Label htmlFor="prod-barcode">Barcode</Label>
   501→              <Input
   502→                id="prod-barcode"
   503→                placeholder="e.g., 1234567890123"
