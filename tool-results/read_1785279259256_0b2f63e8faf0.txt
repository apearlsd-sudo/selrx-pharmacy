   142→   142→export function StockTakeReportView({ stockTakeId: stockTakeIdProp }: { stockTakeId?: string }) {
   143→   143→  // Read stockTakeReportId from store reactively as fallback when prop is missing
   144→   144→  const storeReportId = useAppStore((s) => s.stockTakeReportId)
   145→   145→  const stockTakeId = stockTakeIdProp || storeReportId || undefined
   146→   146→
   147→   147→  const [report, setReport] = useState<ReportData | null>(null)
   148→   148→  const [loading, setLoading] = useState(true)
   149→   149→  const [error, setError] = useState<string | null>(null)
   150→   150→  const [expandedSections, setExpandedSections] = useState<Record<string, boolean>>({
   151→   151→    expired: true,
   152→   152→    nearExpiry: true,
   153→   153→    variance: true,
   154→   154→    reorder: true,
   155→   155→    valuation: true,
   156→   156→  })
   157→   157→  const [sortField, setSortField] = useState<string>('')
   158→   158→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('asc')
   159→   159→  const addToast = useAppStore((s) => s.addToast)
   160→   160→
   161→   161→  const fetchReport = useCallback(async () => {
   162→   162→    if (!stockTakeId) {
   163→   163→      setLoading(false)
   164→   164→      setError('No stock take selected')
   165→   165→      return
   166→   166→    }
   167→   167→    setLoading(true)
   168→   168→    setError(null)
   169→   169→    try {
   170→   170→      const res = await fetch(`/api/stock-take?action=report&id=${stockTakeId}`, { headers: authHeaders() })
   171→   171→      if (res.ok) {
   172→   172→        const data = await res.json()
   173→   173→        // Validate that the response has the expected structure before setting state
   174→   174→        if (data && data.stockTakeRef && data.inventoryValuation) {
   175→   175→          setReport(data)
   176→   176→        } else {
   177→   177→          console.error('[StockTakeReport] Invalid report data:', data)
   178→   178→          setError('Invalid report data received from server')
   179→   179→        }
   180→   180→      } else {
   181→   181→        const err = await res.json().catch(() => ({ error: 'Failed to load report' }))
   182→   182→        setError(err.error || 'Failed to load report')
   183→   183→      }
   184→   184→    } catch {
   185→   185→      setError('Network error while loading report')
   186→   186→    } finally {
   187→   187→      setLoading(false)
   188→   188→    }
   189→   189→  }, [stockTakeId])
   190→   190→
   191→   191→  useEffect(() => { fetchReport() }, [fetchReport])
   192→   192→
   193→   193→  const toggleSection = (section: string) => {
   194→   194→    setExpandedSections((prev) => ({ ...prev, [section]: !prev[section] }))
   195→   195→  }
   196→   196→
   197→   197→  const handlePrint = () => window.print()
   198→   198→  const handleExportCSV = () => {
   199→   199→    if (!report) return
   200→   200→    try {
   201→   201→      const lines: string[] = []
   202→   202→      // Expired goods
   203→   203→      lines.push('=== EXPIRED GOODS ===')
   204→   204→      lines.push('Product,NDC,Category,Strength,Form,Qty,Cost Price,Selling Price,Total Cost,Potential Revenue,Expiry Date,Days Since Expiry,Manufacturer,Vendor')
   205→   205→      for (const item of report.expiredGoods.items) {
   206→   206→        lines.push(`"${item.productName}","${item.ndc || ''}","${item.category}","${item.strength || ''}","${item.dosageForm || ''}",${item.countedQty},${item.costPrice},${item.sellingPrice},${item.totalCost},${item.potentialRevenue},"${item.expiryDate || ''}",${item.daysSinceExpiry},"${item.manufacturer || ''}","${item.vendor || ''}"`)
   207→   207→      }

   209→   209→      // Near expiry

   213→   213→        lines.push(`"${item.productName}","${item.ndc || ''}","${item.category}","${item.strength || ''}","${item.dosageForm || ''}",${item.countedQty},${item.costPrice},${item.sellingPrice},${item.totalCost},${item.potentialRevenue},"${item.expiryDate || ''}",${item.daysToExpiry},"${item.manufacturer || ''}","${item.vendor || ''}"`)
   214→   214→      }
   215→   215→      lines.push('')
   216→   216→      // Variance

   221→   221→      }
   222→   222→      lines.push('')
   223→   223→      // Reorder alerts
   224→   224→      lines.push('=== REORDER ALERTS ===')

   228→   228→      }
   229→   229→
   230→   230→      const blob = new Blob([lines.join('\n')], { type: 'text/csv' })

   232→   232→      const a = document.createElement('a')

   235→   235→      a.click()

   240→   240→    }
   241→   241→  }

   247→   247→        <Skeleton className="h-8 w-64" />

   268→   268→  const accuracyRate = report.totalItemsChecked > 0
   269→   269→    ? ((report.itemsMatched / report.totalItemsChecked) * 100).toFixed(1)
   270→   270→    : '0'
   271→   271→
   272→   272→  const SectionHeader = ({
   273→   273→    id, icon: Icon, title, badge, badgeColor, count, subtitle,

   284→   284→          <Icon className="h-4.5 w-4.5 text-emerald-600" />
   285→   285→        </div>

   287→   287→          <div className="flex items-center gap-2">

   303→   303→  return (

   308→   308→          <Button variant="ghost" size="sm" onClick={() => useAppStore.getState().setCurrentView('stock-take')} className="print:hidden">
   309→   309→            <ArrowLeft className="h-4 w-4 mr-1" /> Back
   310→   310→          </Button>
   311→   311→          <div className="h-10 w-10 rounded-lg bg-emerald-100 flex items-center justify-center">
   312→   312→            <FileText className="h-5 w-5 text-emerald-600" />
