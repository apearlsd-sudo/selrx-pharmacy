   106→function CategoryModal({
   107→  open,
   108→  onOpenChange,
   109→  editingCategory,
   110→  onSaved,
   111→}: {
   112→  open: boolean
   113→  onOpenChange: (open: boolean) => void
   114→  editingCategory: Category | null
   115→  onSaved: (category: Category) => void
   116→}) {
   117→  const [name, setName] = useState('')
   118→  const [description, setDescription] = useState('')
   119→  const [saving, setSaving] = useState(false)
   120→  const addToast = useAppStore((s) => s.addToast)
   121→  const isEditing = !!editingCategory
   122→
   123→  useEffect(() => {
   124→    if (open) {
   125→      setName(editingCategory?.name.replace(/_/g, ' ') || '')
   126→      setDescription(editingCategory?.description || '')
   127→      setSaving(false)
   128→    }
   129→  }, [open, editingCategory])
   130→
   131→  const handleSave = async () => {
   132→    if (!name.trim()) return
   133→    setSaving(true)
   134→    try {
   135→      if (isEditing) {
   136→        const res = await fetch(`/api/categories/${editingCategory.id}`, {
   137→          method: 'PUT',
   138→          headers: { 'Content-Type': 'application/json' },
   139→          body: JSON.stringify({ name: name.trim(), description: description.trim() || null }),
   140→        })
   141→        if (!res.ok) {
   142→          const err = await res.json()
   143→          throw new Error(err.error || 'Failed to update')
   144→        }
   145→        const updated = await res.json()
   146→        addToast({ title: 'Category Updated', description: `"${name.trim()}" updated successfully`, variant: 'success' })
   147→        onSaved(updated)
   148→      } else {
   149→        const res = await fetch('/api/categories', {
   150→          method: 'POST',
   151→          headers: { 'Content-Type': 'application/json' },
   152→          body: JSON.stringify({ name: name.trim(), description: description.trim() || null }),
   153→        })
   154→        if (!res.ok) {
   155→          const err = await res.json()
   156→          throw new Error(err.error || 'Failed to create')
   157→        }
   158→        const created = await res.json()
   159→        addToast({ title: 'Category Created', description: `"${name.trim()}" added successfully`, variant: 'success' })
   160→        onSaved(created)
   161→      }
   162→      onOpenChange(false)
   163→    } catch (err: any) {
   164→      addToast({ title: 'Error', description: err.message, variant: 'destructive' })
   165→    } finally {
   166→      setSaving(false)
   167→    }
   168→  }
   169→
   170→  return (
   171→    <Dialog open={open} onOpenChange={onOpenChange}>
   172→      <DialogContent className="sm:max-w-lg">
   173→        <DialogHeader>
   174→          <DialogTitle>{isEditing ? 'Edit Category' : 'Add Category'}</DialogTitle>
   175→          <DialogDescription>{isEditing ? 'Update category details' : 'Create a new drug category'}</DialogDescription>
   176→        </DialogHeader>
   177→        <div className="grid gap-3">
   178→          <div>
   179→            <Label className="text-xs">Category Name <span className="text-red-500">*</span></Label>
   180→            <Input value={name} onChange={(e) => setName(e.target.value)} placeholder="e.g., ANALGESICS" className="mt-1" onKeyDown={(e) => { if (e.key === 'Enter') handleSave() }} />
   181→          </div>
   182→          <div>
   183→            <Label className="text-xs">Description</Label>
   184→            <Textarea value={description} onChange={(e) => setDescription(e.target.value)} placeholder="Brief description of this category" className="mt-1" rows={2} />
   185→          </div>
   186→        </div>
   187→        <DialogFooter>
   188→          <Button variant="outline" onClick={() => onOpenChange(false)}>Cancel</Button>
   189→          <Button onClick={handleSave} disabled={!name.trim() || saving} className="bg-emerald-600 hover:bg-emerald-700">
   190→            {saving ? 'Saving...' : <><Save className="h-4 w-4 mr-2" /> {isEditing ? 'Update' : 'Create'}</>}
   191→          </Button>
   192→        </DialogFooter>
   193→      </DialogContent>
   194→    </Dialog>
   195→  )
   196→}
