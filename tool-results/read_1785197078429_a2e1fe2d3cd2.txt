   127→function CategorySection() {
   128→  const [categories, setCategories] = useState<Category[]>([])
   129→  const [loading, setLoading] = useState(true)
   130→  const [newName, setNewName] = useState('')
   131→  const [newDesc, setNewDesc] = useState('')
   132→  const [saving, setSaving] = useState(false)
   133→  const [search, setSearch] = useState('')
   134→  const addToast = useAppStore((s) => s.addToast)
   135→
   136→  const fetchCategories = useCallback(async () => {
   137→    setLoading(true)
   138→    try {
   139→      const res = await fetch('/api/categories')
   140→      if (res.ok) setCategories(await res.json())
   141→    } catch {
   142→      addToast({ title: 'Error', description: 'Failed to load categories', variant: 'destructive' })
   143→    } finally {
   144→      setLoading(false)
   145→    }
   146→  }, [addToast])
   147→
   148→  useEffect(() => { fetchCategories() }, [fetchCategories])
   149→
   150→  const handleCreate = async () => {
   151→    if (!newName.trim()) return
   152→    setSaving(true)
   153→    try {
   154→      const res = await fetch('/api/categories', {
   155→        method: 'POST',
   156→        headers: { 'Content-Type': 'application/json' },
   157→        body: JSON.stringify({ name: newName.trim(), description: newDesc.trim() || null }),
   158→      })
   159→      if (!res.ok) {
   160→        const err = await res.json()
   161→        throw new Error(err.error || 'Failed')
   162→      }
   163→      addToast({ title: 'Category Created', description: `"${newName.trim()}" added successfully`, variant: 'success' })
   164→      setNewName('')
   165→      setNewDesc('')
   166→      fetchCategories()
   167→    } catch (err: any) {
   168→      addToast({ title: 'Error', description: err.message, variant: 'destructive' })
   169→    } finally {
   170→      setSaving(false)
   171→    }
   172→  }
   173→
   174→  const handleDelete = async (cat: Category) => {
   175→    try {
   176→      const res = await fetch(`/api/categories/${cat.id}`, { method: 'DELETE' })
   177→      if (!res.ok) {
   178→        const err = await res.json()
   179→        throw new Error(err.error || 'Failed')
   180→      }
   181→      addToast({ title: 'Deleted', description: `"${cat.name.replace(/_/g, ' ')}" removed`, variant: 'success' })
   182→      fetchCategories()
   183→    } catch (err: any) {
   184→      addToast({ title: 'Error', description: err.message, variant: 'destructive' })
   185→    }
   186→  }
   187→
   188→  const filtered = categories.filter((c) =>
   189→    c.name.toLowerCase().includes(search.toLowerCase()) ||
   190→    (c.description && c.description.toLowerCase().includes(search.toLowerCase()))
   191→  )
   192→
   193→  return (
   194→    <div className="space-y-4">
   195→      {/* Create Form */}
   196→      <Card>
   197→        <CardContent className="p-4">
   198→          <h3 className="text-sm font-semibold mb-3 flex items-center gap-2">
   199→            <Tags className="h-4 w-4 text-emerald-600" />
   200→            Create New Category
   201→          </h3>
   202→          <div className="grid grid-cols-1 sm:grid-cols-12 gap-3">
   203→            <div className="sm:col-span-3">
   204→              <Label className="text-xs">Category Name *</Label>
   205→              <Input
   206→                placeholder="e.g., ANALGESICS"
   207→                value={newName}
   208→                onChange={(e) => setNewName(e.target.value)}
   209→                className="mt-1"
   210→                onKeyDown={(e) => { if (e.key === 'Enter') handleCreate() }}
   211→              />
   212→            </div>
   213→            <div className="sm:col-span-6">
   214→              <Label className="text-xs">Description</Label>
   215→              <Input
   216→                placeholder="Brief description of this category"
   217→                value={newDesc}
   218→                onChange={(e) => setNewDesc(e.target.value)}
   219→                className="mt-1"
   220→                onKeyDown={(e) => { if (e.key === 'Enter') handleCreate() }}
   221→              />
   222→            </div>
   223→            <div className="sm:col-span-3 flex items-end">
   224→              <Button
   225→                onClick={handleCreate}
   226→                disabled={!newName.trim() || saving}
   227→                className="w-full bg-emerald-600 hover:bg-emerald-700"
   228→              >
   229→                {saving ? 'Saving...' : <><Plus className="h-4 w-4 mr-2" /> Add Category</>}
   230→              </Button>
   231→            </div>
   232→          </div>
   233→        </CardContent>
   234→      </Card>
   235→
   236→      {/* Search */}
   237→      <div className="relative max-w-sm">
   238→        <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   239→        <Input placeholder="Search categories..." value={search} onChange={(e) => setSearch(e.target.value)} className="pl-9" />
   240→      </div>
   241→
   242→      {/* Categories Table */}
   243→      <Card>
   244→        <CardContent className="p-0">
   245→          <Table>
   246→            <TableHeader>
   247→              <TableRow>
   248→                <TableHead>Category Name</TableHead>
   249→                <TableHead className="hidden sm:table-cell">Description</TableHead>
   250→                <TableHead className="text-center">Products</TableHead>
   251→                <TableHead className="text-center">Type</TableHead>
   252→                <TableHead className="text-right">Actions</TableHead>
   253→              </TableRow>
   254→            </TableHeader>
   255→            <TableBody>
   256→              {loading ? (
   257→                Array.from({ length: 5 }).map((_, i) => (
   258→                  <TableRow key={i}>
   259→                    {Array.from({ length: 5 }).map((_, j) => (
   260→                      <TableCell key={j}><Skeleton className="h-5 w-full" /></TableCell>
   261→                    ))}
   262→                  </TableRow>
   263→                ))
   264→              ) : filtered.length === 0 ? (
   265→                <TableRow>
   266→                  <TableCell colSpan={5} className="text-center py-8 text-muted-foreground">No categories found</TableCell>
   267→                </TableRow>
   268→              ) : (
   269→                filtered.map((cat) => {
   270→                  const isBuiltIn = BUILT_IN_CATEGORIES.includes(cat.name)
   271→                  const prodCount = cat._count?.products || 0
   272→                  return (
   273→                    <TableRow key={cat.id}>
   274→                      <TableCell>
   275→                        <Badge variant="outline" className="font-mono text-xs">{cat.name.replace(/_/g, ' ')}</Badge>
   276→                      </TableCell>
   277→                      <TableCell className="hidden sm:table-cell text-sm text-muted-foreground">{cat.description || '—'}</TableCell>
   278→                      <TableCell className="text-center">
   279→                        <Badge variant="secondary" className="text-xs">{prodCount}</Badge>
   280→                      </TableCell>
   281→                      <TableCell className="text-center">
   282→                        {isBuiltIn ? (
   283→                          <Badge className="text-[10px] bg-blue-100 text-blue-700 border-blue-200">System</Badge>
   284→                        ) : (
   285→                          <Badge className="text-[10px] bg-purple-100 text-purple-700 border-purple-200">Custom</Badge>
   286→                        )}
   287→                      </TableCell>
   288→                      <TableCell className="text-right">
   289→                        <Button
   290→                          size="sm" variant="ghost"
   291→                          className="text-red-400 hover:text-red-600 hover:bg-red-50"
   292→                          onClick={() => handleDelete(cat)}
   293→                          disabled={isBuiltIn || prodCount > 0}
   294→                          title={isBuiltIn ? 'System category' : prodCount > 0 ? `${prodCount} products linked` : 'Delete'}
   295→                        >
   296→                          <Trash2 className="h-3.5 w-3.5" />
   297→                        </Button>
   298→                      </TableCell>
   299→                    </TableRow>
   300→                  )
   301→                })
   302→              )}
   303→            </TableBody>
   304→          </Table>
   305→        </CardContent>
   306→      </Card>
   307→    </div>
   308→  )
   309→}

   313→function DrugSection() {
   314→  const [drugs, setDrugs] = useState<DrugProduct[]>([])
   315→  const [categories, setCategories] = useState<Category[]>([])
   316→  const [vendors, setVendors] = useState<Vendor[]>([])
   317→  const [loading, setLoading] = useState(true)
   318→  const [saving, setSaving] = useState(false)
   319→  const [search, setSearch] = useState('')
   320→  const addToast = useAppStore((s) => s.addToast)
   321→
   322→  const [form, setForm] = useState({
   323→    name: '', sku: '', category: 'OTC', dosageForm: '', manufacturer: '', costPrice: '', sellingPrice: '',
   324→    stockQuantity: '0', minStockLevel: '10', expiryDate: '', barcode: '', vendorId: '',
   325→  })
   326→  const [showNewManufacturer, setShowNewManufacturer] = useState(false)
   327→  const [newManufacturerName, setNewManufacturerName] = useState('')
   328→  const [showNewVendor, setShowNewVendor] = useState(false)
   329→  const [newVendorName, setNewVendorName] = useState('')
   330→  const [showNewDosageForm, setShowNewDosageForm] = useState(false)
   331→  const [newDosageFormName, setNewDosageFormName] = useState('')
   332→  const [showNewCategory, setShowNewCategory] = useState(false)
   333→  const [newCategoryName, setNewCategoryName] = useState('')
   334→
   335→  const fetchData = useCallback(async () => {
   336→    setLoading(true)
   337→    try {
   338→      const [prodRes, catRes, venRes] = await Promise.all([
   339→        fetch('/api/products'),
   340→        fetch('/api/categories'),
   341→        fetch('/api/vendors'),
   342→      ])
   343→      if (prodRes.ok) {
   344→        const data = await prodRes.json()
   345→        const list = Array.isArray(data) ? data : data.products || []
   346→        setDrugs(list)
   347→      }
   348→      if (catRes.ok) setCategories(await catRes.json())
   349→      if (venRes.ok) setVendors(await venRes.json())
   350→    } catch {
   351→      addToast({ title: 'Error', description: 'Failed to load data', variant: 'destructive' })
   352→    } finally {
   353→      setLoading(false)
   354→    }
   355→  }, [addToast])
   356→
   357→  useEffect(() => { fetchData() }, [fetchData])
   358→
   359→  const handleCreate = async () => {
   360→    if (!form.name || !form.sellingPrice) return
   361→    setSaving(true)
   362→    try {
   363→      const res = await fetch('/api/products', {
   364→        method: 'POST',
   365→        headers: {
   366→          'Content-Type': 'application/json',
   367→          'x-user-role': 'SUPER_ADMIN',
   368→        },
   369→        body: JSON.stringify({
   370→          name: form.name,
   371→          ndc: form.sku || null,
   372→          category: form.category,
   373→          dosageForm: form.dosageForm || null,
   374→          manufacturer: form.manufacturer || null,
   375→          costPrice: form.costPrice ? parseFloat(form.costPrice) : null,
   376→          sellingPrice: parseFloat(form.sellingPrice),
   377→          reorderPoint: parseInt(form.minStockLevel) || 10,
   378→          expiryDate: form.expiryDate || null,
   379→          batchNumber: form.barcode || null,
   380→          vendorId: form.vendorId || null,
   381→        }),
   382→      })
   383→      if (!res.ok) {
   384→        const err = await res.json()
   385→        throw new Error(err.error || 'Failed to create product')
   386→      }
   387→      const newProduct = await res.json()
   388→
   389→      // Set initial stock
   390→      const qty = parseInt(form.stockQuantity) || 0
   391→      if (qty > 0) {
   392→        await fetch('/api/inventory', {
   393→          method: 'PUT',
   394→          headers: { 'Content-Type': 'application/json' },
   395→          body: JSON.stringify({ productId: newProduct.id, adjustment: qty, reason: 'Initial stock on creation' }),
   396→        })
   397→      }
   398→
   399→      addToast({ title: 'Drug Added', description: `"${form.name}" registered in inventory`, variant: 'success' })
   400→      setForm({
   401→        name: '', sku: '', category: 'OTC', dosageForm: '', manufacturer: '', costPrice: '', sellingPrice: '',
   402→        stockQuantity: '0', minStockLevel: '10', expiryDate: '', barcode: '', vendorId: '',
   403→      })
   404→      setShowNewManufacturer(false)
   405→      setNewManufacturerName('')
   406→      fetchData()
   407→    } catch (err: any) {
   408→      addToast({ title: 'Error', description: err.message || 'Failed to add drug', variant: 'destructive' })
   409→    } finally {
   410→      setSaving(false)
   411→    }
   412→  }
   413→
   414→  const filtered = drugs.filter((d) =>
   415→    d.name.toLowerCase().includes(search.toLowerCase()) ||
   416→    (d.ndc && d.ndc.toLowerCase().includes(search.toLowerCase()))
   417→  )
   418→
   419→  // Extract unique manufacturers from existing drugs
   420→  const existingManufacturers = [...new Set(drugs.map((d) => d.manufacturer).filter(Boolean))] as string[]
   421→
   422→  const handleAddNewManufacturer = () => {
   423→    const trimmed = newManufacturerName.trim()
   424→    if (!trimmed) return
   425→    setForm({ ...form, manufacturer: trimmed })
   426→    setNewManufacturerName('')
   427→    setShowNewManufacturer(false)
   428→  }
   429→
   430→  const handleAddNewVendor = async () => {
   431→    const trimmed = newVendorName.trim()
   432→    if (!trimmed) return
   433→    try {
   434→      const res = await fetch('/api/vendors', {
   435→        method: 'POST',
   436→        headers: { 'Content-Type': 'application/json' },
   437→        body: JSON.stringify({ name: trimmed }),
   438→      })
   439→      if (res.ok) {
   440→        const vendor = await res.json()
   441→        setForm({ ...form, vendorId: vendor.id })
   442→        setVendors((prev) => [...prev, { id: vendor.id, name: trimmed, contactPerson: null, email: null, phone: null, address: null, notes: null }])
   443→        setNewVendorName('')
   444→        setShowNewVendor(false)
   445→      } else {
   446→        const err = await res.json()
   447→        addToast({ title: 'Error', description: err.error || 'Failed to create vendor', variant: 'destructive' })
   448→      }
   449→    } catch {
   450→      addToast({ title: 'Error', description: 'Failed to create vendor', variant: 'destructive' })
   451→    }
   452→  }
   453→
   454→  const handleAddNewDosageForm = () => {
   455→    const trimmed = newDosageFormName.trim()
   456→    if (!trimmed) return
   457→    setForm({ ...form, dosageForm: trimmed })
   458→    setNewDosageFormName('')
   459→    setShowNewDosageForm(false)
   460→  }
   461→
   462→  const handleAddNewCategory = async () => {
   463→    const trimmed = newCategoryName.trim()
   464→    if (!trimmed) return
   465→    try {
   466→      const res = await fetch('/api/categories', {
   467→        method: 'POST',
   468→        headers: { 'Content-Type': 'application/json' },
   469→        body: JSON.stringify({ name: trimmed }),
   470→      })
   471→      if (res.ok) {
   472→        const cat = await res.json()
   473→        setForm({ ...form, category: cat.name })
   474→        setCategories((prev) => [...prev, { id: cat.id, name: cat.name, description: null }])
   475→        setNewCategoryName('')
   476→        setShowNewCategory(false)
   477→      } else {
   478→        const err = await res.json()
   479→        addToast({ title: 'Error', description: err.error || 'Failed to create category', variant: 'destructive' })
   480→      }
   481→    } catch {
   482→      addToast({ title: 'Error', description: 'Failed to create category', variant: 'destructive' })
   483→    }
   484→  }
   485→
   486→  return (
   487→    <div className="space-y-4">
   488→      {/* Drug Registration Form */}
   489→      <Card>
   490→        <CardContent className="p-4">
   491→          <h3 className="text-sm font-semibold mb-3 flex items-center gap-2">
   492→            <Pill className="h-4 w-4 text-teal-600" />
   493→            Register New Drug / Product
   494→          </h3>
   495→          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-3">
   496→            {/* Product Name - full width */}
   497→            <div className="sm:col-span-2 lg:col-span-3">
   498→              <Label className="text-xs">Product Name <span className="text-red-500">*</span></Label>
   499→              <Input placeholder="e.g., Paracetamol 500mg" value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} className="mt-1" />
   500→            </div>
   501→
   502→            {/* SKU */}
   503→            <div>
   504→              <Label className="text-xs">SKU / NDC</Label>
   505→              <Input placeholder="e.g., SKU-00123" value={form.sku} onChange={(e) => setForm({ ...form, sku: e.target.value })} className="mt-1" />
   506→            </div>
   507→
   508→            {/* Manufacturer Dropdown */}
   509→            {showNewManufacturer ? (
   510→              <div>
   511→                <Label className="text-xs">New Manufacturer</Label>
   512→                <div className="flex gap-1 mt-1">
   513→                  <Input
   514→                    placeholder="Enter manufacturer name"
   515→                    value={newManufacturerName}
   516→                    onChange={(e) => setNewManufacturerName(e.target.value)}
   517→                    onKeyDown={(e) => { if (e.key === 'Enter') handleAddNewManufacturer() }}
   518→                    autoFocus
   519→                  />
   520→                  <Button type="button" size="sm" variant="outline" onClick={handleAddNewManufacturer} className="shrink-0 px-3">
   521→                    <CheckCircle2 className="h-4 w-4" />
   522→                  </Button>
   523→                  <Button type="button" size="sm" variant="ghost" onClick={() => { setShowNewManufacturer(false); setNewManufacturerName('') }} className="shrink-0 px-2">
   524→                    <Trash2 className="h-3.5 w-3.5" />
   525→                  </Button>

   867→        method: 'POST',
