     8→} from 'lucide-react'

    11→import { Badge } from '@/components/ui/badge'

   100→interface ReportData {

   102→  stockTakeRef: string

   106→  startedAt: string | null

   140→// ── Component ──
   141→
   142→export function StockTakeReportView({ stockTakeId }: { stockTakeId?: string }) {
   143→  const [report, setReport] = useState<ReportData | null>(null)
   144→  const [loading, setLoading] = useState(true)
   145→  const [error, setError] = useState<string | null>(null)
   146→  const [expandedSections, setExpandedSections] = useState<Record<string, boolean>>({
   147→    expired: true,
   148→    nearExpiry: true,
   149→    variance: true,
   150→    reorder: true,
   151→    valuation: true,
   152→  })
   153→  const [sortField, setSortField] = useState<string>('')
   154→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('asc')
   155→  const addToast = useAppStore((s) => s.addToast)
   156→
   157→  const fetchReport = useCallback(async () => {
   158→    if (!stockTakeId) return
   159→    setLoading(true)
   160→    setError(null)
   161→    try {
   162→      const res = await fetch(`/api/stock-take?action=report&id=${stockTakeId}`, { headers: authHeaders() })
   163→      if (res.ok) {
   164→        setReport(await res.json())
   165→      } else {
   166→        const err = await res.json().catch(() => ({ error: 'Failed to load report' }))
   167→        setError(err.error || 'Failed to load report')
   168→      }
   169→    } catch {
   170→      setError('Network error while loading report')
   171→    } finally {
   172→      setLoading(false)
   173→    }
   174→  }, [stockTakeId])
   175→
   176→  useEffect(() => { fetchReport() }, [fetchReport])
   177→
   178→  const toggleSection = (section: string) => {
   179→    setExpandedSections((prev) => ({ ...prev, [section]: !prev[section] }))
   180→  }
   181→
   182→  const handlePrint = () => window.print()
   183→  const handleExportCSV = () => {
   184→    if (!report) return
   185→    try {
   186→      const lines: string[] = []
   187→      // Expired goods
   188→      lines.push('=== EXPIRED GOODS ===')
   189→      lines.push('Product,NDC,Category,Strength,Form,Qty,Cost Price,Selling Price,Total Cost,Potential Revenue,Expiry Date,Days Since Expiry,Manufacturer,Vendor')
   190→      for (const item of report.expiredGoods.items) {
   191→        lines.push(`"${item.productName}","${item.ndc || ''}","${item.category}","${item.strength || ''}","${item.dosageForm || ''}",${item.countedQty},${item.costPrice},${item.sellingPrice},${item.totalCost},${item.potentialRevenue},"${item.expiryDate || ''}",${item.daysSinceExpiry},"${item.manufacturer || ''}","${item.vendor || ''}"`)
   192→      }
   193→      lines.push('')
   194→      // Near expiry
   195→      lines.push('=== NEAR EXPIRY (90 DAYS) ===')
   196→      lines.push('Product,NDC,Category,Strength,Form,Qty,Cost Price,Selling Price,Total Cost,Potential Revenue,Expiry Date,Days To Expiry,Manufacturer,Vendor')
   197→      for (const item of report.nearExpiryGoods.items) {
   198→        lines.push(`"${item.productName}","${item.ndc || ''}","${item.category}","${item.strength || ''}","${item.dosageForm || ''}",${item.countedQty},${item.costPrice},${item.sellingPrice},${item.totalCost},${item.potentialRevenue},"${item.expiryDate || ''}",${item.daysToExpiry},"${item.manufacturer || ''}","${item.vendor || ''}"`)
   199→      }
   200→      lines.push('')
   201→      // Variance
   202→      lines.push('=== STOCK VARIANCE ===')
   203→      lines.push('Product,NDC,Category,System Qty,Counted Qty,Variance,Variance %,Type,Unit Cost,Total Cost,Manufacturer,Vendor')
   204→      for (const item of report.stockVariance.items) {
   205→        lines.push(`"${item.productName}","${item.ndc || ''}","${item.category}",${item.systemQty},${item.countedQty},${item.variance},${item.variancePercent},"${item.varianceType}",${item.unitCost},${item.totalCost},"${item.manufacturer || ''}","${item.vendor || ''}"`)
   206→      }
   207→      lines.push('')
   208→      // Reorder alerts
   209→      lines.push('=== REORDER ALERTS ===')
   210→      lines.push('Product,NDC,Category,Current Qty,Reorder Point,Reorder Qty,Deficit,Unit Cost,Reorder Cost,Manufacturer,Vendor')
   211→      for (const item of report.reorderAlerts.items) {
   212→        lines.push(`"${item.productName}","${item.ndc || ''}","${item.category}",${item.countedQty},${item.reorderPoint},${item.reorderQty},${item.deficit},${item.unitCost},${item.reorderCost},"${item.manufacturer || ''}","${item.vendor || ''}"`)
   213→      }
   214→
   215→      const blob = new Blob([lines.join('\n')], { type: 'text/csv' })
   216→      const url = URL.createObjectURL(blob)
   217→      const a = document.createElement('a')
   218→      a.href = url
   219→      a.download = `stock-take-report-${report.stockTakeRef}.csv`
   220→      a.click()
   221→      URL.revokeObjectURL(url)
   222→      addToast({ title: 'Exported', description: 'Report exported as CSV', variant: 'success' })
   223→    } catch {
   224→      addToast({ title: 'Export Failed', description: 'Failed to export report', variant: 'destructive' })
   225→    }
   226→  }
   227→
   228→  // ── Loading State ──
   229→  if (loading) {
   230→    return (
   231→      <div className="space-y-4">
   232→        <Skeleton className="h-8 w-64" />
   233→        <Skeleton className="h-24 w-full" />
   234→        <Skeleton className="h-64 w-full" />
   235→        <Skeleton className="h-48 w-full" />
   236→      </div>
   237→    )
   238→  }
   239→
   240→  // ── Error State ──
   241→  if (error || !report) {
   242→    return (
   243→      <div className="text-center py-12">
   244→        <AlertCircle className="h-12 w-12 mx-auto mb-3 text-red-400" />
   245→        <p className="text-sm font-medium text-gray-500">{error || 'Report not found'}</p>
   246→        <Button variant="outline" className="mt-4" onClick={() => useAppStore.getState().setCurrentView('stock-take')}>
   247→          <ArrowLeft className="h-4 w-4 mr-1" /> Back to Stock Takes
   248→        </Button>
   249→      </div>
   250→    )
   251→  }
   252→
   253→  const accuracyRate = report.totalItemsChecked > 0
   254→    ? ((report.itemsMatched / report.totalItemsChecked) * 100).toFixed(1)
   255→    : '0'
   256→
   257→  const SectionHeader = ({
   258→    id, icon: Icon, title, badge, badgeColor, count, subtitle,
   259→  }: {
   260→    id: string; icon: React.ElementType; title: string; badge?: string; badgeColor?: string
   261→    count?: number; subtitle?: string
   262→  }) => (
   263→    <button
   264→      className="w-full flex items-center justify-between p-4 hover:bg-gray-50/50 rounded-lg transition-colors cursor-pointer"
   265→      onClick={() => toggleSection(id)}
   266→    >
   267→      <div className="flex items-center gap-3">
   268→        <div className="h-9 w-9 rounded-lg bg-emerald-100 flex items-center justify-center">
   269→          <Icon className="h-4.5 w-4.5 text-emerald-600" />
   270→        </div>
   271→        <div className="text-left">
   272→          <div className="flex items-center gap-2">
   273→            <h3 className="text-sm font-semibold">{title}</h3>
   274→            {badge !== undefined && (
   275→              <Badge className={`text-[10px] px-1.5 py-0 ${badgeColor || 'bg-gray-100 text-gray-700'}`}>{badge}</Badge>
   276→            )}
   277→          </div>
   278→          {subtitle && <p className="text-xs text-muted-foreground">{subtitle}</p>}
   279→        </div>
   280→      </div>
   281→      <div className="flex items-center gap-3">
   282→        {count !== undefined && <span className="text-xs font-medium text-muted-foreground">{count} items</span>}
   283→        {expandedSections[id] ? <ChevronUp className="h-4 w-4 text-muted-foreground" /> : <ChevronDown className="h-4 w-4 text-muted-foreground" />}
   284→      </div>
   285→    </button>
   286→  )
   287→
   288→  return (
   289→    <div className="space-y-4 print:space-y-2">
   290→      {/* ── Report Header ── */}
   291→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   292→        <div className="flex items-center gap-3">
   293→          <Button variant="ghost" size="sm" onClick={() => useAppStore.getState().setCurrentView('stock-take')} className="print:hidden">
   294→            <ArrowLeft className="h-4 w-4 mr-1" /> Back
   295→          </Button>
   296→          <div className="h-10 w-10 rounded-lg bg-emerald-100 flex items-center justify-center">
   297→            <FileText className="h-5 w-5 text-emerald-600" />
   298→          </div>
   299→          <div>
   300→            <h1 className="text-lg font-bold">Stock Take Report</h1>
   301→            <p className="text-xs text-muted-foreground">{report.stockTakeRef} · Generated {report.completedAt ? new Date(report.completedAt).toLocaleString() : new Date(report.generatedAt).toLocaleString()}</p>
   302→          </div>
   303→        </div>
   304→        <div className="flex gap-2 print:hidden">
   305→          <Button variant="outline" size="sm" onClick={handlePrint}>
   306→            <Printer className="h-3.5 w-3.5 mr-1" /> Print
   307→          </Button>
   308→          <Button variant="outline" size="sm" onClick={handleExportCSV}>
   309→            <Download className="h-3.5 w-3.5 mr-1" /> Export CSV
   310→          </Button>
   311→        </div>
   312→      </div>
   313→
   314→      {/* ── Meta Info Bar ── */}
   315→      <Card className="print:border-none print:shadow-none">
   316→        <CardContent className="p-4">
   317→          <div className="flex flex-wrap gap-x-6 gap-y-1 text-xs text-muted-foreground">
   318→            <span><strong className="text-foreground">Reference:</strong> {report.stockTakeRef}</span>
   319→            {report.countedBy && <span><strong className="text-foreground">Counted By:</strong> {report.countedBy}</span>}
   320→            <span><strong className="text-foreground">Date:</strong> {report.completedAt ? new Date(report.completedAt).toLocaleDateString() : '—'}</span>
   321→            <span><strong className="text-foreground">Items Checked:</strong> {report.totalItemsChecked}</span>
   322→            {report.notes && <span><strong className="text-foreground">Notes:</strong> {report.notes}</span>}
   323→          </div>
   324→        </CardContent>
   325→      </Card>
   326→
   327→      {/* ── Key Metrics Cards ── */}
   328→      <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-6 gap-3 print:grid-cols-3">
   329→        <Card>
   330→          <CardContent className="p-3 text-center">
   331→            <p className="text-2xl font-bold">{report.totalItemsChecked}</p>
   332→            <p className="text-[11px] text-muted-foreground">Items Checked</p>
   333→          </CardContent>
   334→        </Card>
   335→        <Card>
   336→          <CardContent className="p-3 text-center">
   337→            <p className="text-2xl font-bold text-emerald-600">{accuracyRate}%</p>
   338→            <p className="text-[11px] text-muted-foreground">Accuracy Rate</p>
   339→          </CardContent>
   340→        </Card>
   341→        <Card>
   342→          <CardContent className="p-3 text-center">
   343→            <p className="text-lg font-bold text-red-600">{report.expiredGoods.count}</p>
   344→            <p className="text-[11px] text-muted-foreground">Expired Goods</p>
   345→            <p className="text-[10px] text-red-500 font-medium">{formatCurrency(report.expiredGoods.totalCost)}</p>
   346→          </CardContent>
   347→        </Card>
   348→        <Card>
   349→          <CardContent className="p-3 text-center">
   350→            <p className="text-lg font-bold text-orange-600">{report.nearExpiryGoods.count}</p>
   351→            <p className="text-[11px] text-muted-foreground">Near Expiry</p>
   352→            <p className="text-[10px] text-orange-500 font-medium">{formatCurrency(report.nearExpiryGoods.totalCost)}</p>
   353→          </CardContent>
   354→        </Card>
   355→        <Card>
   356→          <CardContent className="p-3 text-center">
   357→            <p className="text-lg font-bold text-orange-600">{report.stockVariance.shortageCount}</p>
   358→            <p className="text-[11px] text-muted-foreground">Shortages</p>
   359→            <p className="text-[10px] text-orange-500 font-medium">{formatCurrency(report.stockVariance.shortageTotalCost)}</p>
   360→          </CardContent>
   361→        </Card>
   362→        <Card>
   363→          <CardContent className="p-3 text-center">
   364→            <p className="text-lg font-bold text-blue-600">{report.reorderAlerts.count}</p>
   365→            <p className="text-[11px] text-muted-foreground">Reorder Alerts</p>
   366→            <p className="text-[10px] text-blue-500 font-medium">{formatCurrency(report.reorderAlerts.totalReorderCost)}</p>
   367→          </CardContent>
   368→        </Card>
   369→      </div>
   370→
   371→      {/* ── Inventory Valuation Summary ── */}
   372→      <Card>
   373→        <SectionHeader
   374→          id="valuation"
   375→          icon={DollarSign}
   376→          title="Inventory Valuation Summary"
   377→          subtitle={`Total cost value: ${formatCurrency(report.inventoryValuation.totalCostValue)} · Retail value: ${formatCurrency(report.inventoryValuation.totalRetailValue)}`}
   378→        />
   379→        {expandedSections.valuation && (
   380→          <CardContent className="pt-0 px-4 pb-4">
   381→            <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   382→              <div className="rounded-lg border p-3 text-center">
   383→                <p className="text-xs text-muted-foreground">Total Cost Value</p>
   384→                <p className="text-lg font-bold">{formatCurrency(report.inventoryValuation.totalCostValue)}</p>
   385→              </div>
   386→              <div className="rounded-lg border p-3 text-center">
   387→                <p className="text-xs text-muted-foreground">Total Retail Value</p>
   388→                <p className="text-lg font-bold">{formatCurrency(report.inventoryValuation.totalRetailValue)}</p>
   389→              </div>
   390→              <div className="rounded-lg border p-3 text-center">
   391→                <p className="text-xs text-muted-foreground">Potential Profit</p>
   392→                <p className="text-lg font-bold text-emerald-600">{formatCurrency(report.inventoryValuation.potentialProfit)}</p>
   393→              </div>
   394→              <div className="rounded-lg border p-3 text-center">
   395→                <p className="text-xs text-muted-foreground">Profit Margin</p>
   396→                <p className="text-lg font-bold text-emerald-600">{report.inventoryValuation.profitMargin.toFixed(1)}%</p>
   397→              </div>
   398→            </div>
   399→          </CardContent>
   400→        )}
   401→      </Card>
   402→
   403→      {/* ── Expired Goods ── */}
   404→      <Card>
   405→        <SectionHeader
   406→          id="expired"
   407→          icon={PackageX}
   408→          title="Expired Goods"
   409→          badge={report.expiredGoods.count.toString()}
   410→          badgeColor="bg-red-100 text-red-700"
   411→          count={report.expiredGoods.count}
   412→          subtitle={`Total cost at risk: ${formatCurrency(report.expiredGoods.totalCost)} · Lost revenue: ${formatCurrency(report.expiredGoods.totalPotentialRevenue)}`}
   413→        />
   414→        {expandedSections.expired && (
   415→          <CardContent className="pt-0 px-0">
   416→            {report.expiredGoods.items.length > 0 ? (
   417→              <div className="overflow-x-auto">
   418→                <Table>
   419→                  <TableHeader>
   420→                    <TableRow className="bg-red-50/50">
   421→                      <TableHead className="text-xs">#</TableHead>
   422→                      <TableHead className="text-xs">Product</TableHead>
   423→                      <TableHead className="text-xs hidden xl:table-cell">NDC</TableHead>
   424→                      <TableHead className="text-xs hidden lg:table-cell">Category</TableHead>
   425→                      <TableHead className="text-xs hidden xl:table-cell">Strength/Form</TableHead>
   426→                      <TableHead className="text-xs hidden xl:table-cell">Manufacturer</TableHead>
   427→                      <TableHead className="text-xs hidden xl:table-cell">Vendor</TableHead>
   428→                      <TableHead className="text-xs text-right">Qty</TableHead>
   429→                      <TableHead className="text-xs text-right hidden md:table-cell">Cost Price</TableHead>
   430→                      <TableHead className="text-xs text-right hidden md:table-cell">Sell Price</TableHead>
   431→                      <TableHead className="text-xs text-right">Total Cost</TableHead>
   432→                      <TableHead className="text-xs text-right hidden lg:table-cell">Lost Revenue</TableHead>
   433→                      <TableHead className="text-xs hidden md:table-cell">Expiry Date</TableHead>
   434→                      <TableHead className="text-xs text-right hidden lg:table-cell">Days Expired</TableHead>
   435→                    </TableRow>
   436→                  </TableHeader>
   437→                  <TableBody>
   438→                    {report.expiredGoods.items.map((item, i) => (
   439→                      <TableRow key={item.productId} className="bg-red-50/20 hover:bg-red-50/40">
   440→                        <TableCell className="text-xs text-muted-foreground">{i + 1}</TableCell>
   441→                        <TableCell className="text-sm font-medium">{item.productName}</TableCell>
   442→                        <TableCell className="text-xs font-mono text-muted-foreground hidden xl:table-cell">{item.ndc || '—'}</TableCell>
   443→                        <TableCell className="text-xs text-muted-foreground hidden lg:table-cell">{item.category.replace(/_/g, ' ')}</TableCell>
   444→                        <TableCell className="text-xs text-muted-foreground hidden xl:table-cell">
   445→                          {[item.strength, item.dosageForm].filter(Boolean).join(' / ') || '—'}
   446→                        </TableCell>
   447→                        <TableCell className="text-xs text-muted-foreground hidden xl:table-cell">{item.manufacturer || '—'}</TableCell>
   448→                        <TableCell className="text-xs text-muted-foreground hidden xl:table-cell">{item.vendor || '—'}</TableCell>
   449→                        <TableCell className="text-right text-sm">{item.countedQty}</TableCell>
   450→                        <TableCell className="text-right text-xs hidden md:table-cell">{formatCurrency(item.costPrice)}</TableCell>
   451→                        <TableCell className="text-right text-xs hidden md:table-cell">{formatCurrency(item.sellingPrice)}</TableCell>
   452→                        <TableCell className="text-right text-sm font-semibold text-red-600">{formatCurrency(item.totalCost)}</TableCell>
   453→                        <TableCell className="text-right text-xs text-red-500 hidden lg:table-cell">{formatCurrency(item.potentialRevenue)}</TableCell>
   454→                        <TableCell className="text-xs text-red-600 font-medium hidden md:table-cell">{item.expiryDate ? new Date(item.expiryDate).toLocaleDateString() : '—'}</TableCell>
   455→                        <TableCell className="text-right text-xs text-red-600 font-bold hidden lg:table-cell">{item.daysSinceExpiry}</TableCell>
