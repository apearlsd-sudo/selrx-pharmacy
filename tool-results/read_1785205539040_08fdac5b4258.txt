   118→   118→  })

   131→   131→        adjustmentType: adjustType,
   132→   132→        reason: adjustReason,

   135→   135→        body.setQuantity = adj
   136→   136→        body.adjustment = 0

   138→   138→        body.adjustment = adj

   175→   175→          name: productForm.name,
   176→   176→          ndc: productForm.sku || null,
   177→   177→          category: productForm.category,

   179→   179→          costPrice: parseFloat(productForm.price) * 0.7,
   180→   180→          reorderPoint: parseInt(productForm.minStockLevel) || 10,
   181→   181→          expiryDate: productForm.expiryDate || null,
   182→   182→          batchNumber: productForm.barcode || null,

   186→   186→        const err = await res.json()

   188→   188→      }
   189→   189→      const newProduct = await res.json()
   190→   190→
   191→   191→      // Set initial stock quantity if specified
   192→   192→      const qty = parseInt(productForm.stockQuantity)
   193→   193→      if (qty > 0) {
   194→   194→        await fetch('/api/inventory', {
   195→   195→          method: 'PUT',
   196→   196→          headers: { 'Content-Type': 'application/json' },
   197→   197→          body: JSON.stringify({
   198→   198→            productId: newProduct.id,
   199→   199→            adjustment: qty,
   200→   200→            reason: 'Initial stock on product creation',
   201→   201→          }),
   202→   202→        })
   203→   203→      }
   204→   204→
   205→   205→      addToast({ title: 'Product Added', description: `${productForm.name} has been added to inventory`, variant: 'success' })
   206→   206→      setAddProductDialog(false)
   207→   207→      setProductForm({ name: '', sku: '', category: 'OTC', price: '', stockQuantity: '', minStockLevel: '10', expiryDate: '', barcode: '' })
   208→   208→      fetchInventory()
   209→   209→    } catch (err: any) {
   210→   210→      addToast({ title: 'Error', description: err.message || 'Failed to add product', variant: 'destructive' })
   211→   211→    } finally {
   212→   212→      setSavingProduct(false)
   213→   213→    }
   214→   214→  }
   215→   215→
   216→   216→  // ── Stock Count: search + set physical quantities ──────────────
   217→   217→  const filteredStockProducts = stockSearch.length > 0
   218→   218→    ? items.filter(i => i.product.name.toLowerCase().includes(stockSearch.toLowerCase()) || i.product.ndc?.toLowerCase().includes(stockSearch.toLowerCase()))
   219→   219→    : []
   220→   220→

   222→   222→    if (stockEntries.find(e => e.productId === item.productId)) return
   223→   223→    setStockEntries(prev => [...prev, {
   224→   224→      productId: item.productId,
   225→   225→      name: item.product.name,
   226→   226→      currentQty: Number(item.quantity) || 0,
   227→   227→      physicalQty: String(Number(item.quantity) || 0),
   228→   228→      unit: item.product.unitOfMeasure,
   229→   229→    }])
   230→   230→    setStockSearch('')
   231→   231→  }
   232→   232→
   233→   233→  const removeStockEntry = (productId: string) => {
   234→   234→    setStockEntries(prev => prev.filter(e => e.productId !== productId))
   235→   235→  }
   236→   236→
   237→   237→  const updateStockEntry = (productId: string, physicalQty: string) => {
   238→   238→    setStockEntries(prev => prev.map(e => e.productId === productId ? { ...e, physicalQty } : e))
   239→   239→  }
   240→   240→
   241→   241→  const handleStockCountSave = async () => {

   243→   243→    setStockSaving(true)
   244→   244→    try {

   247→   247→        const physicalQty = parseInt(entry.physicalQty)
   248→   248→        if (isNaN(physicalQty)) continue
   249→   249→        if (physicalQty === entry.currentQty) continue // no change needed
   250→   250→        await fetch('/api/inventory', {
   251→   251→          method: 'PUT',

   253→   253→          body: JSON.stringify({
   254→   254→            productId: entry.productId,
   255→   255→            adjustmentType: 'SET',
   256→   256→            setQuantity: physicalQty,
   257→   257→            adjustment: 0,
   258→   258→            reason: 'Physical stock count',
   259→   259→          }),

   261→   261→        updated++
   262→   262→      }
   263→   263→      addToast({ title: 'Stock Count Saved', description: `${updated} product${updated !== 1 ? 's' : ''} updated to physical count`, variant: 'success' })
   264→   264→      setStockCountDialog(false)
   265→   265→      setStockEntries([])
   266→   266→      fetchInventory()
   267→   267→    } catch (err: any) {
   268→   268→      addToast({ title: 'Error', description: err.message || 'Failed to save stock count', variant: 'destructive' })
   269→   269→    } finally {
   270→   270→      setStockSaving(false)

   280→   280→            <div className="h-10 w-10 rounded-lg bg-emerald-100 flex items-center justify-center">
   281→   281→              <Package className="h-5 w-5 text-emerald-600" />
   282→   282→            </div>
   283→   283→            <div>
   284→   284→              <p className="text-2xl font-bold">{items.length}</p>
   285→   285→              <p className="text-xs text-muted-foreground">Total Products</p>
   286→   286→            </div>
   287→   287→          </CardContent>
   288→   288→        </Card>
   289→   289→        <Card>
   290→   290→          <CardContent className="p-4 flex items-center gap-3">
   291→   291→            <div className="h-10 w-10 rounded-lg bg-red-100 flex items-center justify-center">
   292→   292→              <AlertTriangle className="h-5 w-5 text-red-600" />
   293→   293→            </div>

   296→   296→              <p className="text-xs text-muted-foreground">Low Stock Alerts</p>

   298→   298→          </CardContent>
   299→   299→        </Card>
   300→   300→        <Card>
   301→   301→          <CardContent className="p-4 flex items-center gap-3">
   302→   302→            <div className="h-10 w-10 rounded-lg bg-amber-100 flex items-center justify-center">
   303→   303→              <AlertTriangle className="h-5 w-5 text-amber-600" />
   304→   304→            </div>

   311→   311→        <Card>
   312→   312→          <CardContent className="p-4 flex items-center gap-3">
   313→   313→            <div className="h-10 w-10 rounded-lg bg-teal-100 flex items-center justify-center">
   314→   314→              <TrendingUp className="h-5 w-5 text-teal-600" />
   315→   315→            </div>
   316→   316→            <div>
   317→   317→              <p className="text-2xl font-bold">${totalValue.toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 2 })}</p>
   318→   318→              <p className="text-xs text-muted-foreground">Inventory Value</p>
   319→   319→            </div>
   320→   320→          </CardContent>
   321→   321→        </Card>
   322→   322→      </div>
   323→   323→
   324→   324→      {/* Filters & Actions */}

   335→   335→              />
   336→   336→            </div>

   338→   338→              <SelectTrigger className="w-full sm:w-40">
   339→   339→                <Filter className="h-4 w-4 mr-2" />
   340→   340→                <SelectValue />
   341→   341→              </SelectTrigger>
   342→   342→              <SelectContent>

   344→   344→                <SelectItem value="OTC">OTC</SelectItem>

   350→   350→              </SelectContent>
   351→   351→            </Select>
   352→   352→            <Select value={stockFilter} onValueChange={setStockFilter}>

   354→   354→                <SelectValue />

   356→   356→              <SelectContent>

   358→   358→                <SelectItem value="LOW">Low Stock</SelectItem>

   360→   360→                <SelectItem value="OK">In Stock OK</SelectItem>
   361→   361→              </SelectContent>
   362→   362→            </Select>
   363→   363→            <Button onClick={() => setAddProductDialog(true)} className="bg-teal-600 hover:bg-teal-700">
   364→   364→              <PackagePlus className="h-4 w-4 mr-2" />
   365→   365→              Add Product

   368→   368→              <ClipboardCheck className="h-4 w-4 mr-2" />
   369→   369→              Stock Count
   370→   370→            </Button>
   371→   371→          </div>
   372→   372→        </CardContent>
   373→   373→      </Card>
   374→   374→
   375→   375→      {/* Inventory Table */}
   376→   376→      <Card>
   377→   377→        <CardContent className="p-0">
   378→   378→          <Table>
   379→   379→            <TableHeader>
   380→   380→              <TableRow>
   381→   381→                <TableHead className="cursor-pointer" onClick={() => { setSortBy('name'); setSortDir(sortDir === 'asc' ? 'desc' : 'asc') }}>
   382→   382→                  <span className="flex items-center gap-1">Product <ArrowUpDown className="h-3 w-3" /></span>
   383→   383→                </TableHead>
   384→   384→                <TableHead className="hidden md:table-cell">NDC</TableHead>
   385→   385→                <TableHead className="cursor-pointer hidden sm:table-cell" onClick={() => { setSortBy('category'); setSortDir(sortDir === 'asc' ? 'desc' : 'asc') }}>
   386→   386→                  <span className="flex items-center gap-1">Category <ArrowUpDown className="h-3 w-3" /></span>
   387→   387→                </TableHead>
   388→   388→                <TableHead className="cursor-pointer text-right" onClick={() => { setSortBy('stock'); setSortDir(sortDir === 'asc' ? 'desc' : 'asc') }}>
   389→   389→                  <span className="flex items-center justify-end gap-1">In Stock <ArrowUpDown className="h-3 w-3" /></span>

   392→   392→                <TableHead className="text-right">Status</TableHead>
   393→   393→                <TableHead className="hidden lg:table-cell text-right">Cost</TableHead>
   394→   394→                <TableHead className="hidden lg:table-cell text-right">Retail</TableHead>
   395→   395→                <TableHead className="text-right">Actions</TableHead>
   396→   396→              </TableRow>
   397→   397→            </TableHeader>
   398→   398→            <TableBody>
   399→   399→              {loading ? (
   400→   400→                Array.from({ length: 8 }).map((_, i) => (
   401→   401→                  <TableRow key={i}>
   402→   402→                    {Array.from({ length: 9 }).map((_, j) => (

   405→   405→                  </TableRow>
   406→   406→                ))
   407→   407→              ) : filteredItems.length === 0 ? (
   408→   408→                <TableRow>
   409→   409→                  <TableCell colSpan={9} className="text-center py-8 text-muted-foreground">

   412→   412→                </TableRow>
   413→   413→              ) : (
   414→   414→                filteredItems.map((item) => {
   415→   415→                  const qty = Number(item.quantity) || 0
   416→   416→                  const reorder = Number(item.product.reorderPoint) || 10
   417→   417→                  const isOut = qty === 0
