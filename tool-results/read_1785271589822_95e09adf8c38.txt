    48→function formatDate(dateStr: string): string {
    49→  return new Date(dateStr).toLocaleDateString('en-US', {
    50→    month: 'short',
    51→    day: 'numeric',
    52→    year: 'numeric',
    53→    hour: '2-digit',
    54→    minute: '2-digit',
    55→  })
    56→}
    57→
    58→function formatTime(dateStr: string): string {
    59→  return new Date(dateStr).toLocaleTimeString('en-US', {
    60→    hour: '2-digit',
    61→    minute: '2-digit',
    62→  })
    63→}

    72→// Status badge color helper
    73→function statusBadge(status: string) {
    74→  switch (status) {
    75→    case 'COMPLETED':
    76→      return <Badge className="bg-emerald-100 text-emerald-700 text-xs border-emerald-200">Completed</Badge>
    77→    case 'PENDING':
    78→      return <Badge className="bg-amber-100 text-amber-700 text-xs border-amber-200">Pending</Badge>
    79→    case 'VOIDED':
    80→      return <Badge className="bg-red-100 text-red-700 text-xs border-red-200">Voided</Badge>
    81→    case 'REFUNDED':
    82→      return <Badge className="bg-gray-100 text-gray-700 text-xs border-gray-200">Refunded</Badge>
    83→    default:
    84→      return <Badge variant="outline" className="text-xs">{status}</Badge>
    85→  }
    86→}

    92→    CREDIT_CARD: 'bg-blue-100 text-blue-700 border-blue-200',

    96→    SPLIT: 'bg-indigo-100 text-indigo-700 border-indigo-200',
