   126→   126→function CategorySection() {
   127→   127→  const [categories, setCategories] = useState<Category[]>([])
   128→   128→  const [loading, setLoading] = useState(true)
   129→   129→  const [newName, setNewName] = useState('')
   130→   130→  const [newDesc, setNewDesc] = useState('')
   131→   131→  const [saving, setSaving] = useState(false)
   132→   132→  const [search, setSearch] = useState('')
   133→   133→  const addToast = useAppStore((s) => s.addToast)
   134→   134→
   135→   135→  const fetchCategories = useCallback(async () => {
   136→   136→    setLoading(true)
   137→   137→    try {
   138→   138→      const res = await fetch('/api/categories')
   139→   139→      if (res.ok) setCategories(await res.json())
   140→   140→    } catch {
   141→   141→      addToast({ title: 'Error', description: 'Failed to load categories', variant: 'destructive' })
   142→   142→    } finally {
   143→   143→      setLoading(false)
   144→   144→    }
   145→   145→  }, [addToast])
   146→   146→
   147→   147→  useEffect(() => { fetchCategories() }, [fetchCategories])
   148→   148→
   149→   149→  const handleCreate = async () => {
   150→   150→    if (!newName.trim()) return
   151→   151→    setSaving(true)
   152→   152→    try {
   153→   153→      const res = await fetch('/api/categories', {
   154→   154→        method: 'POST',
   155→   155→        headers: { 'Content-Type': 'application/json' },
   156→   156→        body: JSON.stringify({ name: newName.trim(), description: newDesc.trim() || null }),
   157→   157→      })
   158→   158→      if (!res.ok) {
   159→   159→        const err = await res.json()
   160→   160→        throw new Error(err.error || 'Failed')
   161→   161→      }
   162→   162→      addToast({ title: 'Category Created', description: `"${newName.trim()}" added successfully`, variant: 'success' })
   163→   163→      setNewName('')
   164→   164→      setNewDesc('')
   165→   165→      fetchCategories()
   166→   166→    } catch (err: any) {
   167→   167→      addToast({ title: 'Error', description: err.message, variant: 'destructive' })
   168→   168→    } finally {
   169→   169→      setSaving(false)
   170→   170→    }
   171→   171→  }
   172→   172→
   173→   173→  const handleDelete = async (cat: Category) => {
   174→   174→    try {
   175→   175→      const res = await fetch(`/api/categories/${cat.id}`, { method: 'DELETE' })
   176→   176→      if (!res.ok) {
   177→   177→        const err = await res.json()
   178→   178→        throw new Error(err.error || 'Failed')
   179→   179→      }
   180→   180→      addToast({ title: 'Deleted', description: `"${cat.name.replace(/_/g, ' ')}" removed`, variant: 'success' })
   181→   181→      fetchCategories()
   182→   182→    } catch (err: any) {
   183→   183→      addToast({ title: 'Error', description: err.message, variant: 'destructive' })
   184→   184→    }
   185→   185→  }
   186→   186→
   187→   187→  const filtered = categories.filter((c) =>
   188→   188→    c.name.toLowerCase().includes(search.toLowerCase()) ||
   189→   189→    (c.description && c.description.toLowerCase().includes(search.toLowerCase()))
   190→   190→  )
   191→   191→
   192→   192→  return (
   193→   193→    <div className="space-y-4">
   194→   194→      {/* Create Form */}
   195→   195→      <Card>
   196→   196→        <CardContent className="p-4">
   197→   197→          <h3 className="text-sm font-semibold mb-3 flex items-center gap-2">
   198→   198→            <Tags className="h-4 w-4 text-emerald-600" />
   199→   199→            Create New Category
   200→   200→          </h3>
   201→   201→          <div className="grid grid-cols-1 sm:grid-cols-12 gap-3">
   202→   202→            <div className="sm:col-span-3">
   203→   203→              <Label className="text-xs">Category Name *</Label>
   204→   204→              <Input
   205→   205→                placeholder="e.g., ANALGESICS"
   206→   206→                value={newName}
   207→   207→                onChange={(e) => setNewName(e.target.value)}
   208→   208→                className="mt-1"
   209→   209→                onKeyDown={(e) => { if (e.key === 'Enter') handleCreate() }}
   210→   210→              />
   211→   211→            </div>
   212→   212→            <div className="sm:col-span-6">
   213→   213→              <Label className="text-xs">Description</Label>
   214→   214→              <Input
   215→   215→                placeholder="Brief description of this category"
   216→   216→                value={newDesc}
   217→   217→                onChange={(e) => setNewDesc(e.target.value)}
   218→   218→                className="mt-1"
   219→   219→                onKeyDown={(e) => { if (e.key === 'Enter') handleCreate() }}
   220→   220→              />
   221→   221→            </div>
   222→   222→            <div className="sm:col-span-3 flex items-end">
   223→   223→              <Button
   224→   224→                onClick={handleCreate}
   225→   225→                disabled={!newName.trim() || saving}
   226→   226→                className="w-full bg-emerald-600 hover:bg-emerald-700"
   227→   227→              >
   228→   228→                {saving ? 'Saving...' : <><Plus className="h-4 w-4 mr-2" /> Add Category</>}
   229→   229→              </Button>
   230→   230→            </div>
   231→   231→          </div>
   232→   232→        </CardContent>
   233→   233→      </Card>
   234→   234→
   235→   235→      {/* Search */}
   236→   236→      <div className="relative max-w-sm">
   237→   237→        <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   238→   238→        <Input placeholder="Search categories..." value={search} onChange={(e) => setSearch(e.target.value)} className="pl-9" />
   239→   239→      </div>
   240→   240→
   241→   241→      {/* Categories Table */}
   242→   242→      <Card>
   243→   243→        <CardContent className="p-0">
   244→   244→          <Table>
   245→   245→            <TableHeader>
   246→   246→              <TableRow>
   247→   247→                <TableHead>Category Name</TableHead>
   248→   248→                <TableHead className="hidden sm:table-cell">Description</TableHead>
   249→   249→                <TableHead className="text-center">Products</TableHead>
   250→   250→                <TableHead className="text-center">Type</TableHead>
   251→   251→                <TableHead className="text-right">Actions</TableHead>
   252→   252→              </TableRow>
   253→   253→            </TableHeader>
   254→   254→            <TableBody>
   255→   255→              {loading ? (
   256→   256→                Array.from({ length: 5 }).map((_, i) => (
   257→   257→                  <TableRow key={i}>
   258→   258→                    {Array.from({ length: 5 }).map((_, j) => (
   259→   259→                      <TableCell key={j}><Skeleton className="h-5 w-full" /></TableCell>
   260→   260→                    ))}
   261→   261→                  </TableRow>
   262→   262→                ))
   263→   263→              ) : filtered.length === 0 ? (
   264→   264→                <TableRow>
   265→   265→                  <TableCell colSpan={5} className="text-center py-8 text-muted-foreground">No categories found</TableCell>
   266→   266→                </TableRow>
   267→   267→              ) : (
   268→   268→                filtered.map((cat) => {
   269→   269→                  const isBuiltIn = BUILT_IN_CATEGORIES.includes(cat.name)
   270→   270→                  const prodCount = cat._count?.products || 0
   271→   271→                  return (
   272→   272→                    <TableRow key={cat.id}>
   273→   273→                      <TableCell>
   274→   274→                        <Badge variant="outline" className="font-mono text-xs">{cat.name.replace(/_/g, ' ')}</Badge>
   275→   275→                      </TableCell>
   276→   276→                      <TableCell className="hidden sm:table-cell text-sm text-muted-foreground">{cat.description || '—'}</TableCell>
   277→   277→                      <TableCell className="text-center">
   278→   278→                        <Badge variant="secondary" className="text-xs">{prodCount}</Badge>
   279→   279→                      </TableCell>
   280→   280→                      <TableCell className="text-center">
   281→   281→                        {isBuiltIn ? (
   282→   282→                          <Badge className="text-[10px] bg-blue-100 text-blue-700 border-blue-200">System</Badge>
   283→   283→                        ) : (
   284→   284→                          <Badge className="text-[10px] bg-purple-100 text-purple-700 border-purple-200">Custom</Badge>
   285→   285→                        )}
   286→   286→                      </TableCell>
   287→   287→                      <TableCell className="text-right">
   288→   288→                        <Button
   289→   289→                          size="sm" variant="ghost"
   290→   290→                          className="text-red-400 hover:text-red-600 hover:bg-red-50"
   291→   291→                          onClick={() => handleDelete(cat)}
   292→   292→                          disabled={isBuiltIn || prodCount > 0}
   293→   293→                          title={isBuiltIn ? 'System category' : prodCount > 0 ? `${prodCount} products linked` : 'Delete'}
   294→   294→                        >
   295→   295→                          <Trash2 className="h-3.5 w-3.5" />
   296→   296→                        </Button>
   297→   297→                      </TableCell>
   298→   298→                    </TableRow>
   299→   299→                  )
   300→   300→                })
   301→   301→              )}
   302→   302→            </TableBody>
   303→   303→          </Table>
   304→   304→        </CardContent>
   305→   305→      </Card>
   306→   306→    </div>
   307→   307→  )
   308→   308→}

   312→   312→function DrugSection() {
   313→   313→  const [drugs, setDrugs] = useState<DrugProduct[]>([])
   314→   314→  const [categories, setCategories] = useState<Category[]>([])
   315→   315→  const [vendors, setVendors] = useState<Vendor[]>([])
   316→   316→  const [loading, setLoading] = useState(true)
   317→   317→  const [saving, setSaving] = useState(false)
   318→   318→  const [search, setSearch] = useState('')
   319→   319→  const addToast = useAppStore((s) => s.addToast)
   320→   320→
   321→   321→  const [form, setForm] = useState({
   322→   322→    name: '', sku: '', category: 'OTC', dosageForm: '', costPrice: '', sellingPrice: '',
   323→   323→    stockQuantity: '0', minStockLevel: '10', expiryDate: '', barcode: '', vendorId: '',
   324→   324→  })
   325→   325→
   326→   326→  const fetchData = useCallback(async () => {
   327→   327→    setLoading(true)
   328→   328→    try {
   329→   329→      const [prodRes, catRes, venRes] = await Promise.all([
   330→   330→        fetch('/api/products'),
   331→   331→        fetch('/api/categories'),
   332→   332→        fetch('/api/vendors'),
   333→   333→      ])
   334→   334→      if (prodRes.ok) {
   335→   335→        const data = await prodRes.json()
   336→   336→        const list = Array.isArray(data) ? data : data.products || []
   337→   337→        setDrugs(list)
   338→   338→      }
   339→   339→      if (catRes.ok) setCategories(await catRes.json())
   340→   340→      if (venRes.ok) setVendors(await venRes.json())
   341→   341→    } catch {
   342→   342→      addToast({ title: 'Error', description: 'Failed to load data', variant: 'destructive' })
   343→   343→    } finally {
   344→   344→      setLoading(false)
   345→   345→    }
   346→   346→  }, [addToast])
   347→   347→
   348→   348→  useEffect(() => { fetchData() }, [fetchData])
   349→   349→
   350→   350→  const handleCreate = async () => {
   351→   351→    if (!form.name || !form.sellingPrice) return
   352→   352→    setSaving(true)
   353→   353→    try {
   354→   354→      const res = await fetch('/api/products', {
   355→   355→        method: 'POST',
   356→   356→        headers: {
   357→   357→          'Content-Type': 'application/json',
   358→   358→          'x-user-role': 'SUPER_ADMIN',
   359→   359→        },
   360→   360→        body: JSON.stringify({
   361→   361→          name: form.name,
   362→   362→          ndc: form.sku || null,
   363→   363→          category: form.category,
   364→   364→          dosageForm: form.dosageForm || null,
   365→   365→          costPrice: form.costPrice ? parseFloat(form.costPrice) : null,
   366→   366→          sellingPrice: parseFloat(form.sellingPrice),
   367→   367→          reorderPoint: parseInt(form.minStockLevel) || 10,
   368→   368→          expiryDate: form.expiryDate || null,
   369→   369→          batchNumber: form.barcode || null,
   370→   370→          vendorId: form.vendorId || null,
   371→   371→        }),
   372→   372→      })
   373→   373→      if (!res.ok) {
   374→   374→        const err = await res.json()
   375→   375→        throw new Error(err.error || 'Failed to create product')
   376→   376→      }
   377→   377→      const newProduct = await res.json()
   378→   378→
   379→   379→      // Set initial stock
   380→   380→      const qty = parseInt(form.stockQuantity) || 0
   381→   381→      if (qty > 0) {
   382→   382→        await fetch('/api/inventory', {
   383→   383→          method: 'PUT',
   384→   384→          headers: { 'Content-Type': 'application/json' },
   385→   385→          body: JSON.stringify({ productId: newProduct.id, adjustment: qty, reason: 'Initial stock on creation' }),
   386→   386→        })
   387→   387→      }
   388→   388→
   389→   389→      addToast({ title: 'Drug Added', description: `"${form.name}" registered in inventory`, variant: 'success' })
   390→   390→      setForm({
   391→   391→        name: '', sku: '', category: 'OTC', dosageForm: '', costPrice: '', sellingPrice: '',
   392→   392→        stockQuantity: '0', minStockLevel: '10', expiryDate: '', barcode: '', vendorId: '',
   393→   393→      })
   394→   394→      fetchData()
   395→   395→    } catch (err: any) {
   396→   396→      addToast({ title: 'Error', description: err.message || 'Failed to add drug', variant: 'destructive' })
   397→   397→    } finally {
   398→   398→      setSaving(false)
   399→   399→    }
   400→   400→  }
   401→   401→
   402→   402→  const filtered = drugs.filter((d) =>
   403→   403→    d.name.toLowerCase().includes(search.toLowerCase()) ||
   404→   404→    (d.ndc && d.ndc.toLowerCase().includes(search.toLowerCase()))
   405→   405→  )
   406→   406→
   407→   407→  return (
   408→   408→    <div className="space-y-4">
   409→   409→      {/* Drug Registration Form */}
   410→   410→      <Card>
   411→   411→        <CardContent className="p-4">
   412→   412→          <h3 className="text-sm font-semibold mb-3 flex items-center gap-2">
   413→   413→            <Pill className="h-4 w-4 text-teal-600" />
   414→   414→            Register New Drug / Product
   415→   415→          </h3>
   416→   416→          <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-3 gap-3">
   417→   417→            {/* Product Name - full width */}
   418→   418→            <div className="sm:col-span-2 lg:col-span-3">
   419→   419→              <Label className="text-xs">Product Name <span className="text-red-500">*</span></Label>
   420→   420→              <Input placeholder="e.g., Paracetamol 500mg" value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} className="mt-1" />
   421→   421→            </div>
   422→   422→
   423→   423→            {/* SKU */}
   424→   424→            <div>
   425→   425→              <Label className="text-xs">SKU / NDC</Label>
   426→   426→              <Input placeholder="e.g., SKU-00123" value={form.sku} onChange={(e) => setForm({ ...form, sku: e.target.value })} className="mt-1" />
   427→   427→            </div>
   428→   428→
   429→   429→            {/* Category Dropdown */}
   430→   430→            <div>
   431→   431→              <Label className="text-xs">Category</Label>
   432→   432→              <Select value={form.category} onValueChange={(v) => setForm({ ...form, category: v })}>
   433→   433→                <SelectTrigger className="mt-1"><SelectValue /></SelectTrigger>
   434→   434→                <SelectContent>
   435→   435→                  {BUILT_IN_CATEGORIES.map((c) => (
   436→   436→                    <SelectItem key={c} value={c}>{c.replace(/_/g, ' ')}</SelectItem>
   437→   437→                  ))}
   438→   438→                  {categories.filter((c) => !BUILT_IN_CATEGORIES.includes(c.name)).map((c) => (
   439→   439→                    <SelectItem key={c.id} value={c.name}>{c.name.replace(/_/g, ' ')}</SelectItem>
   440→   440→                  ))}
   441→   441→                </SelectContent>
