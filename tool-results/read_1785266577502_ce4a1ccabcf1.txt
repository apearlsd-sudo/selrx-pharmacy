     1→'use client'
     2→
     3→import { useState, useEffect, useCallback } from 'react'
     4→import {
     5→  Package, Search, AlertTriangle, Edit, ArrowUpDown,
     6→  Download, Filter, TrendingUp, PackagePlus, ClipboardCheck, X
     7→} from 'lucide-react'
     8→import { Button } from '@/components/ui/button'
     9→import { Input } from '@/components/ui/input'
    10→import { Card, CardContent, CardHeader, CardTitle } from '@/components/ui/card'
    11→import { Badge } from '@/components/ui/badge'
    12→import { Skeleton } from '@/components/ui/skeleton'
    13→import {
    14→  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
    15→} from '@/components/ui/table'
    16→import {
    17→  Dialog, DialogContent, DialogHeader, DialogTitle, DialogDescription, DialogFooter,
    18→} from '@/components/ui/dialog'
    19→import { Label } from '@/components/ui/label'
    20→import { Textarea } from '@/components/ui/textarea'
    21→import {
    22→  Select, SelectContent, SelectItem, SelectTrigger, SelectValue,
    23→} from '@/components/ui/select'
    24→import { useAppStore } from '@/store/app-store'
    25→
    26→interface InventoryItem {
    27→  id: string
    28→  productId: string
    29→  quantity: number
    30→  lastCounted: string | null
    31→  product: {
    32→    id: string
    33→    name: string
    34→    ndc: string | null
    35→    category: string
    36→    sellingPrice: number
    37→    costPrice: number | null
    38→    status: string
    39→    reorderPoint: number
    40→    maxStock: number | null
    41→    dosageForm: string | null
    42→    strength: string | null
    43→    unitOfMeasure: string
    44→    expiryDate: string | null
    45→    batchNumber: string | null
    46→  }
    47→}
    48→
    49→export function InventoryView() {
    50→  const [items, setItems] = useState<InventoryItem[]>([])
    51→  const [loading, setLoading] = useState(true)
    52→  const [searchQuery, setSearchQuery] = useState('')
    53→  const [categoryFilter, setCategoryFilter] = useState('ALL')
    54→  const [stockFilter, setStockFilter] = useState('ALL')
    55→  const [adjustDialog, setAdjustDialog] = useState(false)
    56→  const [addProductDialog, setAddProductDialog] = useState(false)
    57→  const [categories, setCategories] = useState<{ id: string; name: string; description: string | null; _count?: { products: number } }[]>([])
    58→  const [selectedItem, setSelectedItem] = useState<InventoryItem | null>(null)
    59→  const [adjustType, setAdjustType] = useState('ADD')
    60→  const [adjustAmount, setAdjustAmount] = useState('')
    61→  const [adjustReason, setAdjustReason] = useState('')
    62→  const [adjustCostPrice, setAdjustCostPrice] = useState('')
    63→  const [adjustSellingPrice, setAdjustSellingPrice] = useState('')
    64→  const [stockCountDialog, setStockCountDialog] = useState(false)
    65→  const [stockSearch, setStockSearch] = useState('')
    66→  const [stockSearchResults, setStockSearchResults] = useState<{ id: string; name: string; ndc: string | null; unitOfMeasure: string; currentQty: number }[]>([])
    67→  const [stockSearching, setStockSearching] = useState(false)
    68→  const [stockEntries, setStockEntries] = useState<{
    69→    productId: string; name: string; ndc: string | null; currentQty: number; physicalQty: string;
    70→    unit: string; costPrice: string; sellingPrice: string
    71→  }[]>([])
    72→  const [stockSaving, setStockSaving] = useState(false)
    73→  const [sortBy, setSortBy] = useState<'name' | 'stock' | 'category'>('name')
    74→  const [sortDir, setSortDir] = useState<'asc' | 'desc'>('asc')
    75→  const [productForm, setProductForm] = useState({
    76→    name: '', sku: '', category: 'OTC', price: '', stockQuantity: '',
    77→    minStockLevel: '10', expiryDate: '', barcode: '',
    78→  })
    79→  const [savingProduct, setSavingProduct] = useState(false)
    80→  const addToast = useAppStore((s) => s.addToast)
    81→  const currentUser = useAppStore((s) => s.user)
    82→  const bumpInventoryVersion = useAppStore((s) => s.bumpInventoryVersion)
    83→
    84→  const fetchInventory = useCallback(async (forceRefresh = false) => {
    85→    setLoading(true)
    86→    try {
    87→      const params = new URLSearchParams()
    88→      if (searchQuery) params.set('search', searchQuery)
    89→      if (categoryFilter !== 'ALL') params.set('category', categoryFilter)
    90→      if (forceRefresh) params.set('_t', String(Date.now()))
    91→      const res = await fetch(`/api/inventory?${params}`)
    92→      if (res.ok) {
    93→        const data = await res.json()
    94→        setItems(data)
    95→      }
    96→    } catch {
    97→      addToast({ title: 'Error', description: 'Failed to load inventory', variant: 'destructive' })
    98→    } finally {
    99→      setLoading(false)
   100→    }
   101→  }, [searchQuery, categoryFilter, addToast])
   102→
   103→  const fetchCategories = useCallback(async () => {
   104→    try {
   105→      const res = await fetch('/api/categories')
   106→      if (res.ok) setCategories(await res.json())
   107→    } catch {
   108→      // silent — categories are optional enhancement
   109→    }
   110→  }, [])
   111→
   112→  useEffect(() => { fetchInventory(); fetchCategories() }, [fetchInventory, fetchCategories])
   113→
   114→  const filteredItems = items.filter((item) => {
   115→    const q = Number(item.quantity) || 0
   116→    const r = Number(item.product.reorderPoint) || 10
   117→    if (stockFilter === 'LOW') return q > 0 && q <= r
   118→    if (stockFilter === 'OUT') return q === 0
   119→    if (stockFilter === 'OK') return q > r
   120→    return true
   121→  }).sort((a, b) => {
   122→    if (sortBy === 'name') return sortDir === 'asc' ? a.product.name.localeCompare(b.product.name) : b.product.name.localeCompare(a.product.name)
   123→    if (sortBy === 'stock') return sortDir === 'asc' ? (Number(a.quantity)||0) - (Number(b.quantity)||0) : (Number(b.quantity)||0) - (Number(a.quantity)||0)
   124→    return sortDir === 'asc' ? a.product.category.localeCompare(b.product.category) : b.product.category.localeCompare(a.product.category)
   125→  })
   126→
   127→  const lowStockCount = items.filter((i) => { const q = Number(i.quantity) || 0; const r = Number(i.product.reorderPoint) || 10; return q > 0 && q <= r }).length
   128→  const outOfStockCount = items.filter((i) => (Number(i.quantity) || 0) === 0).length
   129→  const totalValue = items.reduce((sum, i) => sum + (Number(i.quantity) || 0) * (i.product.costPrice || i.product.sellingPrice), 0)
   130→
   131→  const handleAdjust = async () => {
   132→    if (!selectedItem || (!adjustAmount && !adjustCostPrice && !adjustSellingPrice) || !adjustReason) return
   133→    try {
   134→      const isSet = adjustType === 'SET'
   135→      const adj = adjustAmount ? (adjustType === 'ADD' ? parseInt(adjustAmount) : adjustType === 'REMOVE' ? -parseInt(adjustAmount) : parseInt(adjustAmount)) : 0
   136→      const body: Record<string, any> = {
   137→        productId: selectedItem.productId,
   138→        adjustmentType: adjustType,
   139→        reason: adjustReason,
   140→      }
   141→      if (isSet) {
   142→        body.setQuantity = adj
   143→        body.adjustment = 0
   144→      } else {
   145→        body.adjustment = adj
   146→      }
   147→      if (adjustCostPrice !== '') body.costPrice = parseFloat(adjustCostPrice)
   148→      if (adjustSellingPrice !== '') body.sellingPrice = parseFloat(adjustSellingPrice)
   149→
   150→      const res = await fetch('/api/inventory', {
   151→        method: 'PUT',
   152→        headers: { 'Content-Type': 'application/json', 'x-user-role': currentUser?.role || 'SUPER_ADMIN' },
   153→        body: JSON.stringify(body),
   154→      })
   155→      if (!res.ok) {
   156→        const err = await res.json()
   157→        throw new Error(err.error || 'Failed to adjust')
   158→      }
   159→      addToast({ title: 'Product Updated', description: `${selectedItem.product.name} adjusted successfully`, variant: 'success' })
   160→      setAdjustDialog(false)
   161→      setAdjustAmount('')
   162→      setAdjustReason('')
   163→      setAdjustCostPrice('')
   164→      setAdjustSellingPrice('')
   165→      fetchInventory(true)
   166→      bumpInventoryVersion()
   167→    } catch (err: any) {
   168→      addToast({ title: 'Error', description: err.message || 'Failed to adjust stock', variant: 'destructive' })
   169→    }
   170→  }
   171→
   172→  const handleAddProduct = async () => {
   173→    if (!productForm.name || !productForm.price) return
   174→    setSavingProduct(true)
   175→    try {
   176→      const res = await fetch('/api/products', {
   177→        method: 'POST',
   178→        headers: {
   179→          'Content-Type': 'application/json',
   180→          'x-user-role': currentUser?.role || 'SUPER_ADMIN',
   181→        },
   182→        body: JSON.stringify({
   183→          name: productForm.name,
   184→          ndc: productForm.sku || null,
   185→          category: productForm.category,
   186→          sellingPrice: parseFloat(productForm.price),
   187→          costPrice: parseFloat(productForm.price) * 0.7,
   188→          reorderPoint: parseInt(productForm.minStockLevel) || 10,
   189→          expiryDate: productForm.expiryDate || null,
   190→          batchNumber: productForm.barcode || null,
   191→        }),
   192→      })
   193→      if (!res.ok) {
   194→        const err = await res.json()
   195→        throw new Error(err.error || 'Failed to create product')
   196→      }
   197→      const newProduct = await res.json()
   198→
   199→      // Set initial stock quantity if specified
   200→      const qty = parseInt(productForm.stockQuantity)
   201→      if (qty > 0) {
   202→        await fetch('/api/inventory', {
   203→          method: 'PUT',
   204→          headers: { 'Content-Type': 'application/json' },
   205→          body: JSON.stringify({
   206→            productId: newProduct.id,
   207→            adjustment: qty,
   208→            reason: 'Initial stock on product creation',
   209→          }),
   210→        })
   211→      }
   212→
   213→      addToast({ title: 'Product Added', description: `${productForm.name} has been added to inventory`, variant: 'success' })
   214→      setAddProductDialog(false)
   215→      setProductForm({ name: '', sku: '', category: 'OTC', price: '', stockQuantity: '', minStockLevel: '10', expiryDate: '', barcode: '' })
   216→      fetchInventory()
   217→    } catch (err: any) {
   218→      addToast({ title: 'Error', description: err.message || 'Failed to add product', variant: 'destructive' })
   219→    } finally {
   220→      setSavingProduct(false)
   221→    }
   222→  }
   223→
   224→  // ── Stock Count: API-based product search + set physical quantities & prices ──
   225→  const searchStockProducts = useCallback(async (query: string) => {
   226→    if (query.length < 1) { setStockSearchResults([]); return }
   227→    setStockSearching(true)
   228→    try {
   229→      const params = new URLSearchParams({ search: query, limit: '15' })
   230→      const res = await fetch(`/api/products?${params}`)
   231→      if (res.ok) {
   232→        const data = await res.json()
   233→        const productList = data.products || data
   234→        // Merge current inventory quantities
   235→        setStockSearchResults(
   236→          (Array.isArray(productList) ? productList : []).map((p: any) => {
   237→            const inv = items.find(i => i.productId === p.id)
   238→            return {
   239→              id: p.id,
   240→              name: p.name,
   241→              ndc: p.ndc || null,
   242→              unitOfMeasure: p.unitOfMeasure || 'EA',
   243→              currentQty: inv ? (Number(inv.quantity) || 0) : (p.inventory ? (Number(p.inventory.quantity) || 0) : 0),
   244→              costPrice: p.costPrice,
   245→              sellingPrice: p.sellingPrice,
   246→            }
   247→          })
   248→        )
   249→      }
   250→    } catch { /* silent */ } finally {
   251→      setStockSearching(false)
   252→    }
   253→  }, [items])
   254→
   255→  // Debounced search
   256→  useEffect(() => {
   257→    if (!stockCountDialog) return
   258→    const timer = setTimeout(() => searchStockProducts(stockSearch), 300)
   259→    return () => clearTimeout(timer)
   260→  }, [stockSearch, stockCountDialog, searchStockProducts])
   261→
   262→  const addStockEntry = (product: { id: string; name: string; ndc: string | null; unitOfMeasure: string; currentQty: number; costPrice?: number | null; sellingPrice?: number | null }) => {
   263→    if (stockEntries.find(e => e.productId === product.id)) return
   264→    setStockEntries(prev => [...prev, {
   265→      productId: product.id,
   266→      name: product.name,
   267→      ndc: product.ndc,
   268→      currentQty: product.currentQty,
   269→      physicalQty: String(product.currentQty),
   270→      unit: product.unitOfMeasure,
   271→      costPrice: product.costPrice ? String(product.costPrice) : '',
   272→      sellingPrice: product.sellingPrice ? String(product.sellingPrice) : '',
   273→    }])
   274→    setStockSearch('')
   275→    setStockSearchResults([])
   276→  }
   277→
   278→  const removeStockEntry = (productId: string) => {
   279→    setStockEntries(prev => prev.filter(e => e.productId !== productId))
   280→  }
   281→
   282→  const updateStockEntry = (productId: string, field: 'physicalQty' | 'costPrice' | 'sellingPrice', value: string) => {
   283→    setStockEntries(prev => prev.map(e => e.productId === productId ? { ...e, [field]: value } : e))
   284→  }
   285→
   286→  const handleStockCountSave = async () => {
   287→    if (stockEntries.length === 0) return
   288→    setStockSaving(true)
   289→    try {
   290→      let updated = 0
   291→      let failed = 0
   292→      const results: { name: string; requested: number; confirmed: number }[] = []
   293→      for (const entry of stockEntries) {
   294→        const physicalQty = parseInt(entry.physicalQty)
   295→        if (isNaN(physicalQty)) continue
   296→        const body: Record<string, any> = {
   297→          productId: entry.productId,
   298→          adjustmentType: 'SET',
   299→          setQuantity: physicalQty,
   300→          adjustment: 0,
   301→          reason: 'Physical stock count',
   302→        }
   303→        // Include price updates if changed
   304→        const cp = parseFloat(entry.costPrice)
   305→        if (!isNaN(cp) && cp >= 0) body.costPrice = cp
   306→        const sp = parseFloat(entry.sellingPrice)
   307→        if (!isNaN(sp) && sp >= 0) body.sellingPrice = sp
   308→        try {
   309→          const res = await fetch('/api/inventory', {
   310→            method: 'PUT',
   311→            headers: { 'Content-Type': 'application/json', 'x-user-role': currentUser?.role || 'SUPER_ADMIN' },
   312→            body: JSON.stringify(body),
   313→          })
   314→          if (res.ok) {
   315→            const json = await res.json()
   316→            const confirmed = Number(json.newQuantity) || Number(json.inventory?.quantity) || 0
   317→            results.push({ name: entry.name, requested: physicalQty, confirmed })
   318→            if (confirmed === physicalQty) {
   319→              updated++
   320→            } else {
   321→              console.error(`Stock count mismatch for ${entry.name}: requested ${physicalQty}, DB confirmed ${confirmed}`)
   322→              failed++
   323→            }
   324→          } else {
   325→            const err = await res.json().catch(() => ({ error: 'Unknown error' }))
   326→            console.error(`Stock count failed for ${entry.name}:`, err)
   327→            failed++
   328→          }
   329→        } catch (e) {
   330→          console.error(`Stock count network error for ${entry.name}:`, e)
   331→          failed++
   332→        }
   333→      }
   334→      console.log('[Stock Count Save] Results:', results)
   335→      if (updated > 0) {
   336→        addToast({
   337→          title: 'Stock Count Saved',
   338→          description: `${updated} product${updated !== 1 ? 's' : ''} updated to physical count${failed > 0 ? ` (${failed} failed)` : ''}`,
   339→          variant: updated === stockEntries.length ? 'success' : 'default',
   340→        })
   341→      } else {
   342→        addToast({ title: 'Save Failed', description: `All ${failed} product updates failed. Check console for details.`, variant: 'destructive' })
   343→      }
   344→      setStockCountDialog(false)
   345→      setStockEntries([])
   346→      setStockSearch('')
   347→      setStockSearchResults([])
   348→      // Wait for Turso read replica to catch up, then force-refresh
   349→      await new Promise(r => setTimeout(r, 1000))
   350→      await fetchInventory(true)
   351→      // Notify all other views (POS, Dashboard, Master Data, Reports) to re-fetch
   352→      bumpInventoryVersion()
   353→    } catch (err: any) {
   354→      addToast({ title: 'Error', description: err.message || 'Failed to save stock count', variant: 'destructive' })
   355→    } finally {
   356→      setStockSaving(false)
   357→    }
   358→  }
   359→
   360→  return (
   361→    <div className="space-y-4">
   362→      {/* Stats Cards */}
   363→      <div className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-4">
   364→        <Card>
   365→          <CardContent className="p-4 flex items-center gap-3">
   366→            <div className="h-10 w-10 rounded-lg bg-emerald-100 flex items-center justify-center">
   367→              <Package className="h-5 w-5 text-emerald-600" />
   368→            </div>
   369→            <div>
   370→              <p className="text-2xl font-bold">{items.length}</p>
   371→              <p className="text-xs text-muted-foreground">Total Products</p>
   372→            </div>
   373→          </CardContent>
   374→        </Card>
   375→        <Card>
   376→          <CardContent className="p-4 flex items-center gap-3">
   377→            <div className="h-10 w-10 rounded-lg bg-red-100 flex items-center justify-center">
   378→              <AlertTriangle className="h-5 w-5 text-red-600" />
   379→            </div>
   380→            <div>
   381→              <p className="text-2xl font-bold text-red-600">{lowStockCount}</p>
   382→              <p className="text-xs text-muted-foreground">Low Stock Alerts</p>
   383→            </div>
   384→          </CardContent>
   385→        </Card>
   386→        <Card>
   387→          <CardContent className="p-4 flex items-center gap-3">
   388→            <div className="h-10 w-10 rounded-lg bg-amber-100 flex items-center justify-center">
   389→              <AlertTriangle className="h-5 w-5 text-amber-600" />
   390→            </div>
   391→            <div>
   392→              <p className="text-2xl font-bold text-amber-600">{outOfStockCount}</p>
   393→              <p className="text-xs text-muted-foreground">Out of Stock</p>
   394→            </div>
   395→          </CardContent>
   396→        </Card>
   397→        <Card>
   398→          <CardContent className="p-4 flex items-center gap-3">
   399→            <div className="h-10 w-10 rounded-lg bg-teal-100 flex items-center justify-center">
   400→              <TrendingUp className="h-5 w-5 text-teal-600" />
   401→            </div>
   402→            <div>
   403→              <p className="text-2xl font-bold">${totalValue.toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 2 })}</p>
   404→              <p className="text-xs text-muted-foreground">Inventory Value</p>
   405→            </div>
   406→          </CardContent>
   407→        </Card>
   408→      </div>
   409→
   410→      {/* Filters & Actions */}
   411→      <Card>
   412→        <CardContent className="p-4">
   413→          <div className="flex flex-col sm:flex-row gap-3">
   414→            <div className="relative flex-1">
   415→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   416→              <Input
   417→                placeholder="Search by product name or NDC..."
   418→                value={searchQuery}
   419→                onChange={(e) => setSearchQuery(e.target.value)}
   420→                className="pl-9"
   421→              />
   422→            </div>
   423→            <Select value={categoryFilter} onValueChange={setCategoryFilter}>
   424→              <SelectTrigger className="w-full sm:w-40">
   425→                <Filter className="h-4 w-4 mr-2" />
   426→                <SelectValue />
   427→              </SelectTrigger>
   428→              <SelectContent>
   429→                <SelectItem value="ALL">All Categories</SelectItem>
   430→                <SelectItem value="OTC">OTC</SelectItem>
   431→                <SelectItem value="PRESCRIPTION">Prescription</SelectItem>
   432→                <SelectItem value="SUPPLEMENT">Supplement</SelectItem>
   433→                <SelectItem value="MEDICAL_DEVICE">Medical Device</SelectItem>
   434→                <SelectItem value="PERSONAL_CARE">Personal Care</SelectItem>
   435→                <SelectItem value="CONSUMABLES">Consumables</SelectItem>
   436→              </SelectContent>
   437→            </Select>
   438→            <Select value={stockFilter} onValueChange={setStockFilter}>
   439→              <SelectTrigger className="w-full sm:w-36">
   440→                <SelectValue />
   441→              </SelectTrigger>
   442→              <SelectContent>
   443→                <SelectItem value="ALL">All Stock</SelectItem>
   444→                <SelectItem value="LOW">Low Stock</SelectItem>
   445→                <SelectItem value="OUT">Out of Stock</SelectItem>
   446→                <SelectItem value="OK">In Stock OK</SelectItem>
   447→              </SelectContent>
   448→            </Select>
   449→            <Button onClick={() => setAddProductDialog(true)} className="bg-teal-600 hover:bg-teal-700">
   450→              <PackagePlus className="h-4 w-4 mr-2" />
   451→              Add Product
   452→            </Button>
   453→            <Button onClick={() => setStockCountDialog(true)} className="bg-indigo-600 hover:bg-indigo-700">
   454→              <ClipboardCheck className="h-4 w-4 mr-2" />
   455→              Stock Count
   456→            </Button>
   457→          </div>
   458→        </CardContent>
   459→      </Card>
   460→
   461→      {/* Inventory Table */}
   462→      <Card>
   463→        <CardContent className="p-0">
   464→          <Table>
   465→            <TableHeader>
   466→              <TableRow>
   467→                <TableHead className="cursor-pointer" onClick={() => { setSortBy('name'); setSortDir(sortDir === 'asc' ? 'desc' : 'asc') }}>
   468→                  <span className="flex items-center gap-1">Product <ArrowUpDown className="h-3 w-3" /></span>
   469→                </TableHead>
   470→                <TableHead className="hidden md:table-cell">NDC</TableHead>
   471→                <TableHead className="cursor-pointer hidden sm:table-cell" onClick={() => { setSortBy('category'); setSortDir(sortDir === 'asc' ? 'desc' : 'asc') }}>
   472→                  <span className="flex items-center gap-1">Category <ArrowUpDown className="h-3 w-3" /></span>
   473→                </TableHead>
   474→                <TableHead className="cursor-pointer text-right" onClick={() => { setSortBy('stock'); setSortDir(sortDir === 'asc' ? 'desc' : 'asc') }}>
   475→                  <span className="flex items-center justify-end gap-1">In Stock <ArrowUpDown className="h-3 w-3" /></span>
   476→                </TableHead>
   477→                <TableHead className="hidden lg:table-cell text-right">Reorder Point</TableHead>
   478→                <TableHead className="text-right">Status</TableHead>
   479→                <TableHead className="hidden lg:table-cell text-right">Cost</TableHead>
   480→                <TableHead className="hidden lg:table-cell text-right">Retail</TableHead>
   481→                <TableHead className="text-right">Actions</TableHead>
   482→              </TableRow>
   483→            </TableHeader>
   484→            <TableBody>
   485→              {loading ? (
   486→                Array.from({ length: 8 }).map((_, i) => (
   487→                  <TableRow key={i}>
   488→                    {Array.from({ length: 9 }).map((_, j) => (
   489→                      <TableCell key={j}><Skeleton className="h-5 w-full" /></TableCell>
   490→                    ))}
   491→                  </TableRow>
   492→                ))
   493→              ) : filteredItems.length === 0 ? (
   494→                <TableRow>

   906→          <DialogFooter className="mt-4">
